     2→// Conventions:
     3→//  - snake_case DB columns via @map, camelCase in TS

     6→//  - Lists (timeframes, sessions, checklist results) are JSON-encoded strings
     7→//    (SQLite has no scalar lists / native JSON type in this Prisma version)

   324→  setupType       String? @map("setup_type")

   326→
   327→  // Entry / exit (stored UTC, displayed in user timezone)

   331→  exitPrice         Decimal?  @map("exit_price")
   332→  stopLoss          Decimal?  @map("stop_loss")
   333→  takeProfit        Decimal?  @map("take_profit")

   337→  maePrice          Decimal?  @map("mae_price") // max adverse excursion (price extreme)
   338→
   339→  // Size & risk
   340→  lots                  Decimal  @default(0)
   341→  units                 Decimal? // = lots * contractSize
   342→  riskAmount            Decimal? @map("risk_amount")
   343→  riskPercent           Decimal? @map("risk_percent")

   345→
   346→  // Results
   347→  pips        Decimal?
   348→  points      Decimal?
   349→  grossPnl    Decimal? @map("gross_pnl")
   350→  commission  Decimal  @default(0) // signed cost: positive = cost, negative = credit
   351→  swap        Decimal  @default(0) // signed cost
   352→  fees        Decimal  @default(0) // signed cost
   353→  netPnl      Decimal? @map("net_pnl")
   354→  pnlCurrency String   @default("USD") @map("pnl_currency")
   355→  rMultiple   Decimal? @map("r_multiple")
   356→  plannedRR   Decimal? @map("planned_rr")
   357→  realizedRR  Decimal? @map("realized_rr")
   358→  resultType  String?  @map("result_type") // WIN | LOSS | BREAKEVEN
   359→  pnlSource   String   @default("CALCULATED") @map("pnl_source") // CALCULATED | MANUAL
   360→  exitReason  String?  @map("exit_reason") // TP_HIT | SL_HIT | MANUAL | TRAILING | TIME | BREAKEVEN_STOP | PARTIAL | OTHER
   361→
   362→  // Psychology & quality
   363→  emotionBefore      String?  @map("emotion_before")
   364→  emotionDuring      String?  @map("emotion_during")
   365→  emotionAfter       String?  @map("emotion_after")
   366→  confidence         Int? // 1-5
   367→  setupQualityRating Int?     @map("setup_quality_rating") // 1-5
   368→  executionRating    Int?     @map("execution_rating") // 1-5
   369→  followedPlan       Boolean? @map("followed_plan")
   370→  lessonLearned      String?  @map("lesson_learned")
   371→
   372→  // Review workflow (addendum §11.3)
   373→  reviewStatus String  @default("NOT_REVIEWED") @map("review_status") // NOT_REVIEWED | REVIEWED
