   242→async function resolveAccount(userId: string, accountId: string): Promise<Account> {
   243→  const account = await db.account.findFirst({ where: { id: accountId, userId } });
   244→  if (!account) throw new TradeValidationError("ACCOUNT_NOT_FOUND", 422);
   245→  return account;

   248→async function resolveInstrument(userId: string, instrumentId: string): Promise<Instrument> {
   249→  const instrument = await db.instrument.findFirst({
   250→    where: { id: instrumentId, OR: [{ userId: null }, { userId }] },
   251→  });
   252→  if (!instrument) throw new TradeValidationError("INSTRUMENT_NOT_FOUND", 422);
   253→  return instrument;
   254→}

   257→  if (!strategyId) return null;
   258→  const strategy = await db.strategy.findFirst({ where: { id: strategyId, userId } });
   259→  if (!strategy) throw new TradeValidationError("STRATEGY_NOT_FOUND", 422);
   260→  return strategy;

   272→/** Balance of an account at a point in time: initial + cash flows + closed P&L. */
   273→export async function computeBalanceAt(
   274→  userId: string,
   275→  accountId: string,
   276→  at: Date
   277→): Promise<Decimal> {
   278→  const account = await resolveAccount(userId, accountId);
   279→  let balance = toD(account.initialBalance);
   280→  const cash = await db.cashTransaction.aggregate({
   281→    _sum: { amount: true },
   282→    where: { accountId, date: { lt: at } },
   283→  });
   284→  balance = balance.plus(toD(cash._sum.amount ?? 0));
   285→  const pnl = await db.trade.aggregate({
   286→    _sum: { netPnl: true },
   287→    where: {
   288→      userId,
   289→      accountId,
   290→      status: "CLOSED",
   291→      isDraft: false,
   292→      exitTime: { lt: at },
   293→    },
   294→  });
   295→  balance = balance.plus(toD(pnl._sum.netPnl ?? 0));
   296→  return balance;
   297→}

   301→export interface ComputeParams {
   302→  instrument: Instrument;
   303→  account: Account;
   304→  direction: string;
   305→  entryPrice: string;
   306→  exitPrice: string | null;
   307→  stopLoss: string | null;
   308→  takeProfit: string | null;
   309→  lots: string;
   310→  commission: string;
   311→  swap: string;
   312→  fees: string;
   313→  quoteToAccountRate: string | null | undefined;
   314→  pnlSource: string;
   315→  manualNetPnl: string | null | undefined;
   316→  entryTime: Date;
   317→  balanceAtEntry: Decimal | null;
   318→}
   319→
   320→export function computeDerived(p: ComputeParams) {
   321→  const spec = toInstrumentSpec(p.instrument);
   322→  const direction = p.direction as "LONG" | "SHORT";
   323→  const closed = p.exitPrice !== null && p.exitPrice !== undefined;
   324→
   325→  let preview: TradePreview | null = null;
   326→  if (p.exitPrice != null) {
   327→    preview = previewTrade({
   328→      instrument: spec,
   329→      direction,
   330→      lots: p.lots,
   331→      entry: p.entryPrice,
   332→      exit: p.exitPrice,
   333→      commission: p.commission,
   334→      swap: p.swap,
   335→      fees: p.fees,
   336→      quoteToAccountRate: p.quoteToAccountRate ?? null,
   337→      accountCurrency: p.account.currency,
   338→      stopLoss: p.stopLoss,
   339→      takeProfit: p.takeProfit,
   340→      accountBalance: p.balanceAtEntry,
   341→    });
   342→  } else {
   343→    // Open trade: compute risk figures only (no exit).
   344→    preview = null;
   345→  }
   346→
   347→  // R multiple from prices even without exit (planned) is NOT stored —
   348→  // stored rMultiple is the realized one (closed only).
   349→  let riskAmount: Decimal | null = null;
   350→  let riskPercent: Decimal | null = null;
   351→  const stopDist =
   352→    p.stopLoss != null
   353→      ? direction === "LONG"
   354→        ? toD(p.entryPrice).minus(toD(p.stopLoss))
   355→        : toD(p.stopLoss).minus(toD(p.entryPrice))
   356→      : null;
   357→  if (stopDist && stopDist.gt(0)) {
   358→    riskAmount = stopDist
   359→      .times(toD(p.lots))
   360→      .times(toD(p.instrument.contractSize))
   361→      .times(p.quoteToAccountRate ? toD(p.quoteToAccountRate) : toD(1));
   362→    if (p.balanceAtEntry && p.balanceAtEntry.gt(0)) {
   363→      // 6 dp: SQLite NUMERIC affinity truncates longer decimals to float64
   364→      // on storage (e.g. "0.21983288304231124") — round to keep values exact.
   365→      riskPercent = riskAmount
   366→        .div(p.balanceAtEntry)
   367→        .times(100)
   368→        .toDecimalPlaces(6);
   369→    }
   370→  }
   371→
   372→  // Planned RR
   373→  let plannedRR: Decimal | null = null;
   374→  if (stopDist && stopDist.gt(0) && p.takeProfit != null) {
   375→    const reward = toD(p.takeProfit).minus(toD(p.entryPrice)).abs();
   376→    if (reward.gt(0)) plannedRR = reward.div(stopDist).toDecimalPlaces(6);
   377→  }
   378→
   379→  const netPnl =
   380→    p.pnlSource !== "CALCULATED" && p.manualNetPnl != null
   381→      ? toD(p.manualNetPnl)
   382→      : preview
   383→        ? preview.netPnl
   384→        : null;
   385→
   386→  const rMultiple =
   387→    preview && riskAmount && riskAmount.gt(0)
   388→      ? preview.netPnl.div(riskAmount).toDecimalPlaces(6)
   389→      : null;
   390→
   391→  const resultType = closed
   392→    ? classifyResult(netPnl ?? toD(0), rMultiple)
   393→    : null;
   394→
   395→  // Units
   396→  const units = toD(p.lots).times(toD(p.instrument.contractSize));
   397→
   398→  return {
   399→    session: detectSessionSafe(p.entryTime),
   400→    pips: preview ? preview.pips : null,
   401→    points: preview ? preview.points : null,
   402→    grossPnl: preview ? preview.grossPnl : null,
   403→    netPnl,
   404→    rMultiple,
   405→    plannedRR,
   406→    realizedRR: rMultiple,
   407→    resultType,
   408→    riskAmount,
   409→    riskPercent,
   410→    units,
   411→    pnlCurrency: p.account.currency,
   412→  };
   413→}

   417→/** Filter → Prisma where, shared by the list endpoint and the CSV export. */
   418→export function tradeListWhere(
   419→  userId: string,
   420→  query: TradeListQuery
   421→): Prisma.TradeWhereInput {
   422→  return {
   423→    userId,
   424→    ...(query.includeDrafts ? {} : { isDraft: false }),
   425→    ...(query.status ? { status: query.status } : {}),
   426→    ...(query.direction ? { direction: query.direction } : {}),
   427→    ...(query.resultType ? { resultType: query.resultType } : {}),
   428→    ...(query.reviewStatus ? { reviewStatus: query.reviewStatus } : {}),
   429→    ...(query.session ? { session: query.session } : {}),
   430→    ...(query.timeframe ? { timeframe: query.timeframe } : {}),
   431→    ...(query.accountId ? { accountId: query.accountId } : {}),
   432→    ...(query.strategyId ? { strategyId: query.strategyId } : {}),
   433→    ...(query.instrumentId ? { instrumentId: query.instrumentId } : {}),
   434→    ...(query.symbol ? { instrument: { symbol: query.symbol } } : {}),
   435→    ...(query.tagId ? { tags: { some: { tagId: query.tagId } } } : {}),
   436→    ...(query.emotion ? { emotionBefore: query.emotion } : {}),
   437→    ...(query.dateFrom || query.dateTo
   438→      ? {
   439→          entryTime: {
   440→            ...(query.dateFrom ? { gte: new Date(query.dateFrom) } : {}),
   441→            ...(query.dateTo ? { lte: new Date(query.dateTo) } : {}),
   442→          },
   443→        }
   444→      : {}),
   445→    ...(query.search
   446→      ? {
   447→          OR: [
   448→            { notes: { contains: query.search } },
   449→            { preTradeAnalysis: { contains: query.search } },
   450→            { postTradeReview: { contains: query.search } },
   451→            { setupType: { contains: query.search } },
   452→            { instrument: { symbol: { contains: query.search } } },
   453→          ],
   454→        }
   455→      : {}),
   456→  };
   457→}

   459→export async function listTrades(
   460→  userId: string,
   461→  query: TradeListQuery
   462→): Promise<PaginatedDTO<TradeListItemDTO>> {
   463→  const where = tradeListWhere(userId, query);
   464→
   465→  // v2.5 M2 drill-down: local-hour / local-weekday filters are timezone-aware
   466→  // and computed in JS (SQLite has no TZ support) — bounded fetch, filter,
   467→  // then paginate in memory. Only used by chart drill-downs.
   468→  if (query.entryHour != null || query.entryWeekday != null) {
   469→    const profile = await db.profile.findUnique({ where: { id: userId } });
   470→    const timezone = profile?.timezone ?? "UTC";
   471→    const rows = await db.trade.findMany({
   472→      where,
   473→      include: {
   474→        instrument: true,
   475→        account: true,
   476→        strategy: true,
   477→        tags: { include: { tag: true } },
   478→      },
   479→      orderBy: { entryTime: "desc" },
   480→      take: 5000,
   481→    });
   482→    const filtered = rows.filter((t) => {
   483→      try {
   484→        const parts = new Intl.DateTimeFormat("en-US", {
   485→          timeZone: timezone,
   486→          hour: "2-digit",
   487→          hourCycle: "h23",
   488→          weekday: "short",
   489→        }).formatToParts(t.entryTime);
   490→        const hour = parts.find((p) => p.type === "hour")?.value;
   491→        const weekday = parts.find((p) => p.type === "weekday")?.value;
   492→        const hourNum = hour != null ? Number.parseInt(hour, 10) : null;
   493→        // Intl weekday "short" en-US: Sun..Sat → ISO 1-7 (Mon..Sun)
   494→        const isoWeekday = weekday
   495→          ? ["Mon", "Tue", "Wed", "Thu", "Fri", "Sat", "Sun"].indexOf(weekday) + 1
   496→          : null;
   497→        if (query.entryHour != null && hourNum !== query.entryHour) return false;
   498→        if (query.entryWeekday != null && isoWeekday !== query.entryWeekday) {
   499→          return false;
   500→        }
   501→        return true;
   502→      } catch {
   503→        return false;
   504→      }
   505→    });
   506→    const total = filtered.length;
   507→    const pageRows = filtered.slice(
   508→      (query.page - 1) * query.pageSize,
   509→      query.page * query.pageSize
   510→    );
   511→    return {
   512→      data: pageRows.map(baseItem),
   513→      page: query.page,
   514→      pageSize: query.pageSize,
   515→      total,
   516→      totalPages: Math.max(1, Math.ceil(total / query.pageSize)),
   517→    };
   518→  }
   519→
   520→  const [total, rows] = await Promise.all([
   521→    db.trade.count({ where }),
   522→    db.trade.findMany({
   523→      where,
   524→      include: {
   525→        instrument: true,
   526→        account: true,
   527→        strategy: true,
   528→        tags: { include: { tag: true } },
   529→      },
   530→      orderBy: { [query.sort]: query.order },
   531→      skip: (query.page - 1) * query.pageSize,
   532→      take: query.pageSize,
   533→    }),
   534→  ]);
   535→
   536→  return {
   537→    data: rows.map(baseItem),
   538→    page: query.page,
   539→    pageSize: query.pageSize,
   540→    total,
   541→    totalPages: Math.max(1, Math.ceil(total / query.pageSize)),
   542→  };
   543→}

   564→export async function createTrade(
   565→  userId: string,
   566→  input: TradeInput
   567→): Promise<CreateTradeResult> {
   568→  // Idempotency: replaying the same mutation returns the stored trade.
   569→  if (input.clientMutationId) {
   570→    const existing = await db.trade.findFirst({
   571→      where: { userId, clientMutationId: input.clientMutationId },
   572→      include: DETAIL_INCLUDE,
   573→    });
   574→    if (existing) return { trade: toTradeDetail(existing), deduplicated: true };
   575→  }
   576→
   577→  const [account, instrument, strategy] = await Promise.all([
   578→    resolveAccount(userId, input.accountId),
   579→    resolveInstrument(userId, input.instrumentId),
   580→    resolveStrategy(userId, input.strategyId ?? null),
   581→  ]);
   582→  const tagIds = await filterOwnTags(userId, input.tagIds);
   583→
   584→  const entryTime = new Date(input.entryTime);
   585→  if (Number.isNaN(entryTime.getTime())) {
   586→    throw new TradeValidationError("INVALID_DATE");
   587→  }
   588→  const closed = input.status === "CLOSED";
   589→  if (closed && !input.exitPrice) {
   590→    throw new TradeValidationError("EXIT_PRICE_REQUIRED");
   591→  }
   592→  const exitTime = input.exitTime ? new Date(input.exitTime) : closed ? new Date() : null;
   593→  if (closed && (!exitTime || Number.isNaN(exitTime.getTime()))) {
   594→    throw new TradeValidationError("INVALID_DATE");
   595→  }
   596→
   597→  const balanceAtEntry = await computeBalanceAt(userId, input.accountId, entryTime);
   598→
   599→  const derived = computeDerived({
   600→    instrument,
   601→    account,
   602→    direction: input.direction,
   603→    entryPrice: input.entryPrice,
   604→    exitPrice: closed ? (input.exitPrice ?? null) : null,
   605→    stopLoss: input.stopLoss ?? null,
   606→    takeProfit: input.takeProfit ?? null,
   607→    lots: input.lots,
   608→    commission: input.commission,
   609→    swap: input.swap,
   610→    fees: input.fees,
   611→    quoteToAccountRate: input.quoteToAccountRate ?? null,
   612→    pnlSource: input.pnlSource,
   613→    manualNetPnl: input.manualNetPnl ?? null,
   614→    entryTime,
   615→    balanceAtEntry,
   616→  });
   617→
   618→  const created = await db.$transaction(async (tx) => {
   619→    const trade = await tx.trade.create({
   620→      data: {
   621→        userId,
   622→        accountId: input.accountId,
   623→        strategyId: strategy?.id ?? null,
   624→        instrumentId: input.instrumentId,
   625→        externalId: null,
   626→        direction: input.direction,
   627→        orderType: input.orderType,
   628→        status: input.status,
   629→        timeframe: input.timeframe ?? null,
   630→        session: derived.session,
   631→        setupType: input.setupType ?? null,
   632→        marketCondition: input.marketCondition ?? null,
   633→        entryTime,
   634→        exitTime,
   635→        entryPrice: input.entryPrice,
   636→        exitPrice: closed ? (input.exitPrice ?? null) : null,
   637→        stopLoss: input.stopLoss ?? null,
   638→        takeProfit: input.takeProfit ?? null,
   639→        initialStopLoss: input.stopLoss ?? null,
   640→        initialTakeProfit: input.takeProfit ?? null,
   641→        mfePrice: input.mfePrice ?? null,
   642→        maePrice: input.maePrice ?? null,
   643→        lots: input.lots,
   644→        units: derived.units,
   645→        riskAmount: derived.riskAmount,
   646→        riskPercent: derived.riskPercent,
   647→        plannedRiskPercent: input.plannedRiskPercent ?? null,
   648→        accountBalanceAtEntry: balanceAtEntry,
   649→        pips: derived.pips,
   650→        points: derived.points,
   651→        grossPnl: derived.grossPnl,
   652→        commission: input.commission,
   653→        swap: input.swap,
   654→        fees: input.fees,
   655→        netPnl: derived.netPnl,
   656→        pnlCurrency: derived.pnlCurrency,
   657→        rMultiple: derived.rMultiple,
   658→        plannedRR: derived.plannedRR,
   659→        realizedRR: derived.realizedRR,
   660→        resultType: derived.resultType,
   661→        pnlSource: input.pnlSource,
   662→        exitReason: input.exitReason ?? null,
   663→        emotionBefore: input.emotionBefore ?? null,
   664→        emotionDuring: input.emotionDuring ?? null,
   665→        emotionAfter: input.emotionAfter ?? null,
   666→        confidence: input.confidence ?? null,
   667→        setupQualityRating: input.setupQualityRating ?? null,
   668→        executionRating: input.executionRating ?? null,
   669→        followedPlan: input.followedPlan ?? null,
   670→        reviewStatus: input.reviewStatus,
   671→        isDraft: input.isDraft,
   672→        preTradeAnalysis: input.preTradeAnalysis,
   673→        postTradeReview: input.postTradeReview,
   674→        notes: input.notes ?? null,
   675→        tradingViewUrl: input.tradingViewUrl || null,
   676→        checklistResults: input.checklistResults
   677→          ? JSON.stringify(input.checklistResults)
   678→          : null,
   679→        clientMutationId: input.clientMutationId ?? null,
   680→        tags: {
   681→          create: tagIds.map((tagId) => ({ tagId })),
   682→        },
   683→      },
   684→    });
   685→    // Initial execution row (aggregateExecutions-compatible).
   686→    await tx.tradeExecution.create({
   687→      data: {
   688→        tradeId: trade.id,
   689→        type: "ENTRY",
   690→        time: entryTime,
   691→        price: input.entryPrice,
   692→        lots: input.lots,
   693→        commission: input.commission,
   694→      },
   695→    });
   696→    if (closed && exitTime) {
   697→      await tx.tradeExecution.create({
   698→        data: {
   699→          tradeId: trade.id,
   700→          type: "FINAL_EXIT",
   701→          time: exitTime,
   702→          price: input.exitPrice ?? input.entryPrice,
   703→          lots: input.lots,
   704→          commission: 0,
   705→          pnl: derived.netPnl,
   706→        },
   707→      });
   708→    }
   709→    return trade;
   710→  });
   711→
   712→  const detail = await getTradeDetail(userId, created.id);
   713→  if (!detail) throw new TradeValidationError("INTERNAL");
   714→  // Phase 6 — rule violation detection (best-effort, never fails the write).
   715→  await evaluateTradeViolations(userId, created.id);
   716→  return { trade: detail, deduplicated: false };
   717→}

   719→// ── Update (last-write-wins with conflict report — v2 §4.2) ─────────────────

   721→export interface UpdateTradeResult {
   722→  trade: TradeDetailDTO;
   723→  /** Set when the client edited a stale version (LWW applied). */
   724→  conflict: { baseUpdatedAt: string; serverUpdatedAt: string } | null;
   725→}
   726→
   727→export async function updateTrade(
   728→  userId: string,
   729→  id: string,
   730→  patch: TradePatch
   731→): Promise<UpdateTradeResult | null> {
   732→  const current = await db.trade.findFirst({
   733→    where: { id, userId },
   734→    include: { instrument: true, account: true },
   735→  });
   736→  if (!current) return null;
   737→
   738→  const conflict =
   739→    patch.expectedUpdatedAt &&
   740→    new Date(patch.expectedUpdatedAt).getTime() !== current.updatedAt.getTime()
   741→      ? {
   742→          baseUpdatedAt: patch.expectedUpdatedAt,
   743→          serverUpdatedAt: current.updatedAt.toISOString(),
   744→        }
   745→      : null;
   746→
   747→  // Merge patch onto current values.
   748→  const direction = (patch.direction ?? current.direction) as "LONG" | "SHORT";
   749→  const status = patch.status ?? current.status;
   750→  const entryPrice = patch.entryPrice ?? current.entryPrice.toString();
   751→  const exitPrice =
   752→    patch.exitPrice !== undefined ? patch.exitPrice : current.exitPrice?.toString() ?? null;
   753→  const stopLoss =
   754→    patch.stopLoss !== undefined ? patch.stopLoss : current.stopLoss?.toString() ?? null;
   755→  const takeProfit =
   756→    patch.takeProfit !== undefined
   757→      ? patch.takeProfit
   758→      : current.takeProfit?.toString() ?? null;
   759→  const lots = patch.lots ?? current.lots.toString();
   760→  const commission = patch.commission ?? current.commission.toString();
   761→  const swap = patch.swap ?? current.swap.toString();
   762→  const fees = patch.fees ?? current.fees.toString();
   763→  const pnlSource = patch.pnlSource ?? current.pnlSource;
   764→  const manualNetPnl =
   765→    patch.manualNetPnl !== undefined
   766→      ? patch.manualNetPnl
   767→      : current.netPnl && current.pnlSource === "MANUAL"
   768→        ? current.netPnl.toString()
   769→        : null;
   770→  const entryTime = patch.entryTime ? new Date(patch.entryTime) : current.entryTime;
   771→  const exitTime =
   772→    patch.exitTime !== undefined
   773→      ? patch.exitTime
   774→        ? new Date(patch.exitTime)
   775→        : null
   776→      : current.exitTime;
   777→
   778→  if (status === "CLOSED" && !exitPrice) {
   779→    throw new TradeValidationError("EXIT_PRICE_REQUIRED");
   780→  }
   781→
   782→  // Ownership of any changed relation.
   783→  if (patch.accountId && patch.accountId !== current.accountId) {
   784→    await resolveAccount(userId, patch.accountId);
   785→  }
   786→  if (patch.strategyId !== undefined) {
   787→    await resolveStrategy(userId, patch.strategyId ?? null);
   788→  }
   789→  if (patch.instrumentId && patch.instrumentId !== current.instrumentId) {
   790→    await resolveInstrument(userId, patch.instrumentId);
   791→  }
   792→  const tagIds =
   793→    patch.tagIds !== undefined ? await filterOwnTags(userId, patch.tagIds) : null;
   794→
   795→  const closed = status === "CLOSED" && exitPrice != null;
   796→  const balanceAtEntry = patch.accountId
   797→    ? await computeBalanceAt(userId, patch.accountId, entryTime)
   798→    : current.accountBalanceAtEntry
   799→      ? new Decimal(current.accountBalanceAtEntry)
   800→      : await computeBalanceAt(userId, current.accountId, entryTime);
   801→
   802→  const account = patch.accountId
   803→    ? await resolveAccount(userId, patch.accountId)
   804→    : current.account;
   805→  const instrument = patch.instrumentId
   806→    ? await resolveInstrument(userId, patch.instrumentId)
   807→    : current.instrument;
   808→
   809→  const derived = computeDerived({
   810→    instrument,
   811→    account,
   812→    direction,
   813→    entryPrice,
   814→    exitPrice: closed ? exitPrice : null,
   815→    stopLoss,
   816→    takeProfit,
   817→    lots,
   818→    commission,
   819→    swap,
   820→    fees,
   821→    quoteToAccountRate: patch.quoteToAccountRate ?? null,
   822→    pnlSource,
   823→    manualNetPnl,
   824→    entryTime,
   825→    balanceAtEntry,
   826→  });
   827→
   828→  // Initial SL/TP: never rewritten once set (moved-stop detection, v2 §4.9).
   829→  const initialStopLoss =
   830→    current.initialStopLoss?.toString() ?? (stopLoss ?? null);
   831→  const initialTakeProfit =
   832→    current.initialTakeProfit?.toString() ?? (takeProfit ?? null);
   833→
   834→  await db.$transaction(async (tx) => {
   835→    await tx.trade.update({
   836→      where: { id: current.id },
   837→      data: {
   838→        accountId: patch.accountId ?? undefined,
   839→        strategyId: patch.strategyId !== undefined ? patch.strategyId : undefined,
   840→        instrumentId: patch.instrumentId ?? undefined,
   841→        direction,
   842→        orderType: patch.orderType ?? undefined,
   843→        status,
   844→        timeframe: patch.timeframe !== undefined ? patch.timeframe : undefined,
   845→        session: derived.session ?? undefined,
   846→        setupType: patch.setupType !== undefined ? patch.setupType : undefined,
   847→        marketCondition:
   848→          patch.marketCondition !== undefined ? patch.marketCondition : undefined,
   849→        entryTime,
   850→        exitTime: closed ? (exitTime ?? new Date()) : null,
   851→        entryPrice,
   852→        exitPrice: closed ? exitPrice : null,
   853→        stopLoss,
   854→        takeProfit,
   855→        initialStopLoss,
   856→        initialTakeProfit,
   857→        mfePrice: patch.mfePrice !== undefined ? patch.mfePrice : undefined,
   858→        maePrice: patch.maePrice !== undefined ? patch.maePrice : undefined,
   859→        lots,
   860→        units: derived.units,
   861→        riskAmount: derived.riskAmount,
   862→        riskPercent: derived.riskPercent,
   863→        plannedRiskPercent:
   864→          patch.plannedRiskPercent !== undefined ? patch.plannedRiskPercent : undefined,
   865→        accountBalanceAtEntry: balanceAtEntry,
   866→        pips: derived.pips,
   867→        points: derived.points,
   868→        grossPnl: derived.grossPnl,
   869→        commission,
   870→        swap,
   871→        fees,
   872→        netPnl: derived.netPnl,
   873→        pnlCurrency: derived.pnlCurrency,
   874→        rMultiple: derived.rMultiple,
   875→        plannedRR: derived.plannedRR,
   876→        realizedRR: derived.realizedRR,
   877→        resultType: derived.resultType,
   878→        pnlSource,
   879→        exitReason: patch.exitReason !== undefined ? patch.exitReason : undefined,
   880→        emotionBefore: patch.emotionBefore !== undefined ? patch.emotionBefore : undefined,
   881→        emotionDuring: patch.emotionDuring !== undefined ? patch.emotionDuring : undefined,
   882→        emotionAfter: patch.emotionAfter !== undefined ? patch.emotionAfter : undefined,
   883→        confidence: patch.confidence !== undefined ? patch.confidence : undefined,
   884→        setupQualityRating:
   885→          patch.setupQualityRating !== undefined ? patch.setupQualityRating : undefined,
   886→        executionRating:
   887→          patch.executionRating !== undefined ? patch.executionRating : undefined,
   888→        followedPlan: patch.followedPlan !== undefined ? patch.followedPlan : undefined,
   889→        reviewStatus: patch.reviewStatus ?? undefined,
   890→        isDraft: patch.isDraft !== undefined ? patch.isDraft : undefined,
   891→        preTradeAnalysis: patch.preTradeAnalysis ?? undefined,
   892→        postTradeReview: patch.postTradeReview ?? undefined,
   893→        notes: patch.notes !== undefined ? patch.notes : undefined,
   894→        tradingViewUrl: patch.tradingViewUrl !== undefined ? patch.tradingViewUrl : undefined,
   895→        checklistResults:
   896→          patch.checklistResults !== undefined
   897→            ? patch.checklistResults
   898→              ? JSON.stringify(patch.checklistResults)
   899→              : null
   900→            : undefined,
   901→      },
   902→    });
   903→    if (tagIds !== null) {
   904→      await tx.tradeTag.deleteMany({ where: { tradeId: current.id } });
   905→      await tx.tradeTag.createMany({
   906→        data: tagIds.map((tagId) => ({ tradeId: current.id, tagId })),
   907→      });
   908→    }
   909→  });
   910→
   911→  const trade = await getTradeDetail(userId, current.id);
   912→  if (!trade) throw new TradeValidationError("INTERNAL");
   913→  // Phase 6 — re-run rule detection (risk / P&L / day may have changed).
   914→  await evaluateTradeViolations(userId, current.id);
   915→  return { trade, conflict };
   916→}

   918→// ── Close an open trade (quick action, v2 §4.1) ─────────────────────────────
   919→
   920→export async function closeTrade(
   921→  userId: string,
   922→  id: string,
   923→  input: CloseTradeInput
   924→): Promise<TradeDetailDTO | null> {
   925→  const current = await db.trade.findFirst({
   926→    where: { id, userId },
   927→    include: { instrument: true, account: true },
   928→  });
   929→  if (!current) return null;
   930→  if (current.status === "CLOSED") {
   931→    throw new TradeValidationError("ALREADY_CLOSED");
   932→  }
   933→
   934→  const exitTime = input.exitTime ? new Date(input.exitTime) : new Date();
   935→  const commission = input.commission ?? current.commission.toString();
   936→  const swap = input.swap ?? current.swap.toString();
   937→  const fees = input.fees ?? current.fees.toString();
   938→  const pnlSource = input.pnlSource ?? current.pnlSource;
   939→
   940→  const balanceAtEntry = current.accountBalanceAtEntry
   941→    ? new Decimal(current.accountBalanceAtEntry)
   942→    : await computeBalanceAt(userId, current.accountId, current.entryTime);
   943→
   944→  const derived = computeDerived({
   945→    instrument: current.instrument,
   946→    account: current.account,
   947→    direction: current.direction,
   948→    entryPrice: current.entryPrice.toString(),
   949→    exitPrice: input.exitPrice,
   950→    stopLoss: current.stopLoss?.toString() ?? null,
   951→    takeProfit: current.takeProfit?.toString() ?? null,
   952→    lots: current.lots.toString(),
   953→    commission,
   954→    swap,
   955→    fees,
   956→    quoteToAccountRate: input.quoteToAccountRate ?? null,
   957→    pnlSource,
   958→    manualNetPnl: input.manualNetPnl ?? null,
   959→    entryTime: current.entryTime,
   960→    balanceAtEntry,
   961→  });
   962→
   963→  await db.$transaction(async (tx) => {
   964→    await tx.trade.update({
   965→      where: { id: current.id },
   966→      data: {
   967→        status: "CLOSED",
   968→        exitTime,
   969→        exitPrice: input.exitPrice,
   970→        pips: derived.pips,
   971→        points: derived.points,
   972→        grossPnl: derived.grossPnl,
   973→        netPnl: derived.netPnl,
   974→        rMultiple: derived.rMultiple,
   975→        plannedRR: derived.plannedRR,
   976→        realizedRR: derived.realizedRR,
   977→        resultType: derived.resultType,
   978→        riskAmount: derived.riskAmount,
   979→        riskPercent: derived.riskPercent,
   980→        pnlSource,
   981→        exitReason: input.exitReason ?? "MANUAL",
   982→        commission,
   983→        swap,
   984→        fees,
   985→      },
   986→    });
   987→    await tx.tradeExecution.create({
   988→      data: {
   989→        tradeId: current.id,
   990→        type: "FINAL_EXIT",
   991→        time: exitTime,
   992→        price: input.exitPrice,
   993→        lots: current.lots,
   994→        commission: 0,

   996→      },
   997→    });
   998→  });
   999→
  1000→  const trade = await getTradeDetail(userId, current.id);
  1001→  if (!trade) throw new TradeValidationError("INTERNAL");
  1002→  // Phase 6 — closing realizes the P&L: daily-loss rules re-evaluated.
  1003→  await evaluateTradeViolations(userId, current.id);
  1004→  return trade;
  1005→}
