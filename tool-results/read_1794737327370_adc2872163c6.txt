   128→/** v2.5 M2 — chart drill-down: apply #/trades?… filters once on mount. */
   129→function drillFiltersFromHash(): Partial<TradeListQuery> | null {
   130→  if (typeof window === "undefined") return null;
   131→  const route = parseHash(window.location.hash);
   132→  if (route.view !== "trades") return null;
   133→  const q = route.query ?? {};
   134→  const patch: Partial<TradeListQuery> = {};
   135→  let any = false;
   136→  const set = <K extends keyof TradeListQuery>(key: K, value: TradeListQuery[K]) => {
   137→    patch[key] = value;
   138→    any = true;
   139→  };
   140→  if (q.symbol) set("symbol", q.symbol);
   141→  if (q.direction === "LONG" || q.direction === "SHORT") {
   142→    set("direction", q.direction);
   143→  }
   144→  if (q.session) set("session", q.session as TradeListQuery["session"]);
   145→  if (q.timeframe) set("timeframe", q.timeframe as TradeListQuery["timeframe"]);
   146→  if (q.emotion) set("emotion", q.emotion as TradeListQuery["emotion"]);
   147→  if (q.search) set("search", q.search);
   148→  if (q.resultType) {
   149→    set("resultType", q.resultType as TradeListQuery["resultType"]);
   150→  }
   151→  if (q.dateFrom) set("dateFrom", q.dateFrom);
   152→  if (q.dateTo) set("dateTo", q.dateTo);
   153→  if (q.entryHour != null && /^\d{1,2}$/.test(q.entryHour)) {
   154→    set("entryHour", Number(q.entryHour));
   155→  }
   156→  if (q.entryWeekday != null && /^[1-7]$/.test(q.entryWeekday)) {
   157→    set("entryWeekday", Number(q.entryWeekday));
   158→  }
   159→  return any ? patch : null;
   160→}

  1093→/** Enum-like filter select with an implicit "all" (undefined) option. */
  1094→function FilterSelect<T extends string>({
  1095→  id,
  1096→  label,
  1097→  value,
  1098→  options,
  1099→  onChange,
  1100→}: {
  1101→  id: string;
  1102→  label: string;
  1103→  value: T | undefined;
  1104→  options: Array<{ value: T; label: string }>;
  1105→  onChange: (value: T | undefined) => void;
  1106→}) {
  1107→  const tCommon = useTranslations("common");
  1108→  return (
  1109→    <div className="space-y-2">
  1110→      <Label htmlFor={id}>{label}</Label>
  1111→      <Select
  1112→        value={value ?? "all"}
  1113→        onValueChange={(v) => onChange(v === "all" ? undefined : (v as T))}
  1114→      >
  1115→        <SelectTrigger id={id} className="w-full">
  1116→          <SelectValue />
  1117→        </SelectTrigger>
  1118→        <SelectContent className="max-h-72">
  1119→          <SelectItem value="all">{tCommon("all")}</SelectItem>
  1120→          {options.map((o) => (
  1121→            <SelectItem key={o.value} value={o.value}>
  1122→              {o.label}
  1123→            </SelectItem>
  1124→          ))}
  1125→        </SelectContent>
  1126→      </Select>
  1127→    </div>
  1128→  );
  1129→}
