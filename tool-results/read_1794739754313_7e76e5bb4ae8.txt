    81→// ── Category key formatting ──────────────────────────────────────────────────
    82→
    83→/** Localize a raw category key (weekday/month names via Intl, buckets via i18n). */
    84→export function formatCategoryKey(
    85→  dimension: XDimension | string,
    86→  key: string,
    87→  locale: string,
    88→  tBucket: (k: string) => string
    89→): string {
    90→  switch (dimension) {
    91→    case "dayOfWeek": {
    92→      const idx = WEEKDAY_KEY_LIST.indexOf(key);
    93→      if (idx >= 0) {
    94→        return new Intl.DateTimeFormat(locale, { weekday: "short" }).format(
    95→          new Date(2024, 0, 8 + idx) // 2024-01-08 = Monday
    96→        );
    97→      }
    98→      return key;
    99→    }
   100→    case "month": {
   101→      const idx = MONTH_KEY_LIST.indexOf(key);
   102→      if (idx >= 0) {
   103→        return new Intl.DateTimeFormat(locale, { month: "short" }).format(
   104→          new Date(2024, idx, 1)
   105→        );
   106→      }
   107→      return key;
   108→    }
   109→    case "hourOfDay":
   110→      return `${key}h`;
   111→    case "holdingTimeBucket":
   112→      return safeT(tBucket, `hold_${key}`, key);
   113→    case "lotSizeBucket":
   114→      return safeT(tBucket, `lot_${key.replace(/\./g, "_")}`, key);
   115→    case "riskPercentBucket":
   116→      return safeT(tBucket, `risk_${key.replace(/\./g, "_")}`, key);
   117→    case "week":
   118→      return `W${key}`;
   119→    default:
   120→      return key;
   121→  }
   122→}
   123→
   124→/** t() wrapper that falls back to the raw key when the message is missing. */
   125→function safeT(t: (k: string) => string, key: string, fallback: string): string {
   126→  try {
   127→    const v = t(key);
   128→    return v === key ? fallback : v;
   129→  } catch {
   130→    return fallback;
   131→  }
   132→}

   134→// ── Value formatting (by measure format) ─────────────────────────────────────
   135→
   136→export type MeasureFormatKind =
   137→  | "R"
   138→  | "CURRENCY"
   139→  | "PIPS"
   140→  | "POINTS"
   141→  | "LOTS"
   142→  | "PERCENT"
   143→  | "NUMBER"
   144→  | "DURATION";
   145→
   146→export function formatMeasureValue(
   147→  v: number | null | undefined,
   148→  format: MeasureFormatKind,
   149→  currency?: string
   150→): string {
   151→  if (v == null || !Number.isFinite(v)) return "—";
   152→  switch (format) {
   153→    case "CURRENCY":
   154→      return new Intl.NumberFormat(undefined, {
   155→        style: "currency",
   156→        currency: currency ?? "USD",
   157→        maximumFractionDigits: 0,
   158→      }).format(v);
   159→    case "R":
   160→      return `${fmtNum(v, 2)} R`;
   161→    case "PIPS":
   162→      return `${fmtNum(v, 1)} pips`;
   163→    case "POINTS":
   164→      return `${fmtNum(v, 1)} pts`;
   165→    case "LOTS":
   166→      return `${fmtNum(v, 2)} lots`;
   167→    case "PERCENT":
   168→      return `${fmtNum(v, 1)} %`;
   169→    case "DURATION": {
   170→      const min = Math.round(v);
   171→      if (min < 60) return `${min} min`;
   172→      if (min < 1440) return `${fmtNum(min / 60, 1)} h`;
   173→      return `${fmtNum(min / 1440, 1)} d`;
   174→    }
   175→    default:
   176→      return fmtNum(v, 2);
   177→  }
   178→}

   182→interface TooltipEntry {
   183→  name?: string;
   184→  value?: number | string | null;
   185→  color?: string;
   186→  payload?: ChartDatum & Record<string, unknown>;
   187→}
   188→
   189→function ChartTooltip({
   190→  active,
   191→  entries,
   192→  seriesDefs,
   193→  format,
   194→  currency,
   195→  formatKey,
   196→  t,
   197→}: {
   198→  active: boolean;
   199→  entries: TooltipEntry[];
   200→  seriesDefs: Array<{ key: string; label: string }>;
   201→  format: MeasureFormatKind;
   202→  currency: string;
   203→  formatKey: (key: string) => string;
   204→  t: (k: string) => string;
   205→}) {
   206→  if (!active || entries.length === 0) return null;
   207→  const datum = entries[0]?.payload as (ChartDatum & Record<string, unknown>) | undefined;
   208→  if (!datum) return null;
   209→  return (
   210→    <div className="min-w-40 rounded-md border bg-popover px-3 py-2 text-xs shadow-md">
   211→      <p className="mb-1 font-medium text-foreground">{formatKey(datum.key)}</p>
   212→      {entries.map((e) => (
   213→        <p key={e.name ?? "?"} className="flex items-center justify-between gap-3">
   214→          <span className="flex items-center gap-1.5 text-muted-foreground">
   215→            <span
   216→              aria-hidden="true"
   217→              className="inline-block h-2 w-2 rounded-full"
   218→              style={{ backgroundColor: e.color }}
   219→            />
   220→            {seriesDefs.find((s) => s.key === e.name)?.label ?? e.name}
   221→          </span>
   222→          <span className="font-mono text-foreground">
   223→            {formatMeasureValue(
   224→              typeof e.value === "number" ? e.value : null,
   225→              format,
   226→              currency
   227→            )}
   228→          </span>
   229→        </p>
   230→      ))}
   231→      <p className="mt-1 border-t pt-1 text-muted-foreground">
   232→        {t("sampleSize")}: {datum.total}
   233→        {datum.lowSample ? ` — ${t("lowSample")}` : ""}
   234→      </p>
   235→    </div>
   236→  );
   237→}

   239→function ScatterTooltip({
   240→  active,
   241→  payload,
   242→  formatX,
   243→  formatY,
   244→  t,
   245→}: {
   246→  active: boolean;
   247→  payload?: Array<{ payload?: ScatterPoint }>;
   248→  formatX: (v: number) => string;
   249→  formatY: (v: number) => string;
   250→  t: (k: string) => string;
   251→}) {
   252→  if (!active || !payload?.[0]?.payload) return null;
   253→  const p = payload[0].payload;
   254→  return (
   255→    <div className="rounded-md border bg-popover px-3 py-2 text-xs shadow-md">
   256→      <p className="font-mono">
   257→        x: {formatX(p.x)} · y: {formatY(p.y)}
   258→      </p>
   259→      <button
   260→        type="button"
   261→        className="mt-1 text-muted-foreground underline-offset-2 hover:underline"
   262→        onClick={() => window.dispatchEvent(
   263→          new CustomEvent("tj:chart-drill-trade", { detail: p.tradeId })
   264→        )}
   265→      >
   266→        {t("openTrade")}
   267→      </button>
   268→    </div>
   269→  );
   270→}

   274→export interface ChartPreviewProps {
   275→  result: ChartResult;
   276→  config: ChartConfig;
   277→  currency: string;
   278→  xFormat: MeasureFormatKind;
   279→  yFormat: MeasureFormatKind;
   280→  y2Format: MeasureFormatKind;
   281→  onDrill?: (drill: { dimension: XDimension; value: string }) => void;
   282→  /** Chart body height in px — defaults to 340 (builder page size).
   283→   *  Dashboard widgets pass their grid-derived height. */
   284→  height?: number;

   287→export function ChartPreview({
   288→  result,
   289→  config,
   290→  currency,
   291→  xFormat,
   292→  yFormat,
   293→  y2Format,
   294→  onDrill,
   295→  height = 340,
   296→}: ChartPreviewProps) {
   297→  const t = useTranslations("chartsBuilder.preview");
   298→  const { locale } = useI18n();
   299→  const tradingLabels = useTradingLabels();
   300→  const tBucket = useTranslations("chartsBuilder.buckets");
   301→
   302→  const formatKey = React.useCallback(
   303→    (key: string) =>
   304→      formatCategoryKey(config.x, key, locale, (k) => tBucket(k as never)),
   305→    [config.x, locale, tBucket]
   306→  );
   307→
   308→  const labelOfDimensionValue = React.useCallback(
   309→    (dim: string, value: string): string => {
   310→      if (dim === "session") return tradingLabels.session(value);
   311→      if (dim === "direction") return tradingLabels.direction(value);
   312→      if (dim === "result") return tradingLabels.result(value);
   313→      if (dim === "exitReason") return tradingLabels.exitReason(value);
   314→      if (dim === "marketCondition") return tradingLabels.marketCondition(value);
   315→      if (
   316→        dim === "emotionBefore" ||
   317→        dim === "emotionDuring" ||
   318→        dim === "emotionAfter"
   319→      ) {
   320→        return tradingLabels.emotion(value);
   321→      }
   322→      if (dim === "followedPlan") {
   323→        return value === "yes" ? t("yesLabel") : t("noLabel");
   324→      }
   325→      return formatKey(value);
   326→    },
   327→    [tradingLabels, formatKey, t]
   328→  );
   329→
   330→  const seriesDefs = result.series.map((s, i) => ({
   331→    key: s.key,
   332→    label:
   333→      config.groupBy === "none"
   334→        ? ""
   335→        : labelOfDimensionValue(config.groupBy, s.key) || s.key,
   336→    color: seriesColor(s.key, i),
   337→  }));
   338→
   339→  if (result.chartType === "scatter") {
   340→    return (
   341→      <ScatterPreview
   342→        result={result}
   343→        config={config}
   344→        seriesDefs={seriesDefs}
   345→        xFormat={xFormat}
   346→        yFormat={yFormat}
   347→        currency={currency}
   348→        t={t}
   349→        height={height}
   350→      />
   351→    );
   352→  }
   353→  if (result.chartType === "histogram") {
   354→    return (
   355→      <HistogramPreview
   356→        result={result}
   357→        xFormat={xFormat}
   358→        t={t}
   359→        height={height}
   360→      />
   361→    );
   362→  }
   363→  if (result.chartType === "box") {
   364→    return (
   365→      <BoxPlotPreview
   366→        result={result}
   367→        config={config}
   368→        seriesDefs={seriesDefs}
   369→        yFormat={yFormat}
   370→        currency={currency}
   371→        formatKey={formatKey}
   372→        labelOf={labelOfDimensionValue}
   373→        t={t}
   374→        onDrill={onDrill}
   375→        height={height}
   376→      />
   377→    );
   378→  }
   379→  return (
   380→    <CategoricalPreview
   381→      result={result}
   382→      config={config}
   383→      seriesDefs={seriesDefs}
   384→      yFormat={yFormat}
   385→      y2Format={y2Format}
   386→      currency={currency}
   387→      formatKey={formatKey}
   388→      t={t}
   389→      onDrill={onDrill}
   390→      height={height}
   391→    />
   392→  );
   393→}

   397→function CategoricalPreview({
   398→  result,
   399→  config,
   400→  seriesDefs,
   401→  yFormat,
   402→  y2Format,
   403→  currency,
   404→  formatKey,
   405→  t,
   406→  onDrill,
   407→  height,
   408→}: {
   409→  result: ChartResult;
   410→  config: ChartConfig;
   411→  seriesDefs: Array<{ key: string; label: string; color: string }>;
   412→  yFormat: MeasureFormatKind;
   413→  y2Format: MeasureFormatKind;
   414→  currency: string;
   415→  formatKey: (key: string) => string;
   416→  t: (k: string) => string;
   417→  onDrill?: (drill: { dimension: XDimension; value: string }) => void;
   418→  height: number;
   419→}) {
   420→  // Recharts data: flatten datum.values into { key, [seriesKey]: value }.
   421→  const data = result.data.map((d) => ({
   422→    key: formatKey(d.key),
   423→    raw: d,
   424→    total: d.total,
   425→    lowSample: d.lowSample,
   426→    ...d.values,
   427→  }));
   428→  const singleSeries = seriesDefs.length === 1 && seriesDefs[0].key === "all";
   429→  const colorBySign =
   430→    singleSeries && SIGNED_MEASURES.includes(config.y ?? "") && config.chartType === "bar";
   431→
   432→  const handleBarClick = (entry: unknown) => {
   433→    const raw = (entry as { raw?: ChartDatum } | undefined)?.raw;
   434→    if (raw && onDrill) onDrill(raw.drill);
   435→  };
   436→
   437→  const xAxisTick = (v: string | number) => String(v);
   438→
   439→  return (
   440→    <ResponsiveContainer width="100%" height={height}>
   441→      <ComposedChart data={data} margin={{ top: 12, right: 12, bottom: 4, left: 0 }}>
   442→        <CartesianGrid strokeDasharray="3 3" strokeOpacity={0.3} vertical={false} />
   443→        <XAxis
   444→          dataKey="key"
   445→          tick={{ fontSize: 11 }}
   446→          tickLine={false}
   447→          axisLine={false}
   448→          minTickGap={12}
   449→          tickFormatter={xAxisTick}
   450→        />
   451→        <YAxis
   452→          yAxisId="left"
   453→          tick={{ fontSize: 11 }}
   454→          tickLine={false}
   455→          axisLine={false}
   456→          width={64}
   457→          tickFormatter={(v: number) => formatMeasureValue(v, yFormat, currency)}
   458→        />
   459→        {config.y2 != null && (
   460→          <YAxis
   461→            yAxisId="right"
   462→            orientation="right"
   463→            tick={{ fontSize: 11 }}
   464→            tickLine={false}
   465→            axisLine={false}
   466→            width={64}
   467→            tickFormatter={(v: number) => formatMeasureValue(v, y2Format, currency)}
   468→          />
   469→        )}
   470→        <Tooltip
   471→          cursor={{ fillOpacity: 0.08 }}
   472→          content={({ active, payload }) => (
   473→            <ChartTooltip
   474→              active={Boolean(active)}
   475→              entries={(payload ?? []) as TooltipEntry[]}
   476→              seriesDefs={seriesDefs}
   477→              format={yFormat}
   478→              currency={currency}
   479→              formatKey={formatKey}
   480→              t={t}
   481→            />
   482→          )}
   483→        />
   484→        <ReferenceLine yAxisId="left" y={0} stroke="var(--border)" />
   485→        {config.chartType === "bar" &&
   486→          seriesDefs.map((s) => (
   487→            <Bar
   488→              key={s.key}
   489→              yAxisId="left"
   490→              dataKey={s.key}
   491→              name={s.key}
   492→              fill={s.color}
   493→              radius={[3, 3, 0, 0]}
   494→              maxBarSize={42}
   495→              onClick={handleBarClick}
   496→              className={onDrill ? "cursor-pointer" : undefined}
   497→            >
   498→              {colorBySign &&
   499→                data.map((d, i) => {
   500→                  const v = d[s.key] as number | null;
   501→                  return (
   502→                    <Cell
   503→                      key={i}
   504→                      fill={v == null ? "var(--muted)" : v >= 0 ? "var(--gain)" : "var(--loss)"}
   505→                      fillOpacity={d.raw.lowSample ? 0.35 : 1}
   506→                      stroke={d.raw.lowSample ? "var(--muted-foreground)" : undefined}
   507→                      strokeDasharray={d.raw.lowSample ? "3 2" : undefined}
   508→                    />
   509→                  );
   510→                })}
   511→              {!colorBySign &&
   512→                data.map((d, i) => (
   513→                  <Cell
   514→                    key={i}
   515→                    fillOpacity={d.raw.lowSample ? 0.35 : 1}
   516→                    stroke={d.raw.lowSample ? "var(--muted-foreground)" : undefined}
   517→                    strokeDasharray={d.raw.lowSample ? "3 2" : undefined}
   518→                  />
   519→                ))}
   520→            </Bar>
   521→          ))}
   522→        {(config.chartType === "line" || config.chartType === "area") &&
   523→          seriesDefs.map((s) =>
   524→            config.chartType === "area" ? (
   525→              <Area
   526→                key={s.key}
   527→                yAxisId="left"
   528→                type="monotone"
   529→                dataKey={s.key}
   530→                name={s.key}
   531→                stroke={s.color}
   532→                strokeWidth={2}
   533→                fill={s.color}
   534→                fillOpacity={0.15}
   535→                dot={data.length <= 30 ? { r: 2.5 } : false}
   536→                activeDot={{ r: 4 }}
   537→                connectNulls
   538→              />
   539→            ) : (
   540→              <Line
   541→                key={s.key}
   542→                yAxisId="left"
   543→                type="monotone"
   544→                dataKey={s.key}
   545→                name={s.key}
   546→                stroke={s.color}
   547→                strokeWidth={2}
   548→                dot={data.length <= 30 ? { r: 2.5 } : false}
   549→                activeDot={{ r: 4 }}
   550→                connectNulls
   551→              />
   552→            )
   553→          )}
   554→        {config.y2 != null && config.chartType === "line" && (
   555→          <Line
   556→            yAxisId="right"
   557→            type="monotone"
   558→            dataKey={config.y2}
   559→            name={config.y2}
   560→            stroke="var(--chart-3)"
   561→            strokeWidth={2}
   562→            strokeDasharray="5 3"
   563→            dot={false}
   564→            connectNulls
   565→          />
   566→        )}
   567→        {config.y2 != null && config.chartType === "bar" && (
   568→          <Bar
   569→            yAxisId="right"
   570→            dataKey={config.y2}
   571→            name={config.y2}
   572→            fill="var(--chart-3)"
   573→            fillOpacity={0.55}
   574→            radius={[3, 3, 0, 0]}
   575→            maxBarSize={42}
   576→          />
   577→        )}
   578→        {config.y2 != null && config.chartType === "area" && (
   579→          <Area
   580→            yAxisId="right"
   581→            type="monotone"
   582→            dataKey={config.y2}
   583→            name={config.y2}
   584→            stroke="var(--chart-3)"
   585→            strokeDasharray="5 3"
   586→            strokeWidth={2}
   587→            fill="var(--chart-3)"
   588→            fillOpacity={0.08}
   589→            dot={false}
   590→            connectNulls
   591→          />
   592→        )}
   593→      </ComposedChart>
   594→    </ResponsiveContainer>
   595→  );
   596→}

   598→// ── Scatter ──────────────────────────────────────────────────────────────────
   599→
   600→function ScatterPreview({
   601→  result,
   602→  config,
   603→  seriesDefs,
   604→  xFormat,
   605→  yFormat,
   606→  currency,
   607→  t,
   608→  height,
   609→}: {
   610→  result: ChartResult;
   611→  config: ChartConfig;
   612→  seriesDefs: Array<{ key: string; label: string; color: string }>;
   613→  xFormat: MeasureFormatKind;
   614→  yFormat: MeasureFormatKind;
   615→  currency: string;
   616→  t: (k: string) => string;
   617→  height: number;
   618→}) {
   619→  const byGroup = new Map<string, ScatterPoint[]>();
   620→  for (const p of result.points) {
   621→    const list = byGroup.get(p.group);
   622→    if (list) list.push(p);
   623→    else byGroup.set(p.group, [p]);
   624→  }
   625→  const fmtX = (v: number) =>
   626→    config.x === "date"
   627→      ? new Intl.DateTimeFormat(undefined, { dateStyle: "short" }).format(new Date(v))
   628→      : formatMeasureValue(v, xFormat, currency);
   629→  const fmtY = (v: number) => formatMeasureValue(v, yFormat, currency);

   758→function BoxPlotPreview({
   759→  result,
   760→  config,
   761→  seriesDefs,
   762→  yFormat,
   763→  currency,
   764→  formatKey,
   765→  labelOf,
   766→  t,
   767→  onDrill,
   768→  height: bodyHeight,
   769→}: {
   770→  result: ChartResult;
   771→  config: ChartConfig;
   772→  seriesDefs: Array<{ key: string; label: string; color: string }>;
   773→  yFormat: MeasureFormatKind;
   774→  currency: string;
   775→  formatKey: (key: string) => string;
   776→  labelOf: (dim: string, value: string) => string;
   777→  t: (k: string) => string;
   778→  onDrill?: (drill: { dimension: XDimension; value: string }) => void;
   779→  height: number;
   780→}) {
   781→  const width = 900;
   782→  const height = Math.max(220, bodyHeight);
   783→  const padL = 64;
   784→  const padR = 16;
   785→  const padT = 16;
   786→  const padB = 56;
   787→  const plotW = width - padL - padR;
   788→  const plotH = height - padT - padB;
   789→
   790→  // Y scale over all box values + outliers.
   791→  let yMin = Infinity;
   792→  let yMax = -Infinity;
   793→  for (const d of result.data) {
   794→    for (const box of Object.values(d.boxes ?? {})) {
   795→      if (!box) continue;
   796→      const vals = [box.whiskerLow, box.whiskerHigh, ...box.outliers];
   797→      yMin = Math.min(yMin, ...vals);
   798→      yMax = Math.max(yMax, ...vals);
   799→    }
   800→  }
   801→  if (!Number.isFinite(yMin) || !Number.isFinite(yMax)) {
   802→    yMin = 0;
   803→    yMax = 1;
   804→  }
   805→  if (yMin === yMax) {
   806→    yMin -= 1;
   807→    yMax += 1;
   808→  }
   809→  const pad = (yMax - yMin) * 0.06;
   810→  yMin -= pad;
   811→  yMax += pad;
   812→  const yOf = (v: number) => padT + plotH - ((v - yMin) / (yMax - yMin)) * plotH;
   813→
   814→  const slotW = plotW / Math.max(1, result.data.length);
   815→  const boxW = Math.min(28, (slotW * 0.62) / Math.max(1, seriesDefs.length));
   816→  const yTicks = 5;
   817→  const ticks = Array.from({ length: yTicks }, (_, i) => yMin + ((yMax - yMin) * i) / (yTicks - 1));
   818→
   819→  const [hover, setHover] = React.useState<string | null>(null);
   820→
   821→  return (
   822→    <svg
   823→      viewBox={`0 0 ${width} ${height}`}
   824→      className="w-full"
   825→      style={{ height }}
   826→      role="img"
   827→      aria-label={t("boxAria")}
   828→    >
   829→      {/* gridlines + y axis */}
   830→      {ticks.map((v, i) => (
   831→        <g key={i}>
   832→          <line
   833→            x1={padL}
   834→            x2={width - padR}
   835→            y1={yOf(v)}
   836→            y2={yOf(v)}
   837→            stroke="var(--border)"
   838→            strokeDasharray="3 3"
   839→            strokeOpacity={0.5}
   840→          />
   841→          <text
   842→            x={padL - 8}
   843→            y={yOf(v) + 3}
   844→            textAnchor="end"
   845→            fontSize={11}
   846→            fill="var(--muted-foreground)"
   847→          >
   848→            {formatMeasureValue(v, yFormat, currency)}
   849→          </text>
   850→        </g>
   851→      ))}
   852→      <line
   853→        x1={padL}
   854→        x2={width - padR}
   855→        y1={yOf(Math.max(yMin, Math.min(0, yMax)))}
   856→        y2={yOf(Math.max(yMin, Math.min(0, yMax)))}
   857→        stroke="var(--border)"
   858→      />
   859→
   860→      {/* boxes */}
   861→      {result.data.map((d, di) => {
   862→        const cx0 = padL + di * slotW;
   863→        const isHover = hover === `${d.key}`;
   864→        return (
   865→          <g key={d.key}>
   866→            {seriesDefs.map((s, si) => {
   867→              const box = d.boxes?.[s.key];
   868→              if (!box) return null;
   869→              const cx = cx0 + slotW / 2 + (si - (seriesDefs.length - 1) / 2) * (boxW + 4);
   870→              const lowY = yOf(box.whiskerHigh);
   871→              const highY = yOf(box.whiskerLow);
   872→              const q1Y = yOf(box.q3);
   873→              const q3Y = yOf(box.q1);
   874→              const medY = yOf(box.median);
   875→              const color = s.color;
   876→              const dim = d.lowSample ? 0.4 : 1;
   877→              return (
   878→                <g
   879→                  key={s.key}
   880→                  tabIndex={0}
   881→                  role="img"
   882→                  aria-label={`${formatKey(d.key)}: Q1 ${fmtNum(box.q1, 2)}, med ${fmtNum(box.median, 2)}, Q3 ${fmtNum(box.q3, 2)}`}
   883→                  className="outline-none focus-visible:outline-2 focus-visible:outline-ring"
   884→                  onMouseEnter={() => setHover(d.key)}
   885→                  onMouseLeave={() => setHover(null)}
   886→                  onClick={() => onDrill?.(d.drill)}
   887→                  style={{ cursor: onDrill ? "pointer" : undefined }}
   888→                >
   889→                  {/* whisker */}
   890→                  <line x1={cx} x2={cx} y1={lowY} y2={q1Y} stroke={color} strokeWidth={1.5} opacity={dim} />
   891→                  <line x1={cx} x2={cx} y1={q3Y} y2={highY} stroke={color} strokeWidth={1.5} opacity={dim} />
   892→                  <line x1={cx - boxW / 3} x2={cx + boxW / 3} y1={lowY} y2={lowY} stroke={color} strokeWidth={1.5} opacity={dim} />
   893→                  <line x1={cx - boxW / 3} x2={cx + boxW / 3} y1={highY} y2={highY} stroke={color} strokeWidth={1.5} opacity={dim} />
   894→                  {/* box */}
   895→                  <rect
   896→                    x={cx - boxW / 2}
   897→                    y={q1Y}
   898→                    width={boxW}
   899→                    height={Math.max(1, q3Y - q1Y)}
   900→                    fill={color}
   901→                    fillOpacity={0.18 * dim}
   902→                    stroke={color}
   903→                    strokeWidth={1.5}
   904→                    strokeDasharray={d.lowSample ? "3 2" : undefined}
   905→                    opacity={dim}
   906→                    rx={2}
   907→                  />
   908→                  {/* median */}
   909→                  <line
   910→                    x1={cx - boxW / 2}
   911→                    x2={cx + boxW / 2}
   912→                    y1={medY}
   913→                    y2={medY}
   914→                    stroke={color}
   915→                    strokeWidth={2.5}
   916→                    opacity={dim}
   917→                  />
   918→                  {/* outliers */}
   919→                  {box.outliers.map((o, oi) => (
   920→                    <circle
   921→                      key={oi}
   922→                      cx={cx}
   923→                      cy={yOf(o)}
   924→                      r={2}
   925→                      fill="var(--muted-foreground)"
   926→                      opacity={dim}
   927→                    />
   928→                  ))}
   929→                  {isHover && (
   930→                    <text
   931→                      x={cx}
   932→                      y={padT - 4}
   933→                      textAnchor="middle"
   934→                      fontSize={11}
   935→                      fill="var(--foreground)"
   936→                    >
   937→                      n={d.total}
   938→                    </text>
   939→                  )}
   940→                </g>
   941→              );
   942→            })}
   943→            {/* x label (skip when crowded) */}
   944→            {(result.data.length <= 24 || di % Math.ceil(result.data.length / 24) === 0) && (
   945→              <text
   946→                x={cx0 + slotW / 2}
   947→                y={height - padB + 18}
   948→                textAnchor="middle"
   949→                fontSize={11}
   950→                fill="var(--muted-foreground)"
   951→              >
   952→                {formatKey(d.key).slice(0, 10)}
   953→              </text>
   954→            )}
   955→          </g>
   956→        );
   957→      })}
   958→      {/* legend */}
   959→      {seriesDefs.length > 1 && (
   960→        <g>
   961→          {seriesDefs.map((s, i) => (
   962→            <g key={s.key} transform={`translate(${padL + i * 120}, ${height - 12})`}>
   963→              <rect width={10} height={10} y={-9} fill={s.color} fillOpacity={0.4} stroke={s.color} rx={2} />
   964→              <text x={16} fontSize={11} fill="var(--muted-foreground)">
   965→                {labelOf(config.groupBy, s.key) || s.key}
   966→              </text>
   967→            </g>
   968→          ))}
   969→        </g>
   970→      )}
   971→    </svg>
   972→  );
   973→}
