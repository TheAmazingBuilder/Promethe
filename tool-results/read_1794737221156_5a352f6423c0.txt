   178→   178→function BreakdownCard({
   179→   179→  title,
   180→   180→  description,
   181→   181→  icon: Icon,
   182→   182→  entries,
   183→   183→  currency,
   184→   184→  renderKey,
   185→   185→  max = 8,
   186→   186→}: {
   187→   187→  title: string;
   188→   188→  description?: string;
   189→   189→  icon: React.ComponentType<{ className?: string }>;
   190→   190→  entries: GroupStat[];
   191→   191→  currency: string;
   192→   192→  renderKey: (key: string) => string;
   193→   193→  max?: number;
   194→   194→}) {
   195→   195→  const t = useTranslations("analytics");
   196→   196→  const shown = entries.slice(0, max);
   197→   197→  const maxAbs = shown.reduce(
   198→   198→    (m, e) => Math.max(m, Math.abs(Number(e.stats.netPnl))),
   199→   199→    0
   200→   200→  );
   201→   201→  return (
   202→   202→    <Card>
   203→   203→      <CardHeader>
   204→   204→        <CardTitle className="flex items-center gap-2 text-base">
   205→   205→          <Icon className="h-4 w-4 text-muted-foreground" aria-hidden="true" />
   206→   206→          {title}
   207→   207→        </CardTitle>
   208→   208→        {description ? <CardDescription>{description}</CardDescription> : null}
   209→   209→      </CardHeader>
   210→   210→      <CardContent>
   211→   211→        {shown.length === 0 ? (
   212→   212→          <p className="text-sm text-muted-foreground">{t("noData")}</p>
   213→   213→        ) : (
   214→   214→          <ul className="space-y-1.5">
   215→   215→            {shown.map((e) => {
   216→   216→              const net = Number(e.stats.netPnl);
   217→   217→              const width =
   218→   218→                maxAbs > 0 ? Math.max(2, (Math.abs(net) / maxAbs) * 100) : 0;
   219→   219→              return (
   220→   220→                <li
   221→   221→                  key={e.key}
   222→   222→                  className="relative flex items-center justify-between gap-2 overflow-hidden rounded-md px-2 py-1.5 text-sm"
   223→   223→                >
   224→   224→                  <span
   225→   225→                    className={`absolute inset-y-0 left-0 ${
   226→   226→                      net >= 0 ? "bg-gain-subtle" : "bg-loss-subtle"
   227→   227→                    }`}
   228→   228→                    style={{ width: `${width}%` }}
   229→   229→                    aria-hidden="true"
   230→   230→                  />
   231→   231→                  <span className="relative z-10 min-w-0 truncate font-medium">
   232→   232→                    {renderKey(e.key)}
   233→   233→                  </span>
   234→   234→                  <span className="relative z-10 flex shrink-0 items-center gap-3 tabular-nums">
   235→   235→                    <span className="text-xs text-muted-foreground">
   236→   236→                      ×{e.stats.tradeCount}
   237→   237→                    </span>
   238→   238→                    <span className="hidden w-16 whitespace-nowrap text-right text-xs text-muted-foreground sm:inline">
   239→   239→                      {e.stats.winRate != null
   240→   240→                        ? fmtPct(e.stats.winRate)
   241→   241→                        : "—"}
   242→   242→                    </span>
   243→   243→                    <span
   244→   244→                      className={`w-24 text-right font-medium ${pnlClass(
   245→   245→                        e.stats.netPnl
   246→   246→                      )}`}
   247→   247→                    >
   248→   248→                      {fmtSignedMoney(e.stats.netPnl, currency)}
   249→   249→                    </span>
   250→   250→                  </span>
   251→   251→                </li>
   252→   252→              );
   253→   253→            })}
   254→   254→            {entries.length > shown.length ? (
   255→   255→              <li className="px-2 pt-1 text-xs text-muted-foreground">
   256→   256→                {t("more", { count: entries.length - shown.length })}
   257→   257→              </li>
   258→   258→            ) : null}
   259→   259→          </ul>
   260→   260→        )}
   261→   261→      </CardContent>
   262→   262→    </Card>
   263→   263→  );
   264→   264→}

   268→   268→export function AnalyticsView() {
   269→   269→  const { session } = useSession();
   270→   270→  const t = useTranslations("analytics");
   271→   271→  const tRoot = useTranslations();

   275→   275→  const [accountId, setAccountId] = React.useState<string>("all");
   276→   276→
   277→   277→  const { data: accountsData } = useAccounts();
   278→   278→  const accounts = accountsData ?? [];
   279→   279→
   280→   280→  const effectiveAccount = accountId === "all" ? undefined : accountId;
   281→   281→  const { data, isLoading, isError, refetch } = useAnalytics({
   282→   282→    period,
   283→   283→    account: effectiveAccount,
   284→   284→  });
   285→   285→
   286→   286→  const timezone = data?.timezone ?? session?.profile.timezone ?? "UTC";
   287→   287→  const currency = data?.currency ?? session?.profile.baseCurrency ?? "USD";
   288→   288→  const stats = data?.stats ?? null;
   289→   289→  const advanced = data?.advanced ?? null;
   290→   290→
   291→   291→  const equityData = React.useMemo<EquityDatum[]>(
   292→   292→    () =>
   293→   293→      (data?.equityCurve ?? []).map((p) => ({
   294→   294→        i: p.i,
   295→   295→        date: p.date,
   296→   296→        equity: Number(p.equity),
   297→   297→        drawdown: Number(p.drawdown),
   298→   298→      })),
   299→   299→    [data]
   300→   300→  );
   301→   301→
   302→   302→  const rDistData = React.useMemo(
   303→   303→    () =>
   304→   304→      (data?.rDistribution.buckets ?? []).map((b) => ({
   305→   305→        id: b.id,
   306→   306→        label: t(`rdist.${b.id}`),
   307→   307→        count: b.count,
   308→   308→        totalR: Number(b.totalR),
   309→   309→        negative: b.to != null ? b.to <= 0 : false,
   310→   310→      })),
   311→   311→    [data, t]
   312→   312→  );
   313→   313→
   314→   314→  const weekdayLabel = React.useCallback(
   315→   315→    (key: string) => {
   316→   316→      try {
   317→   317→        // ISO weekday 1-7 → localized name via a stable reference week
   318→   318→        // (2024-01-01 was a Monday).
   319→   319→        const ref = new Date(Date.UTC(2024, 0, Number(key)));
   320→   320→        return new Intl.DateTimeFormat(locale, {
   321→   321→          weekday: "long",
   322→   322→          timeZone: "UTC",
   323→   323→        }).format(ref);
   324→   324→      } catch {
   325→   325→        return key;
   326→   326→      }
   327→   327→    },
   328→   328→    [locale]
   329→   329→  );
   330→   330→
   331→   331→  const hourLabel = React.useCallback((key: string) => {
   332→   332→    const h = Number(key);
   333→   333→    if (!Number.isFinite(h)) return key;
   334→   334→    const two = (n: number) => String(n).padStart(2, "0");
   335→   335→    return `${two(h)}:00–${two(h + 1)}:59`;
   336→   336→  }, []);
   337→   337→
   338→   338→  /** Formats raw insight params for i18n interpolation (money, R, labels…). */
   339→   339→  const formatInsightParams = React.useCallback(
   340→   340→    (insight: Insight): Record<string, string> => {
   341→   341→      const money = new Set(["netPnl", "fees", "dd", "pnl"]);
   342→   342→      const pct = new Set(["winRate", "pct"]);
   343→   343→      const r = new Set([
   344→   344→        "avgR",
   345→   345→        "avgRPlan",
   346→   346→        "avgRNoPlan",
   347→   347→        "busyAvgR",
   348→   348→        "lightAvgR",
   349→   349→      ]);
   350→   350→      const num = new Set(["payoff", "rf"]);
   351→   351→      const out: Record<string, string> = {};
   352→   352→      for (const [k, v] of Object.entries(insight.params)) {
   353→   353→        if (money.has(k) && typeof v === "string") {
   354→   354→          out[k] = fmtSignedMoney(v, currency);
   355→   355→        } else if (pct.has(k) && typeof v === "string") {
   356→   356→          out[k] = fmtPct(v);
   357→   357→        } else if (r.has(k) && typeof v === "string") {
   358→   358→          out[k] = `${fmtSigned(v)} R`;
   359→   359→        } else if (num.has(k) && typeof v === "string") {
   360→   360→          out[k] = fmtNum(v);
   361→   361→        } else if (k === "weekday") {
   362→   362→          out[k] = weekdayLabel(String(v));
   363→   363→        } else if (k === "hour") {
   364→   364→          out[k] = hourLabel(String(v));
   365→   365→        } else if (k === "session") {
   366→   366→          out[k] = labels.session(String(v));
   367→   367→        } else if (k === "emotion") {
   368→   368→          out[k] = labels.emotion(String(v));
   369→   369→        } else {
   370→   370→          out[k] = String(v);
   371→   371→        }
   372→   372→      }
   373→   373→      return out;
   374→   374→    },
   375→   375→    [currency, weekdayLabel, hourLabel, labels]
   376→   376→  );
   377→   377→
   378→   378→  const renderInsight = (insight: Insight) =>
   379→   379→    tRoot(`analytics.insights.${insight.rule}` as Parameters<typeof tRoot>[0], {
   380→   380→      ...formatInsightParams(insight),
   381→   381→    });
   382→   382→
   383→   383→  const goodInsights = (data?.insights ?? []).filter((i) => i.tone === "good");
   384→   384→  const badInsights = (data?.insights ?? []).filter((i) => i.tone === "bad");
   385→   385→  const infoInsights = (data?.insights ?? []).filter((i) => i.tone === "info");
   386→   386→
   387→   387→  /** KPI display model. */
   388→   388→  const kpi = React.useMemo(() => {
   389→   389→    const s = stats;
   390→   390→    const pfNum = s?.profitFactor != null ? Number(s.profitFactor) : null;
   391→   391→    const streak = s?.currentStreak ?? 0;
   392→   392→    return {
   393→   393→      netPnl: s ? fmtSignedMoney(s.netPnl, currency) : "—",
   394→   394→      netPnlTone: s ? pnlClass(s.netPnl) : "text-muted-foreground",
   395→   395→      winRate: s?.winRate != null ? fmtPct(s.winRate) : "—",
   396→   396→      profitFactor:
   397→   397→        pfNum == null
   398→   398→          ? "—"
   399→   399→          : Number.isFinite(pfNum)
   400→   400→            ? fmtNum(pfNum)
   401→   401→            : "∞",
   402→   402→      profitFactorTone:
   403→   403→        pfNum == null
   404→   404→          ? ""
   405→   405→          : pfNum >= 2
   406→   406→            ? EMERALD
   407→   407→            : pfNum < 1
   408→   408→              ? ROSE
   409→   409→              : "",
   410→   410→      expectancy: s?.expectancyR != null ? `${fmtSigned(s.expectancyR)} R` : "—",
   411→   411→      expectancyTone:
   412→   412→        s?.expectancyR != null ? pnlClass(s.expectancyR) : "text-muted-foreground",
   413→   413→      totalR: s?.totalR != null ? `${fmtSigned(s.totalR)} R` : "—",
   414→   414→      maxDrawdown: s?.maxDrawdown != null ? fmtMoney(s.maxDrawdown, currency) : "—",
   415→   415→      maxDrawdownTone: s?.maxDrawdown != null ? ROSE : "",
   416→   416→      sqn: advanced?.sqn != null ? fmtNum(advanced.sqn) : "—",
   417→   417→      sqnTone:
   418→   418→        advanced?.sqn == null
   419→   419→          ? ""
   420→   420→          : Number(advanced.sqn) >= 2
   421→   421→            ? EMERALD
   422→   422→            : Number(advanced.sqn) < 1
   423→   423→              ? ROSE
   424→   424→              : "",
   425→   425→      tradeCount: s ? fmtNum(s.tradeCount, 0) : "—",
   426→   426→      winsLosses: s
   427→   427→        ? t("kpiWinsLosses", { wins: s.wins, losses: s.losses, be: s.breakevens })
   428→   428→        : null,
   429→   429→      sharpe: advanced?.sharpe != null ? fmtNum(advanced.sharpe) : "—",
   430→   430→      sortino: advanced?.sortino != null ? fmtNum(advanced.sortino) : "—",
   431→   431→      recovery:
   432→   432→        advanced?.recoveryFactor != null ? fmtNum(advanced.recoveryFactor) : "—",
   433→   433→      payoff: advanced?.payoffRatio != null ? fmtNum(advanced.payoffRatio) : "—",
   434→   434→      avgDuration:
   435→   435→        advanced?.avgDurationMin != null
   436→   436→          ? advanced.avgDurationMin >= 60
   437→   437→            ? t("durationHm", {
   438→   438→                h: Math.floor(advanced.avgDurationMin / 60),
   439→   439→                m: advanced.avgDurationMin % 60,
   440→   440→              })
   441→   441→            : t("durationM", { m: advanced.avgDurationMin })
   442→   442→          : "—",
   443→   443→      totalFees:
   444→   444→        advanced != null ? fmtMoney(advanced.totalFees, currency) : "—",

   446→   446→      maxWinStreak: s ? fmtNum(s.maxWinStreak, 0) : "—",

   448→   448→      streakText:
   449→   449→        streak > 0
   450→   450→          ? t("streakWin", { count: streak })
   451→   451→          : streak < 0
   452→   452→            ? t("streakLoss", { count: Math.abs(streak) })
   453→   453→            : "—",
   454→   454→      streakTone: streak > 0 ? EMERALD : streak < 0 ? ROSE : "",
   455→   455→    };
   456→   456→  }, [stats, advanced, currency, t]);
   457→   457→
   458→   458→  if (!session) return null;
   459→   459→
   460→   460→  const hasAnyTrade = stats != null;
   461→   461→  const showAllCta = !hasAnyTrade && period !== "all";
   462→   462→
   463→   463→  return (
   464→   464→    <div className="space-y-6">
   465→   465→      {/* ── Header + filters ── */}
   466→   466→      <div className="flex flex-col gap-3">
   467→   467→        <div className="flex flex-col gap-1">
   468→   468→          <h2 className="text-xl font-bold tracking-tight sm:text-2xl">
   469→   469→            {t("title")}
   470→   470→          </h2>
   471→   471→          <p className="text-sm text-muted-foreground">
   472→   472→            {hasAnyTrade
   473→   473→              ? t("subtitle", { count: stats.tradeCount })
   474→   474→              : t("subtitleEmpty")}
   475→   475→          </p>
   476→   476→        </div>
   477→   477→        <div className="flex flex-wrap items-center gap-2">
