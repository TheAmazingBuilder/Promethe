   149→   149→        i: acc.length + 1,
   150→   150→        date: tr.exitTime,
   151→   151→        ms: new Date(tr.exitTime).getTime(),

   153→   153→      });

   155→   155→    }, []);
   156→   156→  }, [trades]);

   170→   170→  };
   171→   171→
   172→   172→  const exportPng = async () => {
   173→   173→    const ok = await exportSvgToPng(chartRef.current, "equity-annotated.png");

   175→   175→  };
   176→   176→
   177→   177→  const selectedStats = React.useMemo(
   178→   178→    () =>
   179→   179→      selected
   180→   180→        ? beforeAfterStats(trades, selected)
   181→   181→        : null,
   182→   182→    [selected, trades]
   183→   183→  );
   184→   184→
   185→   185→  const fmtDay = (iso: string) =>
   186→   186→    new Intl.DateTimeFormat(locale, { dateStyle: "medium" }).format(
   187→   187→      new Date(iso)
   188→   188→    );
   189→   189→
   190→   190→  return (
   191→   191→    <div className="space-y-6">
   192→   192→      {/* Header */}
   193→   193→      <div className="flex flex-wrap items-start justify-between gap-3">

   196→   196→          <p className="text-sm text-muted-foreground">{t("description")}</p>
   197→   197→        </div>
   198→   198→        <div className="flex gap-1">
   199→   199→          <Button variant="outline" size="sm" onClick={() => navigate("#/analytics")}>
   200→   200→            {tCommon("back")}
   201→   201→          </Button>
   202→   202→          <Button variant="outline" size="sm" onClick={() => setCreating(new Date().toISOString())}>
   203→   203→            <Plus aria-hidden="true" />
   204→   204→            {t("create")}
   205→   205→          </Button>
   206→   206→        </div>
   207→   207→      </div>
   208→   208→
   209→   209→      {/* Annotated equity chart */}
   210→   210→      <Card>
   211→   211→        <CardHeader className="flex-row items-start justify-between space-y-0">
   212→   212→          <div className="space-y-1.5">
   213→   213→            <CardTitle className="text-base">{t("chartTitle")}</CardTitle>
   214→   214→            <CardDescription>{t("chartDesc")}</CardDescription>
   215→   215→          </div>
   216→   216→          <div className="flex items-center gap-1.5">
   217→   217→            <Select value={kindFilter} onValueChange={setKindFilter}>

   240→   240→              {t("noData")}
   241→   241→            </p>

   243→   243→            <div ref={chartRef} className="h-72 w-full" role="img" aria-label={t("chartTitle")}>
   244→   244→              <ResponsiveContainer width="100%" height="100%">
   245→   245→                <AreaChart
   246→   246→                  data={chartData}
   247→   247→                  margin={{ top: 8, right: 12, bottom: 0, left: 0 }}

   306→   306→                        fillOpacity={0.12}
   307→   307→                        stroke={a.color ?? KIND_COLORS[a.kind] ?? KIND_COLORS.OTHER}

   309→   309→                        strokeDasharray="4 2"
   310→   310→                      />
   311→   311→                    ))}

   314→   314→                    <ReferenceLine
   315→   315→                      key={a.id}
   316→   316→                      x={(() => {
   317→   317→                        const idx = chartData.findIndex(
   318→   318→                          (p) => p.ms >= new Date(a.date).getTime()
   319→   319→                        );
   320→   320→                        return idx >= 0 ? idx + 1 : 1;
   321→   321→                      })()}
   322→   322→                      stroke={a.color ?? KIND_COLORS[a.kind] ?? KIND_COLORS.OTHER}
   323→   323→                      strokeWidth={a.endDate == null ? 2 : 0}
   324→   324→                      strokeDasharray="6 3"
   325→   325→                      label={{
   326→   326→                        value: a.label.slice(0, 14),

   328→   328→                        fill: "var(--muted-foreground)",
   329→   329→                        fontSize: 10,
   330→   330→                      }}
   331→   331→                      ifOverflow="extendDomain"
   332→   332→                    />
   333→   333→                  ))}
   334→   334→                  <Area
   335→   335→                    type="monotone"
   336→   336→                    dataKey="equity"
   337→   337→                    stroke="var(--chart-1)"
   338→   338→                    strokeWidth={2}
   339→   339→                    fill="var(--chart-1)"
   340→   340→                    fillOpacity={0.12}
   341→   341→                    dot={false}
   342→   342→                    activeDot={{ r: 4 }}
   343→   343→                  />
   344→   344→                </AreaChart>
   345→   345→              </ResponsiveContainer>
   346→   346→            </div>

   369→   369→                {t("notSignificant")}
   370→   370→              </p>
   371→   371→            )}
   372→   372→            <div className="grid grid-cols-3 gap-2 text-center text-xs">
   373→   373→              <div />
   374→   374→              <div className="font-medium">{t("before")}</div>
   375→   375→              <div className="font-medium">{t("after")}</div>

   377→   377→                { label: t("sTrades"), b: selectedStats.before.n, a: selectedStats.after.n },
   378→   378→                {
   379→   379→                  label: t("sWinRate"),
   380→   380→                  b: selectedStats.before.winRate,
   381→   381→                  a: selectedStats.after.winRate,
   382→   382→                  pct: true,
   383→   383→                },
   384→   384→                {
   385→   385→                  label: t("sExpectancy"),
   386→   386→                  b: selectedStats.before.expectancyR,
   387→   387→                  a: selectedStats.after.expectancyR,
   388→   388→                  r: true,
   389→   389→                },
   390→   390→                {
   391→   391→                  label: t("sNetPnl"),
   392→   392→                  b: selectedStats.before.netPnl,
   393→   393→                  a: selectedStats.after.netPnl,
   394→   394→                  money: true,
   395→   395→                },
   396→   396→                {
   397→   397→                  label: t("sMaxDd"),
   398→   398→                  b: selectedStats.before.maxDrawdown,
   399→   399→                  a: selectedStats.after.maxDrawdown,
   400→   400→                  money: true,
   401→   401→                },
   402→   402→              ].map((row) => (
   403→   403→                <React.Fragment key={row.label}>
   404→   404→                  <div className="text-left text-muted-foreground">{row.label}</div>
   405→   405→                  <div className="font-mono">{fmtSide(row.b, row, fmt, currency)}</div>

   407→   407→                </React.Fragment>
   408→   408→              ))}
   409→   409→            </div>
   410→   410→            {selected.note && (

   417→   417→      {/* List */}

   422→   422→        </CardHeader>
   423→   423→        <CardContent>
   424→   424→          {annotations.length === 0 ? (

   427→   427→            </p>
   428→   428→          ) : (
   429→   429→            <div className="space-y-2">
   430→   430→              {annotations.map((a) => (
   431→   431→                <div
   432→   432→                  key={a.id}
   433→   433→                  className="flex flex-wrap items-center gap-2 rounded-md border p-3"

   435→   435→                  <span

   439→   439→                      backgroundColor:
   440→   440→                        a.color ?? KIND_COLORS[a.kind] ?? KIND_COLORS.OTHER,

   442→   442→                  />
   443→   443→                  <div className="min-w-0 flex-1">
   444→   444→                    <p className="truncate text-sm font-medium">{a.label}</p>
   445→   445→                    <p className="text-xs text-muted-foreground">
   446→   446→                      {fmtDay(a.date)}
   447→   447→                      {a.endDate ? ` → ${fmtDay(a.endDate)}` : ""} ·{" "}

   451→   451→                        : a.scope === "ACCOUNT"

   458→   458→                          : strategies.data?.find((x) => x.id === a.strategyId)?.name ?? "?"}
   459→   459→                    </p>
   460→   460→                  </div>
   461→   461→                  <Button
   462→   462→                    variant="ghost"
   463→   463→                    size="sm"
   464→   464→                    aria-label={visible.has(a.id) ? t("hide") : t("show")}
   465→   465→                    onClick={() =>
   466→   466→                      setVisible((prev) => {
   467→   467→                        const next = new Set(prev);
   468→   468→                        if (next.has(a.id)) next.delete(a.id);
   469→   469→                        else next.add(a.id);
   470→   470→                        return next;
   471→   471→                      })
   472→   472→                    }
   473→   473→                  >
   474→   474→                    {visible.has(a.id) ? (
   475→   475→                      <Eye aria-hidden="true" />
   476→   476→                    ) : (
   477→   477→                      <EyeOff aria-hidden="true" />
   478→   478→                    )}
   479→   479→                  </Button>
   480→   480→                  <Button
   481→   481→                    variant="ghost"
   482→   482→                    size="sm"
   483→   483→                    onClick={() => setSelected(selected?.id === a.id ? null : a)}
   484→   484→                  >
   485→   485→                    {t("stats")}
   486→   486→                  </Button>
   487→   487→                  <Button variant="ghost" size="sm" onClick={() => setEditing(a)}>
   488→   488→                    <Pencil aria-hidden="true" />
   489→   489→                  </Button>
   490→   490→                  <Button
   491→   491→                    variant="ghost"
   492→   492→                    size="sm"
   493→   493→                    aria-label={tCommon("delete")}
   494→   494→                    onClick={() => void removeAnnotation.mutateAsync(a.id)}
   495→   495→                  >
   496→   496→                    <Trash2 aria-hidden="true" className="text-destructive" />
   497→   497→                  </Button>
   498→   498→                </div>
   499→   499→              ))}
   500→   500→            </div>
   501→   501→          )}
   502→   502→        </CardContent>
   503→   503→      </Card>
   504→   504→
   505→   505→      {/* Create / edit dialog */}
   506→   506→      <AnnotationDialog
   507→   507→        open={creating != null || editing != null}
   508→   508→        initialDate={creating ?? editing?.date ?? new Date().toISOString()}
   509→   509→        annotation={editing}
   510→   510→        accounts={accounts.data ?? []}
   511→   511→        strategies={strategies.data ?? []}
   512→   512→        onClose={() => {
   513→   513→          setCreating(null);
   514→   514→          setEditing(null);
   515→   515→        }}
   516→   516→      />
   517→   517→    </div>
   518→   518→  );
   519→   519→}

   521→   521→function fmtSide(
   522→   522→  v: number | null,
   523→   523→  row: { pct?: boolean; r?: boolean; money?: boolean },
   524→   524→  fmt: FormatApi,
   525→   525→  currency: string
   526→   526→): string {
   527→   527→  if (v == null) return "—";
   528→   528→  if (row.pct) return fmt.pct(v, 1);
   529→   529→  if (row.r) return `${fmtNum(v, 2)} R`;
   530→   530→  if (row.money) return fmt.money(v, currency, 0);
   531→   531→  return String(v);
   532→   532→}

   536→   536→function AnnotationDialog({
   537→   537→  open,
   538→   538→  initialDate,
   539→   539→  annotation,
   540→   540→  accounts,
   541→   541→  strategies,
   542→   542→  onClose,
   543→   543→}: {
   544→   544→  open: boolean;
   545→   545→  initialDate: string;
   546→   546→  annotation: Annotation | null;
   547→   547→  accounts: Array<{ id: string; name: string }>;
   548→   548→  strategies: Array<{ id: string; name: string }>;
   549→   549→  onClose: () => void;
   550→   550→}) {
   551→   551→  const t = useTranslations("chartsBuilder.annotations");
   552→   552→  const tCommon = useTranslations("common");
   553→   553→  const tRoot = useTranslations();
   554→   554→  const { toast } = useToast();
   555→   555→  const qc = useQueryClient();
   556→   556→
   557→   557→  const [label, setLabel] = React.useState("");
   558→   558→  const [note, setNote] = React.useState("");
   559→   559→  const [kind, setKind] = React.useState<string>("OTHER");
   560→   560→  const [scope, setScope] = React.useState<string>("GLOBAL");
   561→   561→  const [accountId, setAccountId] = React.useState("");
   562→   562→  const [strategyId, setStrategyId] = React.useState("");
   563→   563→  const [endDate, setEndDate] = React.useState("");
   564→   564→  const [busy, setBusy] = React.useState(false);
   565→   565→
   566→   566→  React.useEffect(() => {
   567→   567→    if (!open) return;
   568→   568→    setLabel(annotation?.label ?? "");
   569→   569→    setNote(annotation?.note ?? "");
   570→   570→    setKind(annotation?.kind ?? "OTHER");
   571→   571→    setScope(annotation?.scope ?? "GLOBAL");
   572→   572→    setAccountId(annotation?.accountId ?? "");
   573→   573→    setStrategyId(annotation?.strategyId ?? "");
   574→   574→    setEndDate(
   575→   575→      annotation?.endDate != null
   576→   576→        ? new Date(annotation.endDate).toISOString().slice(0, 10)
   577→   577→        : ""
   578→   578→    );
   579→   579→  }, [open, annotation]);
   580→   580→
   581→   581→  const submit = async () => {
   582→   582→    if (!label.trim()) return;
   583→   583→    setBusy(true);
   584→   584→    try {
   585→   585→      const body = {
   586→   586→        date: annotation?.date ?? initialDate,
   587→   587→        endDate: endDate ? new Date(`${endDate}T12:00:00.000Z`).toISOString() : null,
   588→   588→        label: label.trim(),
   589→   589→        note: note.trim() || null,
   590→   590→        kind,
   591→   591→        scope,
   592→   592→        accountId: scope === "ACCOUNT" ? accountId || null : null,
   593→   593→        strategyId: scope === "STRATEGY" ? strategyId || null : null,
   594→   594→      };
   595→   595→      if (annotation) {
   596→   596→        await apiFetch(`/api/charts/annotations/${annotation.id}`, {
   597→   597→          method: "PATCH",
   598→   598→          body: JSON.stringify(body),
   599→   599→        });
   600→   600→      } else {
   601→   601→        await apiFetch("/api/charts/annotations", {
   602→   602→          method: "POST",
   603→   603→          body: JSON.stringify(body),
   604→   604→        });
   605→   605→      }
   606→   606→      toast({ title: annotation ? t("updatedToast") : t("createdToast") });
   607→   607→      void qc.invalidateQueries({ queryKey: ["charts", "annotations"] });
   608→   608→      onClose();
   609→   609→    } catch (error) {
   610→   610→      const err = error instanceof ApiClientError ? error : null;
   611→   611→      toast({
   612→   612→        title: tCommon("error"),
   613→   613→        description: apiErrorMessage(
   614→   614→          tRoot,
   615→   615→          err?.code,
   616→   616→          error instanceof Error ? error.message : ""
   617→   617→        ),
   618→   618→        variant: "destructive",
   619→   619→      });
   620→   620→    } finally {
   621→   621→      setBusy(false);
   622→   622→    }
   623→   623→  };
   624→   624→
   625→   625→  const dayValue = new Date(initialDate).toISOString().slice(0, 10);
   626→   626→
   627→   627→  return (

   632→   632→            {annotation ? t("editTitle") : t("createTitle")}
   633→   633→          </DialogTitle>
   634→   634→          <DialogDescription>
   635→   635→            {annotation
   636→   636→              ? t("editDesc")
   637→   637→              : t("createDesc", { date: dayValue })}
   638→   638→          </DialogDescription>
   639→   639→        </DialogHeader>
   640→   640→        <div className="space-y-3">
   641→   641→          <div className="space-y-1.5">
   642→   642→            <Label htmlFor="ann-label">{t("labelLabel")}</Label>
   643→   643→            <Input
   644→   644→              id="ann-label"
   645→   645→              value={label}
   646→   646→              placeholder={t("labelPlaceholder")}
   647→   647→              onChange={(e) => setLabel(e.target.value)}
   648→   648→            />
   649→   649→          </div>
   650→   650→          <div className="grid grid-cols-2 gap-3">
   651→   651→            <div className="space-y-1.5">
   652→   652→              <Label>{t("kindLabel")}</Label>
