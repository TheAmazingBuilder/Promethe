     2→// Conventions:
     3→//  - snake_case DB columns via @map, camelCase in TS
     4→//  - Decimal for ALL money / price / lots / pip values (never floats)
     5→//  - SQLite has no enums: status-like columns are String, validated by Zod at the edge
     6→//  - Lists (timeframes, sessions, checklist results) are JSON-encoded strings
     7→//    (SQLite has no scalar lists / native JSON type in this Prisma version)

   329→  setupType       String? @map("setup_type")

   336→  exitPrice         Decimal?  @map("exit_price")
   337→  stopLoss          Decimal?  @map("stop_loss")
   338→  takeProfit        Decimal?  @map("take_profit")

   345→  lots                  Decimal  @default(0)
   346→  units                 Decimal? // = lots * contractSize
   347→  riskAmount            Decimal? @map("risk_amount")
   348→  riskPercent           Decimal? @map("risk_percent")

   353→  points      Decimal?

   356→  swap        Decimal  @default(0) // signed cost
   357→  fees        Decimal  @default(0) // signed cost
   358→  netPnl      Decimal? @map("net_pnl")
   359→  pnlCurrency String   @default("USD") @map("pnl_currency")
   360→  rMultiple   Decimal? @map("r_multiple")
   361→  plannedRR   Decimal? @map("planned_rr")
   362→  realizedRR  Decimal? @map("realized_rr")
   363→  resultType  String?  @map("result_type") // WIN | LOSS | BREAKEVEN
   364→  pnlSource   String   @default("CALCULATED") @map("pnl_source") // CALCULATED | MANUAL
   365→  exitReason  String?  @map("exit_reason") // TP_HIT | SL_HIT | MANUAL | TRAILING | TIME | BREAKEVEN_STOP | PARTIAL | OTHER
   366→

   369→  emotionDuring      String?  @map("emotion_during")

   372→  setupQualityRating Int?     @map("setup_quality_rating") // 1-5
   373→  executionRating    Int?     @map("execution_rating") // 1-5
   374→  followedPlan       Boolean? @map("followed_plan")
   375→  lessonLearned      String?  @map("lesson_learned")
   376→
