     1→// TradeJournal — Prisma schema (SQLite in sandbox / designed for Postgres parity)
     2→// Conventions:
     3→//  - snake_case DB columns via @map, camelCase in TS
     4→//  - Decimal for ALL money / price / lots / pip values (never floats)
     5→//  - SQLite has no enums: status-like columns are String, validated by Zod at the edge
     6→//  - Lists (timeframes, sessions, checklist results) are JSON-encoded strings
     7→//    (SQLite has no scalar lists / native JSON type in this Prisma version)
     8→

   304→model Trade {
   305→  id           String  @id @default(cuid())
   306→  userId       String  @map("user_id")
   307→  accountId    String  @map("account_id")
   308→  strategyId   String? @map("strategy_id")
   309→  instrumentId String  @map("instrument_id")
   310→  externalId   String? @map("external_id") // broker ticket — import dedupe
   311→
   312→  // Setup
   313→  direction       String // LONG | SHORT
   314→  orderType       String  @default("MARKET") @map("order_type") // MARKET | LIMIT | STOP | STOP_LIMIT
   315→  status          String  @default("OPEN") // OPEN | CLOSED | PENDING | CANCELLED
   316→  timeframe       String?
   317→  session         String? // ASIAN | LONDON | NEW_YORK | OVERLAP | OTHER (from entry time UTC)
   318→  setupType       String? @map("setup_type")
   319→  marketCondition String? @map("market_condition") // TRENDING | RANGING | VOLATILE | NEWS | OTHER
   320→
   321→  // Entry / exit (stored UTC, displayed in user timezone)
   322→  entryTime         DateTime  @map("entry_time")
   323→  exitTime          DateTime? @map("exit_time")
   324→  entryPrice        Decimal   @map("entry_price")
   325→  exitPrice         Decimal?  @map("exit_price")
   326→  stopLoss          Decimal?  @map("stop_loss")
   327→  takeProfit        Decimal?  @map("take_profit")
   328→  initialStopLoss   Decimal?  @map("initial_stop_loss")
   329→  initialTakeProfit Decimal?  @map("initial_take_profit")
   330→  mfePrice          Decimal?  @map("mfe_price") // max favorable excursion (price extreme)
   331→  maePrice          Decimal?  @map("mae_price") // max adverse excursion (price extreme)
   332→
   333→  // Size & risk
   334→  lots                  Decimal  @default(0)
   335→  units                 Decimal? // = lots * contractSize
   336→  riskAmount            Decimal? @map("risk_amount")
   337→  riskPercent           Decimal? @map("risk_percent")
   338→  accountBalanceAtEntry Decimal? @map("account_balance_at_entry")
   339→
   340→  // Results
   341→  pips        Decimal?
   342→  points      Decimal?
   343→  grossPnl    Decimal? @map("gross_pnl")
   344→  commission  Decimal  @default(0) // signed cost: positive = cost, negative = credit
   345→  swap        Decimal  @default(0) // signed cost
   346→  fees        Decimal  @default(0) // signed cost
   347→  netPnl      Decimal? @map("net_pnl")
   348→  pnlCurrency String   @default("USD") @map("pnl_currency")
   349→  rMultiple   Decimal? @map("r_multiple")
   350→  plannedRR   Decimal? @map("planned_rr")
   351→  realizedRR  Decimal? @map("realized_rr")
   352→  resultType  String?  @map("result_type") // WIN | LOSS | BREAKEVEN
   353→  pnlSource   String   @default("CALCULATED") @map("pnl_source") // CALCULATED | MANUAL
   354→  exitReason  String?  @map("exit_reason") // TP_HIT | SL_HIT | MANUAL | TRAILING | TIME | BREAKEVEN_STOP | PARTIAL | OTHER
   355→
   356→  // Psychology & quality
   357→  emotionBefore      String?  @map("emotion_before")
   358→  emotionDuring      String?  @map("emotion_during")
   359→  emotionAfter       String?  @map("emotion_after")
   360→  confidence         Int? // 1-5
   361→  setupQualityRating Int?     @map("setup_quality_rating") // 1-5
   362→  executionRating    Int?     @map("execution_rating") // 1-5
   363→  followedPlan       Boolean? @map("followed_plan")
   364→  lessonLearned      String?  @map("lesson_learned")
   365→
   366→  // Review workflow (addendum §11.3)
   367→  reviewStatus String  @default("NOT_REVIEWED") @map("review_status") // NOT_REVIEWED | REVIEWED
   368→  isDraft      Boolean @default(false) @map("is_draft") // autosaved drafts persist as Trade rows
   369→
   370→  // v2 — offline idempotency: dedupe key for the sync engine (4.2)
   371→  clientMutationId   String?  @unique @map("client_mutation_id")
   372→  /// v2 — planned risk (strategy/template) vs actual riskPercent (4.9)
   373→  plannedRiskPercent Decimal? @map("planned_risk_percent")
   374→
   375→  // ── v3 — broker sync / import provenance (schema §3.1) ──
   376→  /// MANUAL | EA_SYNC | FILE_IMPORT
   377→  source         String  @default("MANUAL")
   378→  /// Broker position identifier (MT5 positionId / stitched MT4 chain head)
   379→  positionId     String? @map("position_id")
   380→  /// EA magic number (strategy auto-tagging via MagicNumberMapping)
   381→  magicNumber    String? @map("magic_number")
   382→  /// Raw broker comment on the order/position
   383→  brokerComment  String? @map("broker_comment")
   384→  /// Synced/imported trade still missing strategy/notes → review queue
   385→  needsDetails   Boolean @default(false) @map("needs_details")
   386→  /// NONE | MANUAL | EA_TRACKED — where MFE/MAE extremes come from
   387→  mfeMaeSource   String  @default("NONE") @map("mfe_mae_source")
   388→  /// True when the EA tracked the position for its whole lifetime
   389→  mfeMaeComplete Boolean @default(false) @map("mfe_mae_complete")
   390→
   391→  // Notes (markdown)
   392→  preTradeAnalysis String  @map("pre_trade_analysis")
   393→  postTradeReview  String  @map("post_trade_review")
   394→  notes            String?
   395→
   396→  // Links
   397→  tradingViewUrl   String? @map("trading_view_url")
   398→  checklistResults String? @map("checklist_results") // JSON: { [itemId]: boolean }
   399→
   400→  createdAt DateTime @default(now()) @map("created_at")
   401→  updatedAt DateTime @updatedAt @map("updated_at")
   402→
   403→  user           User                 @relation(fields: [userId], references: [id], onDelete: Cascade)
   404→  account        Account              @relation(fields: [accountId], references: [id], onDelete: Cascade)
   405→  strategy       Strategy?            @relation(fields: [strategyId], references: [id], onDelete: SetNull)
   406→  instrument     Instrument           @relation(fields: [instrumentId], references: [id])
   407→  executions     TradeExecution[]
   408→  tags           TradeTag[]
   409→  screenshots    TradeScreenshot[]
   410→  ruleViolations TradeRuleViolation[]
   411→
   412→  // ── v2 ──
   413→  review        TradeReview?
   414→  eventLinks    TradeEventLink[]
   415→  marketContext MarketContextSnapshot?
   416→
   417→  // ── v3 ──
   418→  modifications TradeModification[]
   419→
   420→  @@unique([accountId, externalId])
   421→  @@index([accountId, positionId])
   422→  @@index([userId, entryTime])
   423→  @@index([userId, strategyId])
