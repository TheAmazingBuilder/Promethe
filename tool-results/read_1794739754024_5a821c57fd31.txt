   101→  return (
   102→    <div className="flex flex-col gap-2 rounded-lg border p-4">
   103→      <p className="text-xs font-medium text-muted-foreground">{label}</p>
   104→      <p
   105→        className={`tabular-nums ${
   106→          compact ? "text-xl font-semibold" : "text-2xl font-bold"
   107→        } ${valueClassName ?? ""}`}
   108→      >
   109→        {value}
   110→      </p>
   111→      {sub != null ? (
   112→        <p className="text-xs tabular-nums text-muted-foreground">{sub}</p>
   113→      ) : null}
   114→    </div>
   115→  );
   116→}

   130→}: {
   131→  active?: boolean;
   132→  payload?: Array<{ payload?: EquityDatum }>;

   178→/** One breakdown dimension — compact rows with a proportional net P&L bar. */
   179→function BreakdownCard({
   180→  title,
   181→  description,
   182→  icon: Icon,
   183→  entries,
   184→  currency,
   185→  renderKey,
   186→  max = 8,
   187→}: {
   188→  title: string;
   189→  description?: string;
   190→  icon: React.ComponentType<{ className?: string }>;
   191→  entries: GroupStat[];
   192→  currency: string;
   193→  renderKey: (key: string) => string;
   194→  max?: number;
   195→}) {
   196→  const t = useTranslations("analytics");
   197→  const fmt = useFormat();
   198→  const shown = entries.slice(0, max);
   199→  const maxAbs = shown.reduce(
   200→    (m, e) => Math.max(m, Math.abs(Number(e.stats.netPnl))),
   201→    0
   202→  );
   203→  return (
   204→    <Card>
   205→      <CardHeader>
   206→        <CardTitle className="flex items-center gap-2 text-base">
   207→          <Icon className="h-4 w-4 text-muted-foreground" aria-hidden="true" />
   208→          {title}
   209→        </CardTitle>
   210→        {description ? <CardDescription>{description}</CardDescription> : null}
   211→      </CardHeader>
   212→      <CardContent>
   213→        {shown.length === 0 ? (
   214→          <p className="text-sm text-muted-foreground">{t("noData")}</p>
   215→        ) : (
   216→          <ul className="space-y-1.5">
   217→            {shown.map((e) => {
   218→              const net = Number(e.stats.netPnl);
   219→              const width =
   220→                maxAbs > 0 ? Math.max(2, (Math.abs(net) / maxAbs) * 100) : 0;
   221→              return (
   222→                <li
   223→                  key={e.key}
   224→                  className="relative flex items-center justify-between gap-2 overflow-hidden rounded-md px-2 py-1.5 text-sm"
   225→                >
   226→                  <span
   227→                    className={`absolute inset-y-0 left-0 ${
   228→                      net >= 0 ? "bg-gain-subtle" : "bg-loss-subtle"
   229→                    }`}
   230→                    style={{ width: `${width}%` }}
   231→                    aria-hidden="true"
   232→                  />
   233→                  <span className="relative z-10 min-w-0 truncate font-medium">
   234→                    {renderKey(e.key)}
   235→                  </span>
   236→                  <span className="relative z-10 flex shrink-0 items-center gap-3 tabular-nums">
   237→                    <span className="text-xs text-muted-foreground">
   238→                      ×{e.stats.tradeCount}
   239→                    </span>
   240→                    <span className="hidden w-16 whitespace-nowrap text-right text-xs text-muted-foreground sm:inline">
   241→                      {e.stats.winRate != null
   242→                        ? fmt.pct(e.stats.winRate)
   243→                        : "—"}
   244→                    </span>
   245→                    <Money
   246→                      signed
   247→                      value={e.stats.netPnl}
   248→                      currency={currency}
   249→                      className={`w-24 text-right font-medium ${pnlClass(
   250→                        e.stats.netPnl
   251→                      )}`}
   252→                    />
   253→                  </span>
   254→                </li>
   255→              );
   256→            })}
   257→            {entries.length > shown.length ? (
   258→              <li className="px-2 pt-1 text-xs text-muted-foreground">
   259→                {t("more", { count: entries.length - shown.length })}
   260→              </li>
   261→            ) : null}
   262→          </ul>
   263→        )}
   264→      </CardContent>
   265→    </Card>
   266→  );
   267→}

   271→export function AnalyticsView() {
   272→  const { session } = useSession();
   273→  const t = useTranslations("analytics");
   274→  const tRoot = useTranslations();
   275→  const labels = useTradingLabels();
   276→  const { locale } = useI18n();
   277→  const fmt = useFormat();
   278→  const [period, setPeriod] = React.useState<AnalyticsPeriod>("30d");
   279→  const [accountId, setAccountId] = React.useState<string>("all");
   280→
   281→  const { data: accountsData } = useAccounts();
   282→  const accounts = accountsData ?? [];
   283→
   284→  const effectiveAccount = accountId === "all" ? undefined : accountId;
   285→  const { data, isLoading, isError, refetch } = useAnalytics({
   286→    period,
   287→    account: effectiveAccount,
   288→  });
   289→
   290→  const timezone = data?.timezone ?? session?.profile.timezone ?? "UTC";
   291→  const currency = data?.currency ?? session?.profile.baseCurrency ?? "USD";
   292→  const stats = data?.stats ?? null;
   293→  const advanced = data?.advanced ?? null;
   294→
   295→  const equityData = React.useMemo<EquityDatum[]>(
   296→    () =>
   297→      (data?.equityCurve ?? []).map((p) => ({
   298→        i: p.i,
   299→        date: p.date,
   300→        equity: Number(p.equity),
   301→        drawdown: Number(p.drawdown),
   302→      })),
   303→    [data]
   304→  );
   305→
   306→  const rDistData = React.useMemo(
   307→    () =>
   308→      (data?.rDistribution.buckets ?? []).map((b) => ({
   309→        id: b.id,
   310→        label: t(`rdist.${b.id}`),
   311→        count: b.count,
   312→        totalR: Number(b.totalR),
   313→        negative: b.to != null ? b.to <= 0 : false,
   314→      })),
   315→    [data, t]
   316→  );
   317→
   318→  const weekdayLabel = React.useCallback(
   319→    (key: string) => {
   320→      try {
   321→        // ISO weekday 1-7 → localized name via a stable reference week
   322→        // (2024-01-01 was a Monday).
   323→        const ref = new Date(Date.UTC(2024, 0, Number(key)));
   324→        return new Intl.DateTimeFormat(locale, {
   325→          weekday: "long",
   326→          timeZone: "UTC",
   327→        }).format(ref);
   328→      } catch {
   329→        return key;
   330→      }
   331→    },
   332→    [locale]
   333→  );
   334→
   335→  const hourLabel = React.useCallback((key: string) => {
   336→    const h = Number(key);
   337→    if (!Number.isFinite(h)) return key;
   338→    const two = (n: number) => String(n).padStart(2, "0");
   339→    return `${two(h)}:00–${two(h + 1)}:59`;
   340→  }, []);
   341→
   342→  /** Formats raw insight params for i18n interpolation (money, R, labels…). */
   343→  const formatInsightParams = React.useCallback(
   344→    (insight: Insight): Record<string, string> => {
   345→      const money = new Set(["netPnl", "fees", "dd", "pnl"]);
   346→      const pct = new Set(["winRate", "pct"]);
   347→      const r = new Set([
   348→        "avgR",
   349→        "avgRPlan",
   350→        "avgRNoPlan",
   351→        "busyAvgR",
   352→        "lightAvgR",
   353→      ]);
   354→      const num = new Set(["payoff", "rf"]);
   355→      const out: Record<string, string> = {};
   356→      for (const [k, v] of Object.entries(insight.params)) {
   357→        if (money.has(k) && typeof v === "string") {
   358→          out[k] = fmt.signedMoney(v, currency);
   359→        } else if (pct.has(k) && typeof v === "string") {
   360→          out[k] = fmt.pct(v);
   361→        } else if (r.has(k) && typeof v === "string") {
   362→          out[k] = `${fmtSigned(v)} R`;
   363→        } else if (num.has(k) && typeof v === "string") {
   364→          out[k] = fmtNum(v);
   365→        } else if (k === "weekday") {
   366→          out[k] = weekdayLabel(String(v));
   367→        } else if (k === "hour") {
   368→          out[k] = hourLabel(String(v));
   369→        } else if (k === "session") {
   370→          out[k] = labels.session(String(v));
   371→        } else if (k === "emotion") {
   372→          out[k] = labels.emotion(String(v));
   373→        } else {
   374→          out[k] = String(v);
   375→        }
   376→      }
   377→      return out;
   378→    },
   379→    [currency, weekdayLabel, hourLabel, labels, fmt]
   380→  );
   381→
   382→  const renderInsight = (insight: Insight) =>
   383→    tRoot(`analytics.insights.${insight.rule}` as Parameters<typeof tRoot>[0], {
   384→      ...formatInsightParams(insight),
   385→    });
   386→
   387→  const goodInsights = (data?.insights ?? []).filter((i) => i.tone === "good");
   388→  const badInsights = (data?.insights ?? []).filter((i) => i.tone === "bad");
   389→  const infoInsights = (data?.insights ?? []).filter((i) => i.tone === "info");
   390→
   391→  /** KPI display model. */
   392→  const kpi = React.useMemo(() => {
   393→    const s = stats;
   394→    const pfNum = s?.profitFactor != null ? Number(s.profitFactor) : null;
   395→    const streak = s?.currentStreak ?? 0;
   396→    return {
   397→      netPnlTone: s ? pnlClass(s.netPnl) : "text-muted-foreground",
   398→      winRate: s?.winRate != null ? fmt.pct(s.winRate) : "—",
   399→      profitFactor:
   400→        pfNum == null
   401→          ? "—"
   402→          : Number.isFinite(pfNum)
   403→            ? fmtNum(pfNum)
   404→            : "∞",
   405→      profitFactorTone:
   406→        pfNum == null
   407→          ? ""
   408→          : pfNum >= 2
   409→            ? EMERALD
   410→            : pfNum < 1
   411→              ? ROSE
   412→              : "",
   413→      expectancy: s?.expectancyR != null ? `${fmtSigned(s.expectancyR)} R` : "—",
   414→      expectancyTone:
   415→        s?.expectancyR != null ? pnlClass(s.expectancyR) : "text-muted-foreground",
   416→      totalR: s?.totalR != null ? `${fmtSigned(s.totalR)} R` : "—",
   417→      maxDrawdownTone: s?.maxDrawdown != null ? ROSE : "",
   418→      sqn: advanced?.sqn != null ? fmtNum(advanced.sqn) : "—",
   419→      sqnTone:
   420→        advanced?.sqn == null
   421→          ? ""
   422→          : Number(advanced.sqn) >= 2
   423→            ? EMERALD
   424→            : Number(advanced.sqn) < 1
   425→              ? ROSE
   426→              : "",
   427→      tradeCount: s ? fmtNum(s.tradeCount, 0) : "—",
   428→      winsLosses: s
   429→        ? t("kpiWinsLosses", { wins: s.wins, losses: s.losses, be: s.breakevens })
   430→        : null,
   431→      sharpe: advanced?.sharpe != null ? fmtNum(advanced.sharpe) : "—",
   432→      sortino: advanced?.sortino != null ? fmtNum(advanced.sortino) : "—",
   433→      recovery:
   434→        advanced?.recoveryFactor != null ? fmtNum(advanced.recoveryFactor) : "—",
   435→      payoff: advanced?.payoffRatio != null ? fmtNum(advanced.payoffRatio) : "—",
   436→      avgDuration:
   437→        advanced?.avgDurationMin != null
   438→          ? advanced.avgDurationMin >= 60
   439→            ? t("durationHm", {
   440→                h: Math.floor(advanced.avgDurationMin / 60),
   441→                m: advanced.avgDurationMin % 60,
   442→              })
   443→            : t("durationM", { m: advanced.avgDurationMin })
   444→          : "—",
   445→      rCoverage: advanced?.rCoverage != null ? fmt.pct(advanced.rCoverage) : "—",
   446→      maxWinStreak: s ? fmtNum(s.maxWinStreak, 0) : "—",
   447→      maxLossStreak: s ? fmtNum(s.maxLossStreak, 0) : "—",
   448→      streakText:
   449→        streak > 0
   450→          ? t("streakWin", { count: streak })
   451→          : streak < 0
   452→            ? t("streakLoss", { count: Math.abs(streak) })
   453→            : "—",
   454→      streakTone: streak > 0 ? EMERALD : streak < 0 ? ROSE : "",
   455→    };
   456→  }, [stats, advanced, currency, t, fmt]);
   457→
   458→  if (!session) return null;
   459→
   460→  const hasAnyTrade = stats != null;
   461→  const showAllCta = !hasAnyTrade && period !== "all";
   462→
   463→  return (
   464→    <div className="space-y-6">
   465→      {/* ── Header + filters ── */}
   466→      <div className="flex flex-col gap-3">
   467→        <div className="flex flex-col gap-1">
   468→          <h2 className="text-xl font-bold tracking-tight sm:text-2xl">
   469→            {t("title")}
   470→          </h2>
   471→          <p className="text-sm text-muted-foreground">
   472→            {hasAnyTrade
   473→              ? t("subtitle", { count: stats.tradeCount })
   474→              : t("subtitleEmpty")}
   475→          </p>
   476→        </div>
   477→        <div className="flex flex-wrap items-center gap-2">
   478→          <Tabs
   479→            value={period}
   480→            onValueChange={(v) => {
   481→              if (PERIODS.includes(v as AnalyticsPeriod)) {
   482→                setPeriod(v as AnalyticsPeriod);
   483→              }
   484→            }}
   485→          >
   486→            <TabsList>
   487→              <TabsTrigger value="30d">{t("period30d")}</TabsTrigger>
   488→              <TabsTrigger value="90d">{t("period90d")}</TabsTrigger>
   489→              <TabsTrigger value="ytd">{t("periodYtd")}</TabsTrigger>
   490→              <TabsTrigger value="all">{t("periodAll")}</TabsTrigger>
   491→            </TabsList>
   492→          </Tabs>
   493→          <Select
   494→            value={accountId}
   495→            onValueChange={(v) => setAccountId(v === "all" ? "all" : v)}
   496→          >
   497→            <SelectTrigger className="w-[180px]" aria-label={t("accountFilter")}>
   498→              <SelectValue />
   499→            </SelectTrigger>
   500→            <SelectContent>
   501→              <SelectItem value="all">{t("allAccounts")}</SelectItem>
   502→              {accounts.map((a) => (
   503→                <SelectItem key={a.id} value={a.id}>
   504→                  <Sensitive kind="account">{a.name}</Sensitive>
   505→                </SelectItem>
   506→              ))}
   507→            </SelectContent>
   508→          </Select>
   509→          {/* v2.5 M2 — chart builder deep link */}
   510→          <Button
   511→            variant="outline"
   512→            size="sm"
   513→            className="ml-auto"
   514→            onClick={() => navigate("#/analytics/charts")}
   515→          >
   516→            <LineChart className="h-4 w-4" aria-hidden="true" />
   517→            {t("openChartBuilder")}
   518→          </Button>
   519→        </div>
   520→      </div>
   521→
   522→      {/* ── Loading ── */}
   523→      {isLoading ? (
   524→        <div className="space-y-6" role="status" aria-live="polite">
   525→          <span className="sr-only">{t("loading")}</span>
   526→          <div className="grid grid-cols-2 gap-3 lg:grid-cols-4">
   527→            {Array.from({ length: 8 }, (_, i) => (
   528→              <Skeleton key={i} className="h-24 w-full rounded-lg" />
   529→            ))}
   530→          </div>
   531→          <Skeleton className="h-72 w-full rounded-xl" />
   532→          <div className="grid gap-6 lg:grid-cols-2">
   533→            <Skeleton className="h-64 w-full rounded-xl" />
   534→            <Skeleton className="h-64 w-full rounded-xl" />
   535→          </div>
   536→        </div>
   537→      ) : isError ? (
   538→        /* ── Error ── */
   539→        <Card>
   540→          <CardContent className="flex flex-col items-start gap-3">
   541→            <div className="flex items-center gap-2">
   542→              <AlertTriangle className="h-5 w-5 text-destructive" aria-hidden="true" />
   543→              <p className="text-sm font-medium">{t("errorTitle")}</p>
   544→            </div>
   545→            <Button variant="outline" size="sm" onClick={() => void refetch()}>
   546→              <RotateCw className="h-4 w-4" aria-hidden="true" />
   547→              {t("errorRetry")}
   548→            </Button>
   549→          </CardContent>
   550→        </Card>
   551→      ) : !hasAnyTrade || !data ? (
   552→        /* ── Empty state ── */
   553→        <Card>
   554→          <CardHeader>
   555→            <CardTitle className="text-lg">{t("emptyTitle")}</CardTitle>
   556→            <CardDescription>{t("emptyDesc")}</CardDescription>
   557→          </CardHeader>
   558→          <CardContent className="flex flex-col gap-2 sm:flex-row sm:flex-wrap">
   559→            {showAllCta ? (
   560→              <Button onClick={() => setPeriod("all")}>
   561→                <BarChart3 className="h-4 w-4" aria-hidden="true" />
   562→                {t("emptyCtaAll")}
   563→              </Button>
   564→            ) : (
   565→              <Button onClick={() => navigate("#/trades/quick")}>
   566→                <Zap className="h-4 w-4" aria-hidden="true" />
   567→                {t("emptyCtaQuick")}
   568→              </Button>
