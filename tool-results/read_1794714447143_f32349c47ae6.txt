    88→function KpiCard({
    89→  label,
    90→  value,
    91→  valueClassName,
    92→  sub,
    93→  compact = false,
    94→}: {
    95→  label: string;
    96→  value: React.ReactNode;
    97→  valueClassName?: string;
    98→  sub?: React.ReactNode;
    99→  compact?: boolean;
   100→}) {
   101→  return (
   102→    <div className="flex flex-col gap-2 rounded-lg border p-4">
   103→      <p className="text-xs font-medium text-muted-foreground">{label}</p>
   104→      <p
   105→        className={`tabular-nums ${
   106→          compact ? "text-xl font-semibold" : "text-2xl font-bold"
   107→        } ${valueClassName ?? ""}`}
   108→      >
   109→        {value}
   110→      </p>
   111→      {sub != null ? (
   112→        <p className="text-xs tabular-nums text-muted-foreground">{sub}</p>
   113→      ) : null}
   114→    </div>
   115→  );
   116→}

   125→function EquityTooltipContent({
   126→  active,
   127→  payload,
   128→  currency,
   129→  timezone,
   130→}: {
   131→  active?: boolean;
   132→  payload?: Array<{ payload?: EquityDatum }>;
   133→  currency: string;
   134→  timezone: string;
   135→}) {
   136→  if (!active) return null;
   137→  const point = payload?.[0]?.payload;
   138→  if (!point) return null;

   140→    <div className="rounded-lg border bg-background px-3 py-2 text-xs shadow-md">
   141→      <p className="font-medium">#{point.i}</p>
   142→      <p className="text-muted-foreground">
   143→        {fmtDateInTz(point.date, timezone)}
   144→      </p>
   145→      <p className="mt-1 tabular-nums">{fmtMoney(point.equity, currency)}</p>
   146→      {point.drawdown > 0 ? (
   147→        <p className={`tabular-nums ${ROSE}`}>
   148→          −{fmtMoney(point.drawdown, currency)}
   149→        </p>
   150→      ) : null}
   151→    </div>
   152→  );

   176→/** One breakdown dimension — compact rows with a proportional net P&L bar. */
   177→function BreakdownCard({
   178→  title,
   179→  description,
   180→  icon: Icon,
   181→  entries,
   182→  currency,
   183→  renderKey,
   184→  max = 8,
   185→}: {
   186→  title: string;
   187→  description?: string;
   188→  icon: React.ComponentType<{ className?: string }>;
   189→  entries: GroupStat[];
   190→  currency: string;
   191→  renderKey: (key: string) => string;
   192→  max?: number;
   193→}) {
   194→  const t = useTranslations("analytics");
   195→  const shown = entries.slice(0, max);
   196→  const maxAbs = shown.reduce(
   197→    (m, e) => Math.max(m, Math.abs(Number(e.stats.netPnl))),
   198→    0
   199→  );
   200→  return (
   201→    <Card>
   202→      <CardHeader>
   203→        <CardTitle className="flex items-center gap-2 text-base">
   204→          <Icon className="h-4 w-4 text-muted-foreground" aria-hidden="true" />
   205→          {title}
   206→        </CardTitle>
   207→        {description ? <CardDescription>{description}</CardDescription> : null}
   208→      </CardHeader>
   209→      <CardContent>
   210→        {shown.length === 0 ? (
   211→          <p className="text-sm text-muted-foreground">{t("noData")}</p>
   212→        ) : (
   213→          <ul className="space-y-1.5">
   214→            {shown.map((e) => {
   215→              const net = Number(e.stats.netPnl);
   216→              const width =
   217→                maxAbs > 0 ? Math.max(2, (Math.abs(net) / maxAbs) * 100) : 0;
   218→              return (
   219→                <li
   220→                  key={e.key}
   221→                  className="relative flex items-center justify-between gap-2 overflow-hidden rounded-md px-2 py-1.5 text-sm"
   222→                >
   223→                  <span
   224→                    className={`absolute inset-y-0 left-0 ${
   225→                      net >= 0 ? "bg-gain-subtle" : "bg-loss-subtle"
   226→                    }`}
   227→                    style={{ width: `${width}%` }}
   228→                    aria-hidden="true"
   229→                  />
   230→                  <span className="relative z-10 min-w-0 truncate font-medium">
   231→                    {renderKey(e.key)}
   232→                  </span>
   233→                  <span className="relative z-10 flex shrink-0 items-center gap-3 tabular-nums">
   234→                    <span className="text-xs text-muted-foreground">
   235→                      ×{e.stats.tradeCount}
   236→                    </span>
   237→                    <span className="hidden w-16 whitespace-nowrap text-right text-xs text-muted-foreground sm:inline">
   238→                      {e.stats.winRate != null
   239→                        ? fmtPct(e.stats.winRate)
   240→                        : "—"}
   241→                    </span>
   242→                    <span
   243→                      className={`w-24 text-right font-medium ${pnlClass(
   244→                        e.stats.netPnl
   245→                      )}`}
   246→                    >
   247→                      {fmtSignedMoney(e.stats.netPnl, currency)}
   248→                    </span>
   249→                  </span>
   250→                </li>
   251→              );
   252→            })}
   253→            {entries.length > shown.length ? (
   254→              <li className="px-2 pt-1 text-xs text-muted-foreground">
   255→                {t("more", { count: entries.length - shown.length })}
   256→              </li>
   257→            ) : null}
   258→          </ul>
   259→        )}
   260→      </CardContent>
   261→    </Card>
   262→  );
   263→}

   267→export function AnalyticsView() {
   268→  const { session } = useSession();
   269→  const t = useTranslations("analytics");
   270→  const tRoot = useTranslations();
   271→  const labels = useTradingLabels();
   272→  const { locale } = useI18n();
   273→  const [period, setPeriod] = React.useState<AnalyticsPeriod>("30d");
   274→  const [accountId, setAccountId] = React.useState<string>("all");
   275→
   276→  const { data: accountsData } = useAccounts();
   277→  const accounts = accountsData ?? [];
   278→
   279→  const effectiveAccount = accountId === "all" ? undefined : accountId;
   280→  const { data, isLoading, isError, refetch } = useAnalytics({
   281→    period,
   282→    account: effectiveAccount,
   283→  });
   284→
   285→  const timezone = data?.timezone ?? session?.profile.timezone ?? "UTC";
   286→  const currency = data?.currency ?? session?.profile.baseCurrency ?? "USD";
   287→  const stats = data?.stats ?? null;
   288→  const advanced = data?.advanced ?? null;
   289→
   290→  const equityData = React.useMemo<EquityDatum[]>(
   291→    () =>
   292→      (data?.equityCurve ?? []).map((p) => ({
   293→        i: p.i,
   294→        date: p.date,
   295→        equity: Number(p.equity),
   296→        drawdown: Number(p.drawdown),
   297→      })),
   298→    [data]
   299→  );
   300→
   301→  const rDistData = React.useMemo(
   302→    () =>
   303→      (data?.rDistribution.buckets ?? []).map((b) => ({
   304→        id: b.id,
   305→        label: t(`rdist.${b.id}`),
   306→        count: b.count,
   307→        totalR: Number(b.totalR),
   308→        negative: b.to != null ? b.to <= 0 : false,
   309→      })),
   310→    [data, t]
   311→  );
   312→
   313→  const weekdayLabel = React.useCallback(
   314→    (key: string) => {
   315→      try {
   316→        // ISO weekday 1-7 → localized name via a stable reference week
   317→        // (2024-01-01 was a Monday).
   318→        const ref = new Date(Date.UTC(2024, 0, Number(key)));
   319→        return new Intl.DateTimeFormat(locale, {
   320→          weekday: "long",
   321→          timeZone: "UTC",
   322→        }).format(ref);
   323→      } catch {
   324→        return key;
   325→      }
   326→    },
   327→    [locale]
   328→  );
   329→
   330→  const hourLabel = React.useCallback((key: string) => {
   331→    const h = Number(key);
   332→    if (!Number.isFinite(h)) return key;
   333→    const two = (n: number) => String(n).padStart(2, "0");
   334→    return `${two(h)}:00–${two(h + 1)}:59`;
   335→  }, []);
   336→
   337→  /** Formats raw insight params for i18n interpolation (money, R, labels…). */
   338→  const formatInsightParams = React.useCallback(
   339→    (insight: Insight): Record<string, string> => {
   340→      const money = new Set(["netPnl", "fees", "dd", "pnl"]);
   341→      const pct = new Set(["winRate", "pct"]);
   342→      const r = new Set([
   343→        "avgR",
   344→        "avgRPlan",
   345→        "avgRNoPlan",
   346→        "busyAvgR",
   347→        "lightAvgR",
   348→      ]);
   349→      const num = new Set(["payoff", "rf"]);
   350→      const out: Record<string, string> = {};
   351→      for (const [k, v] of Object.entries(insight.params)) {
   352→        if (money.has(k) && typeof v === "string") {
   353→          out[k] = fmtSignedMoney(v, currency);
   354→        } else if (pct.has(k) && typeof v === "string") {
   355→          out[k] = fmtPct(v);
   356→        } else if (r.has(k) && typeof v === "string") {
   357→          out[k] = `${fmtSigned(v)} R`;
   358→        } else if (num.has(k) && typeof v === "string") {
   359→          out[k] = fmtNum(v);
   360→        } else if (k === "weekday") {
   361→          out[k] = weekdayLabel(String(v));
   362→        } else if (k === "hour") {
   363→          out[k] = hourLabel(String(v));
   364→        } else if (k === "session") {
   365→          out[k] = labels.session(String(v));
   366→        } else if (k === "emotion") {
   367→          out[k] = labels.emotion(String(v));
   368→        } else {
   369→          out[k] = String(v);
   370→        }
   371→      }
   372→      return out;
   373→    },
   374→    [currency, weekdayLabel, hourLabel, labels]
   375→  );
   376→
   377→  const renderInsight = (insight: Insight) =>
   378→    tRoot(`analytics.insights.${insight.rule}` as Parameters<typeof tRoot>[0], {
   379→      ...formatInsightParams(insight),
   380→    });
   381→
   382→  const goodInsights = (data?.insights ?? []).filter((i) => i.tone === "good");
   383→  const badInsights = (data?.insights ?? []).filter((i) => i.tone === "bad");
   384→  const infoInsights = (data?.insights ?? []).filter((i) => i.tone === "info");
   385→
   386→  /** KPI display model. */
   387→  const kpi = React.useMemo(() => {
   388→    const s = stats;
   389→    const pfNum = s?.profitFactor != null ? Number(s.profitFactor) : null;
   390→    const streak = s?.currentStreak ?? 0;
   391→    return {
   392→      netPnl: s ? fmtSignedMoney(s.netPnl, currency) : "—",
   393→      netPnlTone: s ? pnlClass(s.netPnl) : "text-muted-foreground",
   394→      winRate: s?.winRate != null ? fmtPct(s.winRate) : "—",
   395→      profitFactor:
   396→        pfNum == null
   397→          ? "—"
   398→          : Number.isFinite(pfNum)
   399→            ? fmtNum(pfNum)
   400→            : "∞",
   401→      profitFactorTone:
   402→        pfNum == null
   403→          ? ""
   404→          : pfNum >= 2
   405→            ? EMERALD
   406→            : pfNum < 1
   407→              ? ROSE
   408→              : "",
   409→      expectancy: s?.expectancyR != null ? `${fmtSigned(s.expectancyR)} R` : "—",
   410→      expectancyTone:
   411→        s?.expectancyR != null ? pnlClass(s.expectancyR) : "text-muted-foreground",
   412→      totalR: s?.totalR != null ? `${fmtSigned(s.totalR)} R` : "—",
   413→      maxDrawdown: s?.maxDrawdown != null ? fmtMoney(s.maxDrawdown, currency) : "—",
   414→      maxDrawdownTone: s?.maxDrawdown != null ? ROSE : "",
   415→      sqn: advanced?.sqn != null ? fmtNum(advanced.sqn) : "—",
   416→      sqnTone:
   417→        advanced?.sqn == null
   418→          ? ""
   419→          : Number(advanced.sqn) >= 2
   420→            ? EMERALD
   421→            : Number(advanced.sqn) < 1
   422→              ? ROSE
   423→              : "",
   424→      tradeCount: s ? fmtNum(s.tradeCount, 0) : "—",
   425→      winsLosses: s
   426→        ? t("kpiWinsLosses", { wins: s.wins, losses: s.losses, be: s.breakevens })
   427→        : null,
   428→      sharpe: advanced?.sharpe != null ? fmtNum(advanced.sharpe) : "—",
   429→      sortino: advanced?.sortino != null ? fmtNum(advanced.sortino) : "—",
   430→      recovery:
   431→        advanced?.recoveryFactor != null ? fmtNum(advanced.recoveryFactor) : "—",
   432→      payoff: advanced?.payoffRatio != null ? fmtNum(advanced.payoffRatio) : "—",
   433→      avgDuration:
   434→        advanced?.avgDurationMin != null
   435→          ? advanced.avgDurationMin >= 60
   436→            ? t("durationHm", {
   437→                h: Math.floor(advanced.avgDurationMin / 60),
   438→                m: advanced.avgDurationMin % 60,
   439→              })
   440→            : t("durationM", { m: advanced.avgDurationMin })
   441→          : "—",
   442→      totalFees:
   443→        advanced != null ? fmtMoney(advanced.totalFees, currency) : "—",
   444→      rCoverage: advanced?.rCoverage != null ? fmtPct(advanced.rCoverage) : "—",
   445→      maxWinStreak: s ? fmtNum(s.maxWinStreak, 0) : "—",
   446→      maxLossStreak: s ? fmtNum(s.maxLossStreak, 0) : "—",
   447→      streakText:
   448→        streak > 0
   449→          ? t("streakWin", { count: streak })
   450→          : streak < 0
   451→            ? t("streakLoss", { count: Math.abs(streak) })
   452→            : "—",
   453→      streakTone: streak > 0 ? EMERALD : streak < 0 ? ROSE : "",
   454→    };
   455→  }, [stats, advanced, currency, t]);
   456→
   457→  if (!session) return null;
   458→
   459→  const hasAnyTrade = stats != null;
   460→  const showAllCta = !hasAnyTrade && period !== "all";
   461→
   462→  return (
   463→    <div className="space-y-6">
   464→      {/* ── Header + filters ── */}
   465→      <div className="flex flex-col gap-3">
   466→        <div className="flex flex-col gap-1">
   467→          <h2 className="text-xl font-bold tracking-tight sm:text-2xl">
   468→            {t("title")}
   469→          </h2>
   470→          <p className="text-sm text-muted-foreground">
   471→            {hasAnyTrade
   472→              ? t("subtitle", { count: stats.tradeCount })
   473→              : t("subtitleEmpty")}
   474→          </p>
   475→        </div>
   476→        <div className="flex flex-wrap items-center gap-2">
   477→          <Tabs
   478→            value={period}
   479→            onValueChange={(v) => {
   480→              if (PERIODS.includes(v as AnalyticsPeriod)) {
   481→                setPeriod(v as AnalyticsPeriod);
   482→              }
   483→            }}
   484→          >
   485→            <TabsList>
   486→              <TabsTrigger value="30d">{t("period30d")}</TabsTrigger>
   487→              <TabsTrigger value="90d">{t("period90d")}</TabsTrigger>
   488→              <TabsTrigger value="ytd">{t("periodYtd")}</TabsTrigger>
   489→              <TabsTrigger value="all">{t("periodAll")}</TabsTrigger>
   490→            </TabsList>
   491→          </Tabs>
   492→          <Select
   493→            value={accountId}
   494→            onValueChange={(v) => setAccountId(v === "all" ? "all" : v)}
   495→          >
   496→            <SelectTrigger className="w-[180px]" aria-label={t("accountFilter")}>
   497→              <SelectValue />
   498→            </SelectTrigger>
   499→            <SelectContent>
   500→              <SelectItem value="all">{t("allAccounts")}</SelectItem>
   501→              {accounts.map((a) => (
   502→                <SelectItem key={a.id} value={a.id}>
   503→                  {a.name}
   504→                </SelectItem>
   505→              ))}
   506→            </SelectContent>
   507→          </Select>
   508→        </div>
   509→      </div>
   510→
   511→      {/* ── Loading ── */}
   512→      {isLoading ? (
   513→        <div className="space-y-6" role="status" aria-live="polite">
   514→          <span className="sr-only">{t("loading")}</span>
   515→          <div className="grid grid-cols-2 gap-3 lg:grid-cols-4">
   516→            {Array.from({ length: 8 }, (_, i) => (
   517→              <Skeleton key={i} className="h-24 w-full rounded-lg" />
   518→            ))}
   519→          </div>
   520→          <Skeleton className="h-72 w-full rounded-xl" />
   521→          <div className="grid gap-6 lg:grid-cols-2">
   522→            <Skeleton className="h-64 w-full rounded-xl" />
   523→            <Skeleton className="h-64 w-full rounded-xl" />
   524→          </div>
   525→        </div>
   526→      ) : isError ? (
   527→        /* ── Error ── */
   528→        <Card>
   529→          <CardContent className="flex flex-col items-start gap-3">
   530→            <div className="flex items-center gap-2">
   531→              <AlertTriangle className="h-5 w-5 text-destructive" aria-hidden="true" />
   532→              <p className="text-sm font-medium">{t("errorTitle")}</p>
   533→            </div>
   534→            <Button variant="outline" size="sm" onClick={() => void refetch()}>
   535→              <RotateCw className="h-4 w-4" aria-hidden="true" />
   536→              {t("errorRetry")}
   537→            </Button>
   538→          </CardContent>
   539→        </Card>
   540→      ) : !hasAnyTrade || !data ? (
   541→        /* ── Empty state ── */
   542→        <Card>
   543→          <CardHeader>
   544→            <CardTitle className="text-lg">{t("emptyTitle")}</CardTitle>
   545→            <CardDescription>{t("emptyDesc")}</CardDescription>
   546→          </CardHeader>
   547→          <CardContent className="flex flex-col gap-2 sm:flex-row sm:flex-wrap">
   548→            {showAllCta ? (
   549→              <Button onClick={() => setPeriod("all")}>
   550→                <BarChart3 className="h-4 w-4" aria-hidden="true" />
   551→                {t("emptyCtaAll")}
   552→              </Button>
   553→            ) : (
   554→              <Button onClick={() => navigate("#/trades/quick")}>
   555→                <Zap className="h-4 w-4" aria-hidden="true" />
   556→                {t("emptyCtaQuick")}
   557→              </Button>
   558→            )}
   559→            <Button variant="outline" onClick={() => navigate("#/trades/new")}>
   560→              <Plus className="h-4 w-4" aria-hidden="true" />
   561→              {t("emptyCtaForm")}
   562→            </Button>
   563→          </CardContent>
   564→        </Card>
   565→      ) : (
   566→        /* ── Data state ── */
   567→        <div className="space-y-6">
