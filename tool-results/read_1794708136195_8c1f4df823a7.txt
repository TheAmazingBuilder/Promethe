   197→/** Local calendar parts in `timezone` — null on invalid TZ (same as analytics). */
   198→function localParts(
   199→  date: Date,
   200→  timezone: string
   201→): { date: string; hour: number } | null {
   202→  try {
   203→    const fmt = new Intl.DateTimeFormat("en-US", {
   204→      timeZone: timezone,
   205→      year: "numeric",
   206→      month: "2-digit",
   207→      day: "2-digit",
   208→      hour: "2-digit",
   209→      hourCycle: "h23",
   210→    });
   211→    const parts: Record<string, string> = {};
   212→    for (const p of fmt.formatToParts(date)) {
   213→      if (p.type !== "literal") parts[p.type] = p.value;
   214→    }
   215→    const hour = Number.parseInt(parts.hour ?? "", 10);
   216→    return {
   217→      date: `${parts.year}-${parts.month}-${parts.day}`,
   218→      hour: Number.isFinite(hour) ? hour : 0,
   219→    };
   220→  } catch {
   221→    return null;
   222→  }
   223→}

   225→function toDashboardTrade(t: TradeRow, timezone: string): DashboardTrade {
   226→  const exit = t.exitTime ?? t.entryTime;
   227→  const exitLocal = localParts(exit, timezone);
   228→  const entryLocal = localParts(t.entryTime, timezone);
   229→  return {
   230→    id: t.id,
   231→    exitTime: exit.toISOString(),
   232→    entryTime: t.entryTime.toISOString(),
   233→    netPnl: t.netPnl?.toString() ?? null,
   234→    rMultiple: t.rMultiple?.toString() ?? null,
   235→    resultType: t.resultType,
   236→    strategyId: t.strategyId,
   237→    strategyName: t.strategy?.name ?? null,
   238→    instrumentId: t.instrumentId,
   239→    symbol: t.instrument.symbol,
   240→    direction: t.direction,
   241→    session: t.session,
   242→    lots: t.lots.toString(),
   243→    accountId: t.accountId,
   244→    accountName: t.account.name,
   245→    timeframe: t.timeframe,
   246→    setupType: t.setupType,
   247→    marketCondition: t.marketCondition,
   248→    exitReason: t.exitReason,
   249→    emotionBefore: t.emotionBefore,
   250→    followedPlan: t.followedPlan,
   251→    commission: t.commission?.toString() ?? null,
   252→    swap: t.swap?.toString() ?? null,
   253→    fees: t.fees?.toString() ?? null,
   254→    localDate: exitLocal?.date ?? null,
   255→    localWeekday:
   256→      entryLocal != null ? weekdayOfIsoDate(entryLocal.date) : null,
   257→    localHour: entryLocal?.hour ?? null,
   258→    durationMin:
   259→      t.exitTime != null
   260→        ? Math.max(
   261→            0,
   262→            Math.round(
   263→              (t.exitTime.getTime() - t.entryTime.getTime()) / (60 * 1000)
   264→            )
   265→          )
   266→        : null,
   267→    pips: t.pips?.toString() ?? null,
   268→    riskPercent: t.riskPercent?.toString() ?? null,
   269→    tags: t.tags.map((tt) => tt.tag.name),
   270→    row: t,
   271→  };
   272→}

   312→function computeKpiExtras(sample: DashboardTrade[], stats: TradeStats): KpiExtras {
   313→  const pips = sample
   314→    .filter((t) => t.pips != null)
   315→    .map((t) => toD(t.pips!));
   316→  const risks = sample
   317→    .filter((t) => t.riskPercent != null)
   318→    .map((t) => toD(t.riskPercent!));
   319→  const pnls = sample.map((t) => toD(t.netPnl ?? 0));
   320→  const reviewed = sample.filter(
   321→    (t) => t.row.reviewStatus === "REVIEWED"
   322→  ).length;
   323→
   324→  // Current drawdown: distance of the final cumulative P&L from its running
   325→  // peak (same basis as the equity curve's per-point `drawdown`).
   326→  let cum = D0;
   327→  let peak = D0;
   328→  for (const t of sortChronologically(sample)) {
   329→    cum = cum.plus(toD(t.netPnl ?? 0));
   330→    if (cum.gt(peak)) peak = cum;
   331→  }
   332→  const drawdown = Decimal.max(D0, peak.minus(cum));
   333→
   334→  return {
   335→    totalPips: pips.length === 0 ? null : pips.reduce((a, b) => a.plus(b), D0).toString(),
   336→    avgRiskPercent:
   337→      risks.length === 0
   338→        ? null
   339→        : risks.reduce((a, b) => a.plus(b), D0).div(risks.length).toString(),
   340→    reviewedPercent:
   341→      sample.length === 0
   342→        ? null
   343→        : new Decimal(reviewed).div(sample.length).times(100).toString(),
   344→    currentDrawdown: drawdown.toString(),
   345→    bestTrade: pnls.length === 0 ? null : Decimal.max(...pnls).toString(),
   346→    worstTrade: pnls.length === 0 ? null : Decimal.min(...pnls).toString(),
   347→    tradingDays: stats.tradingDays,
   348→  };
   349→}

   377→function dayOfYear(isoDay: string): number {
   378→  const date = Date.parse(`${isoDay}T00:00:00Z`);
   379→  const yearStart = Date.parse(`${isoDay.slice(0, 4)}-01-01T00:00:00Z`);
   380→  if (!Number.isFinite(date) || !Number.isFinite(yearStart)) return 1;
   381→  return Math.floor((date - yearStart) / 86_400_000) + 1;
   382→}

   396→interface WidgetContext {
   397→  userId: string;

   399→  timezone: string;

   404→
   405→async function computeWidgetData(
   406→  ctx: WidgetContext,
   407→  widget: DashboardWidgetDTO,
   408→  account: AccountRow | null,
   409→  scoped: DashboardTrade[],
   410→  sample: DashboardTrade[],
   411→  period: ResolvedPeriod
   412→): Promise<unknown> {
   413→  switch (widget.type) {
   414→    case "kpi-strip":
   415→    case "single-kpi":
   416→    case "period-compare": {
   417→      const compareSample =
   418→        widget.periodOverride?.compareToPrevious === true &&
   419→        period.compareFrom != null
   420→          ? applyComparePeriod(scoped, period)
   421→          : null;
   422→      const data: KpiWidgetData = {
   423→        ...computeKpiBlock(sample),
   424→        compare: compareSample != null ? computeKpiBlock(compareSample) : null,
   425→      };
   426→      return data;
   427→    }
   428→
   429→    case "equity-curve":
   430→    case "drawdown-curve": {
   431→      // Curve convention (dashboard.ts): accounts in the profile currency —
   432→      // or the single filtered account when one is active.
   433→      const curveAccounts =
   434→        account != null
   435→          ? [account]
   436→          : ctx.accounts.filter((a) => a.currency === ctx.currency);
   437→      const initialBalance = curveAccounts.reduce(
   438→        (acc, a) => acc.plus(toD(a.initialBalance)),
   439→        D0
   440→      );
   441→      const curveAccountIds = new Set(curveAccounts.map((a) => a.id));
   442→      const curveTrades = sample.filter((t) =>
   443→        curveAccountIds.has(t.accountId)
   444→      );
   445→      const data: CurveWidgetData = {
   446→        curve: computeEquityCurve(curveTrades, initialBalance),
   447→        initialBalance: initialBalance.toString(),
   448→      };
   449→      return data;
   450→    }
   451→
   452→    case "pnl-calendar": {
   453→      const { months: monthCount } = widgetConfig(
   454→        widget.config,
   455→        pnlCalendarConfigSchema,
   456→        { months: 1 }
   457→      );
   458→      // Local-day buckets (P&L attribution on the local exit day).
   459→      const daily = new Map<
   460→        string,
   461→        { pnl: Decimal; count: number; wins: number; losses: number; rSum: Decimal; rCount: number }
   462→      >();
   463→      for (const t of sample) {
   464→        const day = t.localDate ?? t.exitTime.slice(0, 10);
   465→        const bucket = daily.get(day) ?? {
   466→          pnl: D0,
   467→          count: 0,
   468→          wins: 0,
   469→          losses: 0,
   470→          rSum: D0,
   471→          rCount: 0,
   472→        };
   473→        const pnl = toD(t.netPnl ?? 0);
   474→        bucket.pnl = bucket.pnl.plus(pnl);
   475→        bucket.count += 1;
   476→        if (pnl.gt(0)) bucket.wins += 1;
   477→        else if (pnl.lt(0)) bucket.losses += 1;
   478→        if (t.rMultiple != null) {
   479→          bucket.rSum = bucket.rSum.plus(toD(t.rMultiple));
   480→          bucket.rCount += 1;
   481→        }
   482→        daily.set(day, bucket);
   483→      }
   484→
   485→      const todayLocal =
   486→        localDateInTz(new Date(), ctx.timezone) ??
   487→        new Date().toISOString().slice(0, 10);
   488→      let y = Number(todayLocal.slice(0, 4));
   489→      let m = Number(todayLocal.slice(5, 7));
   490→      for (let i = 1; i < monthCount; i++) {
   491→        m -= 1;
   492→        if (m === 0) {
   493→          m = 12;
   494→          y -= 1;
   495→        }
   496→      }
   497→
   498→      const months: PnlCalendarWidgetData["months"] = [];
   499→      for (let i = 0; i < monthCount; i++) {
   500→        const { first, last } = monthBounds(y, m);
   501→        const days: CalendarDay[] = [...daily.entries()]
   502→          .filter(([day]) => isoDayBetween(day, first, last))
   503→          .map(([day, v]) => ({
   504→            date: day,
   505→            netPnl: v.pnl.toString(),
   506→            tradeCount: v.count,
   507→            wins: v.wins,
   508→            losses: v.losses,
   509→            totalR: v.rCount > 0 ? v.rSum.toString() : null,
   510→            hasJournal: false,
   511→          }));
   512→        const grid = buildMonthGrid(y, m).map((week) =>
   513→          week.map((cell) => {
   514→            const bucket = daily.get(cell.date);
   515→            const cellDto: CalendarCellDTO = {
   516→              date: cell.date,
   517→              inMonth: cell.inMonth,
   518→              netPnl: bucket ? bucket.pnl.toString() : "0",
   519→              tradeCount: bucket ? bucket.count : 0,
   520→            };
   521→            return cellDto;
   522→          })
   523→        );
   524→        months.push({ year: y, month: m, grid, summary: computeMonthSummary(days) });
   525→        m += 1;
   526→        if (m === 13) {
   527→          m = 1;
   528→          y += 1;
   529→        }
   530→      }
   531→      const data: PnlCalendarWidgetData = { months };
   532→      return data;
   533→    }
   534→
   535→    case "open-trades": {
   536→      const openRows = await db.trade.findMany({
   537→        where: {
   538→          userId: ctx.userId,
   539→          status: "OPEN",
   540→          isDraft: false,
   541→          ...(account != null ? { accountId: account.id } : {}),
   542→        },
   543→        include: TRADE_INCLUDE,
   544→        orderBy: { entryTime: "desc" },
   545→      });
   546→      return { openPositions: openRows.map(toOpenPositionDTO) };
   547→    }
   548→
   549→    case "recent-trades": {
   550→      const { limit } = widgetConfig(
   551→        widget.config,
   552→        recentTradesConfigSchema,
   553→        { limit: 6 }
   554→      );
   555→      const recent = [...sample]
   556→        .sort(
   557→          (a, b) =>
   558→            b.entryTime.localeCompare(a.entryTime) || b.id.localeCompare(a.id)
   559→        )
   560→        .slice(0, limit);
   561→      const trades: TradeListItemDTO[] = recent.map((t) => toTradeListItem(t.row));
   562→      return { trades };
   563→    }
   564→
   565→    case "session-performance":
   566→    case "weekday-performance":
   567→    case "hour-performance": {
   568→      const breakdowns = computeBreakdowns(sample);
   569→      const groups: GroupStat[] =
   570→        widget.type === "session-performance"
   571→          ? breakdowns.session
   572→          : widget.type === "weekday-performance"
   573→            ? breakdowns.weekday
   574→            : breakdowns.hour;
   575→      return { groups };
   576→    }
   577→
   578→    case "strategy-table": {
   579→      const breakdowns = computeBreakdowns(sample);
   580→      const groups: Array<GroupStat & { color: string }> =
   581→        breakdowns.strategy.map((g) => ({
   582→          ...g,
   583→          color: ctx.strategyColors.get(g.key) ?? "#64748b",
   584→        }));
   585→      return { groups };
   586→    }
   587→
   588→    case "instrument-table": {
   589→      const breakdowns = computeBreakdowns(sample);
   590→      const groups: GroupStat[] = breakdowns.symbol;
   591→      return { groups };
   592→    }
   593→
   594→    case "r-histogram": {
   595→      const distribution: { buckets: RDistBucket[]; withoutR: number } =
   596→        computeRDistribution(sample);
   597→      return { buckets: distribution };
   598→    }
   599→
   600→    case "win-loss": {
   601→      const stats = computeTradeStats(sample);
   602→      return {
   603→        wins: stats.wins,
   604→        losses: stats.losses,
   605→        breakevens: stats.breakevens,
   606→      };
   607→    }
   608→
   609→    case "streak": {
   610→      const stats = computeTradeStats(sample);
   611→      return {
   612→        current: stats.currentStreak,
   613→        maxWin: stats.maxWinStreak,
   614→        maxLoss: stats.maxLossStreak,
   615→      };
   616→    }
   617→
   618→    case "goals-progress": {
   619→      const [goalsPayload, violations] = await Promise.all([
   620→        listGoalsWithProgress(ctx.userId),
   621→        recentViolations(ctx.userId, 30),
   622→      ]);
   623→      return { goals: goalsPayload.goals, violations };
   624→    }
   625→
   626→    case "journal-prompt": {
   627→      const todayLocal =
   628→        localDateInTz(new Date(), ctx.timezone) ??
   629→        new Date().toISOString().slice(0, 10);
   630→      const entry = await db.journalEntry.findFirst({
   631→        where: { userId: ctx.userId, date: todayLocal },
   632→        select: { id: true },
   633→      });
   634→      return {
   635→        prompt: JOURNAL_PROMPTS[dayOfYear(todayLocal) % JOURNAL_PROMPTS.length],
   636→        hasEntryToday: entry != null,
   637→      };
   638→    }
   639→
   640→    case "review-queue": {
   641→      const accountFilter = account != null ? { accountId: account.id } : {};
   642→      const [toReviewCount, draftCount] = await Promise.all([
   643→        db.trade.count({
   644→          where: {
   645→            userId: ctx.userId,
   646→            reviewStatus: "NOT_REVIEWED",
   647→            isDraft: false,
   648→            ...accountFilter,
   649→          },
   650→        }),
   651→        db.trade.count({
   652→          where: { userId: ctx.userId, isDraft: true, ...accountFilter },
   653→        }),
   654→      ]);
   655→      return { toReviewCount, draftCount };
   656→    }
   657→
   658→    case "next-news": {
   659→      const { limit } = widgetConfig(widget.config, nextNewsConfigSchema, {
   660→        limit: 5,
   661→      });
   662→      const events = await db.economicEvent.findMany({
   663→        where: {
   664→          scheduledAt: { gte: new Date() },
   665→          impact: "HIGH",
   666→          OR: [{ userId: null }, { userId: ctx.userId }],
   667→        },
   668→        orderBy: { scheduledAt: "asc" },
   669→        take: limit,
   670→      });
   671→      const dto: NewsEventDTO[] = events.map((e) => ({
   672→        id: e.id,
   673→        title: e.title,
   674→        currency: e.currency,
   675→        impact: e.impact,
   676→        category: e.category,
   677→        scheduledAt: e.scheduledAt.toISOString(),
   678→      }));
   679→      return { events: dto };
   680→    }
   681→
   682→    case "prop-status": {
   683→      const payload = await getPropStatus(ctx.userId, account?.id);
   684→      return { status: payload ?? { account: null, status: null } };
   685→    }
   686→
   687→    case "sync-status": {
   688→      return { overview: await getSyncOverview(ctx.userId) };
   689→    }
   690→
   691→    case "custom-metric": {
   692→      const { metricId } = widgetConfig(widget.config, customMetricConfigSchema, {
   693→        metricId: "",
   694→      });
   695→      if (!metricId) {
   696→        throw new DashboardValidationError("METRIC_NOT_FOUND", 422);
   697→      }
   698→      const metric = await db.customMetric.findFirst({
   699→        where: { id: metricId, userId: ctx.userId },
   700→      });
   701→      if (!metric) {
   702→        throw new DashboardValidationError("METRIC_NOT_FOUND", 422);
   703→      }
   704→      let definition: unknown = null;
   705→      try {
   706→        definition = JSON.parse(metric.definition);
   707→      } catch {
   708→        definition = null;
   709→      }
   710→      if (!isMetricDefinition(definition)) {
   711→        throw new Error(`Corrupt metric definition on custom metric ${metric.id}`);
   712→      }
   713→      const metricTrades: MetricTrade[] = sample;
   714→      const result = evaluateMetric(definition, metricTrades);
   715→      const data: CustomMetricWidgetData = {
   716→        name: metric.name,
   717→        format: metric.format,
   718→        value: result.value,
   719→        sampleSize: result.sampleSize,
   720→        lowSample: result.lowSample,
   721→      };
   722→      return data;
   723→    }
   724→
   725→    default:
   726→      // Static widgets (sticky-note, quick-actions) and unknown types are
   727→      // filtered before the loader runs — unreachable safety net.
   728→      return null;
   729→  }
   730→}

   732→// ── Entry point ──────────────────────────────────────────────────────────────
   733→
   734→export async function getDashboardData(
   735→  userId: string,
   736→  dashboardId: string,
   737→  query: { preset: PeriodPreset; accountId: string | null }
   738→): Promise<DashboardWidgetDataPayload> {
   739→  const [profile, accounts, dashboardRow] = await Promise.all([
   740→    db.profile.findUnique({ where: { id: userId } }),
   741→    db.account.findMany({ where: { userId }, orderBy: { createdAt: "asc" } }),
   742→    db.dashboard.findFirst({
   743→      where: { id: dashboardId, userId },
   744→      include: { widgets: { orderBy: { createdAt: "asc" } } },
   745→    }),
   746→  ]);
   747→  if (!dashboardRow) {
   748→    throw new DashboardValidationError("DASHBOARD_NOT_FOUND", 404);
   749→  }
   750→
   751→  const timezone = profile?.timezone ?? "UTC";
   752→  const currency = profile?.baseCurrency ?? "USD";
   753→  const globalAccount =
   754→    query.accountId != null
   755→      ? (accounts.find((a) => a.id === query.accountId) ?? null)
   756→      : null;
   757→
   758→  // The dataset is fetched ONCE — unscoped — so a widget accountOverride can
   759→  // REPLACE the global account filter (override semantics) instead of
   760→  // intersecting it. Per-widget scoping happens in memory.
   761→  const [closedRows, strategyRows, requirements] = await Promise.all([
   762→    db.trade.findMany({
   763→      where: { userId, status: "CLOSED", isDraft: false },
   764→      include: TRADE_INCLUDE,
   765→    }),
   766→    db.strategy.findMany({
   767→      where: { userId },
   768→      select: { name: true, color: true },
   769→    }),
   770→    getRequirements(userId),
   771→  ]);
   772→
   773→  const trades = closedRows.map((t) => toDashboardTrade(t, timezone));
   774→  const dto = toDashboardDTO(dashboardRow);
   775→
   776→  const ctx: WidgetContext = {
   777→    userId,
   778→    currency,
   779→    timezone,
   780→    accounts,
   781→    strategyColors: new Map(strategyRows.map((s) => [s.name, s.color])),
   782→  };
   783→
   784→  const widgets: Record<string, { data: unknown | null; error: string | null }> =
   785→    {};
   786→
   787→  for (const widget of dto.widgets) {
   788→    const definition = getWidgetDefinition(widget.type);
   789→    if (!definition || !definition.hasLoader) {
   790→      // Static widget (sticky-note, quick-actions) or unknown type → the
   791→      // client renders it from config alone.
   792→      widgets[widget.id] = { data: null, error: null };
   793→      continue;
   794→    }
   795→
   796→    try {
   797→      const account =
   798→        widget.accountOverride != null
   799→          ? (accounts.find((a) => a.id === widget.accountOverride) ?? null)
   800→          : globalAccount;
   801→      const scoped =
   802→        account != null
   803→          ? trades.filter((t) => t.accountId === account.id)
   804→          : trades;
   805→      const period = resolvePeriod(
   806→        { preset: query.preset, accountId: widget.accountOverride ?? null },
   807→        widget.periodOverride,
   808→        timezone
   809→      );
   810→      const sample = applyPeriod(scoped, period);
   811→      widgets[widget.id] = {
   812→        data: await computeWidgetData(ctx, widget, account, scoped, sample, period),
   813→        error: null,
   814→      };
   815→    } catch (error) {
   816→      if (error instanceof DashboardValidationError) {
   817→        // Typed widget-level error (e.g. METRIC_NOT_FOUND) — surfaced as-is.
   818→        widgets[widget.id] = { data: null, error: error.code };
   819→      } else {
   820→        console.error(
   821→          `[dashboard-data] widget ${widget.type} (${widget.id}) failed:`,
   822→          error
   823→        );
   824→        widgets[widget.id] = { data: null, error: "WIDGET_DATA_FAILED" };
   825→      }
   826→    }
   827→  }
   828→
   829→  return {
   830→    currency,
   831→    timezone,
   832→    generatedAt: new Date().toISOString(),
   833→    widgets,
   834→    requirements,
   835→  };
   836→}
