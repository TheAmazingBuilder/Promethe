    89→function KpiCard({
    90→  label,
    91→  value,
    92→  valueClassName,
    93→  sub,
    94→  compact = false,
    95→}: {
    96→  label: string;
    97→  value: React.ReactNode;
    98→  valueClassName?: string;
    99→  sub?: React.ReactNode;
   100→  compact?: boolean;
   101→}) {
   102→  return (
   103→    <div className="flex flex-col gap-2 rounded-lg border p-4">
   104→      <p className="text-xs font-medium text-muted-foreground">{label}</p>
   105→      <p
   106→        className={`tabular-nums ${
   107→          compact ? "text-xl font-semibold" : "text-2xl font-bold"
   108→        } ${valueClassName ?? ""}`}
   109→      >
   110→        {value}
   111→      </p>
   112→      {sub != null ? (
   113→        <p className="text-xs tabular-nums text-muted-foreground">{sub}</p>
   114→      ) : null}
   115→    </div>
   116→  );
   117→}

   126→function EquityTooltipContent({
   127→  active,
   128→  payload,
   129→  currency,
   130→  timezone,
   131→}: {
   132→  active?: boolean;
   133→  payload?: Array<{ payload?: EquityDatum }>;
   134→  currency: string;
   135→  timezone: string;
   136→}) {
   137→  if (!active) return null;
   138→  const point = payload?.[0]?.payload;
   139→  if (!point) return null;
   140→  return (
   141→    <div className="rounded-lg border bg-background px-3 py-2 text-xs shadow-md">
   142→      <p className="font-medium">#{point.i}</p>
   143→      <p className="text-muted-foreground">
   144→        {fmtDateInTz(point.date, timezone)}
   145→      </p>
   146→      <p className="mt-1 tabular-nums">{fmtMoney(point.equity, currency)}</p>
   147→      {point.drawdown > 0 ? (
   148→        <p className={`tabular-nums ${ROSE}`}>
   149→          −{fmtMoney(point.drawdown, currency)}
   150→        </p>
   151→      ) : null}
   152→    </div>
   153→  );

   177→/** One breakdown dimension — compact rows with a proportional net P&L bar. */
   178→function BreakdownCard({
   179→  title,
   180→  description,
   181→  icon: Icon,
   182→  entries,
   183→  currency,
   184→  renderKey,
   185→  max = 8,
   186→}: {
   187→  title: string;
   188→  description?: string;
   189→  icon: React.ComponentType<{ className?: string }>;
   190→  entries: GroupStat[];
   191→  currency: string;
   192→  renderKey: (key: string) => string;
   193→  max?: number;
   194→}) {
   195→  const t = useTranslations("analytics");
   196→  const shown = entries.slice(0, max);
   197→  const maxAbs = shown.reduce(
   198→    (m, e) => Math.max(m, Math.abs(Number(e.stats.netPnl))),
   199→    0
   200→  );
   201→  return (
   202→    <Card>
   203→      <CardHeader>
   204→        <CardTitle className="flex items-center gap-2 text-base">
   205→          <Icon className="h-4 w-4 text-muted-foreground" aria-hidden="true" />
   206→          {title}
   207→        </CardTitle>
   208→        {description ? <CardDescription>{description}</CardDescription> : null}
   209→      </CardHeader>
   210→      <CardContent>
   211→        {shown.length === 0 ? (
   212→          <p className="text-sm text-muted-foreground">{t("noData")}</p>
   213→        ) : (
   214→          <ul className="space-y-1.5">
   215→            {shown.map((e) => {
   216→              const net = Number(e.stats.netPnl);
   217→              const width =
   218→                maxAbs > 0 ? Math.max(2, (Math.abs(net) / maxAbs) * 100) : 0;
   219→              return (
   220→                <li
   221→                  key={e.key}
   222→                  className="relative flex items-center justify-between gap-2 overflow-hidden rounded-md px-2 py-1.5 text-sm"
   223→                >
   224→                  <span
   225→                    className={`absolute inset-y-0 left-0 ${
   226→                      net >= 0 ? "bg-gain-subtle" : "bg-loss-subtle"
   227→                    }`}
   228→                    style={{ width: `${width}%` }}
   229→                    aria-hidden="true"
   230→                  />
   231→                  <span className="relative z-10 min-w-0 truncate font-medium">
   232→                    {renderKey(e.key)}
   233→                  </span>
   234→                  <span className="relative z-10 flex shrink-0 items-center gap-3 tabular-nums">
   235→                    <span className="text-xs text-muted-foreground">
   236→                      ×{e.stats.tradeCount}
   237→                    </span>
   238→                    <span className="hidden w-16 whitespace-nowrap text-right text-xs text-muted-foreground sm:inline">
   239→                      {e.stats.winRate != null
   240→                        ? fmtPct(e.stats.winRate)
   241→                        : "—"}
   242→                    </span>
   243→                    <span
   244→                      className={`w-24 text-right font-medium ${pnlClass(
   245→                        e.stats.netPnl
   246→                      )}`}
   247→                    >
   248→                      {fmtSignedMoney(e.stats.netPnl, currency)}
   249→                    </span>
   250→                  </span>
   251→                </li>
   252→              );
   253→            })}
   254→            {entries.length > shown.length ? (
   255→              <li className="px-2 pt-1 text-xs text-muted-foreground">
   256→                {t("more", { count: entries.length - shown.length })}
   257→              </li>
   258→            ) : null}
   259→          </ul>
   260→        )}
   261→      </CardContent>
   262→    </Card>
   263→  );
   264→}

   268→export function AnalyticsView() {
   269→  const { session } = useSession();
   270→  const t = useTranslations("analytics");
   271→  const tRoot = useTranslations();
   272→  const labels = useTradingLabels();
   273→  const { locale } = useI18n();
   274→  const [period, setPeriod] = React.useState<AnalyticsPeriod>("30d");
   275→  const [accountId, setAccountId] = React.useState<string>("all");
   276→
   277→  const { data: accountsData } = useAccounts();
   278→  const accounts = accountsData ?? [];
   279→
   280→  const effectiveAccount = accountId === "all" ? undefined : accountId;
   281→  const { data, isLoading, isError, refetch } = useAnalytics({
   282→    period,
   283→    account: effectiveAccount,
   284→  });
   285→
   286→  const timezone = data?.timezone ?? session?.profile.timezone ?? "UTC";
   287→  const currency = data?.currency ?? session?.profile.baseCurrency ?? "USD";
   288→  const stats = data?.stats ?? null;
   289→  const advanced = data?.advanced ?? null;
   290→
   291→  const equityData = React.useMemo<EquityDatum[]>(
   292→    () =>
   293→      (data?.equityCurve ?? []).map((p) => ({
   294→        i: p.i,
   295→        date: p.date,
   296→        equity: Number(p.equity),
   297→        drawdown: Number(p.drawdown),
   298→      })),
   299→    [data]
   300→  );
   301→
   302→  const rDistData = React.useMemo(
   303→    () =>
   304→      (data?.rDistribution.buckets ?? []).map((b) => ({
   305→        id: b.id,
   306→        label: t(`rdist.${b.id}`),
   307→        count: b.count,
   308→        totalR: Number(b.totalR),
   309→        negative: b.to != null ? b.to <= 0 : false,
   310→      })),
   311→    [data, t]
   312→  );
   313→
   314→  const weekdayLabel = React.useCallback(
   315→    (key: string) => {
   316→      try {
   317→        // ISO weekday 1-7 → localized name via a stable reference week
   318→        // (2024-01-01 was a Monday).
   319→        const ref = new Date(Date.UTC(2024, 0, Number(key)));
   320→        return new Intl.DateTimeFormat(locale, {
   321→          weekday: "long",
   322→          timeZone: "UTC",
   323→        }).format(ref);
   324→      } catch {
   325→        return key;
   326→      }
   327→    },
   328→    [locale]
   329→  );
   330→
   331→  const hourLabel = React.useCallback((key: string) => {
   332→    const h = Number(key);
   333→    if (!Number.isFinite(h)) return key;
   334→    const two = (n: number) => String(n).padStart(2, "0");
   335→    return `${two(h)}:00–${two(h + 1)}:59`;
   336→  }, []);
   337→
   338→  /** Formats raw insight params for i18n interpolation (money, R, labels…). */
   339→  const formatInsightParams = React.useCallback(
   340→    (insight: Insight): Record<string, string> => {
   341→      const money = new Set(["netPnl", "fees", "dd", "pnl"]);
   342→      const pct = new Set(["winRate", "pct"]);
   343→      const r = new Set([
   344→        "avgR",
   345→        "avgRPlan",
   346→        "avgRNoPlan",
   347→        "busyAvgR",
   348→        "lightAvgR",
   349→      ]);
   350→      const num = new Set(["payoff", "rf"]);
   351→      const out: Record<string, string> = {};
   352→      for (const [k, v] of Object.entries(insight.params)) {
   353→        if (money.has(k) && typeof v === "string") {
   354→          out[k] = fmtSignedMoney(v, currency);
   355→        } else if (pct.has(k) && typeof v === "string") {
   356→          out[k] = fmtPct(v);
   357→        } else if (r.has(k) && typeof v === "string") {
   358→          out[k] = `${fmtSigned(v)} R`;
   359→        } else if (num.has(k) && typeof v === "string") {
   360→          out[k] = fmtNum(v);
   361→        } else if (k === "weekday") {
   362→          out[k] = weekdayLabel(String(v));
   363→        } else if (k === "hour") {
   364→          out[k] = hourLabel(String(v));
   365→        } else if (k === "session") {
   366→          out[k] = labels.session(String(v));
   367→        } else if (k === "emotion") {
   368→          out[k] = labels.emotion(String(v));
   369→        } else {
   370→          out[k] = String(v);
   371→        }
   372→      }
   373→      return out;
   374→    },
   375→    [currency, weekdayLabel, hourLabel, labels]
   376→  );
   377→
   378→  const renderInsight = (insight: Insight) =>
   379→    tRoot(`analytics.insights.${insight.rule}` as Parameters<typeof tRoot>[0], {
   380→      ...formatInsightParams(insight),
   381→    });
   382→
   383→  const goodInsights = (data?.insights ?? []).filter((i) => i.tone === "good");
   384→  const badInsights = (data?.insights ?? []).filter((i) => i.tone === "bad");
   385→  const infoInsights = (data?.insights ?? []).filter((i) => i.tone === "info");
   386→
   387→  /** KPI display model. */
   388→  const kpi = React.useMemo(() => {
   389→    const s = stats;
   390→    const pfNum = s?.profitFactor != null ? Number(s.profitFactor) : null;
   391→    const streak = s?.currentStreak ?? 0;
   392→    return {
   393→      netPnl: s ? fmtSignedMoney(s.netPnl, currency) : "—",
   394→      netPnlTone: s ? pnlClass(s.netPnl) : "text-muted-foreground",
   395→      winRate: s?.winRate != null ? fmtPct(s.winRate) : "—",
   396→      profitFactor:
   397→        pfNum == null
   398→          ? "—"
   399→          : Number.isFinite(pfNum)
   400→            ? fmtNum(pfNum)
   401→            : "∞",
   402→      profitFactorTone:
   403→        pfNum == null
   404→          ? ""
   405→          : pfNum >= 2
   406→            ? EMERALD
   407→            : pfNum < 1
   408→              ? ROSE
   409→              : "",
   410→      expectancy: s?.expectancyR != null ? `${fmtSigned(s.expectancyR)} R` : "—",
   411→      expectancyTone:
   412→        s?.expectancyR != null ? pnlClass(s.expectancyR) : "text-muted-foreground",
   413→      totalR: s?.totalR != null ? `${fmtSigned(s.totalR)} R` : "—",
   414→      maxDrawdown: s?.maxDrawdown != null ? fmtMoney(s.maxDrawdown, currency) : "—",
   415→      maxDrawdownTone: s?.maxDrawdown != null ? ROSE : "",
   416→      sqn: advanced?.sqn != null ? fmtNum(advanced.sqn) : "—",
   417→      sqnTone:
   418→        advanced?.sqn == null
   419→          ? ""
   420→          : Number(advanced.sqn) >= 2
   421→            ? EMERALD
   422→            : Number(advanced.sqn) < 1
   423→              ? ROSE
   424→              : "",
   425→      tradeCount: s ? fmtNum(s.tradeCount, 0) : "—",
   426→      winsLosses: s
   427→        ? t("kpiWinsLosses", { wins: s.wins, losses: s.losses, be: s.breakevens })
   428→        : null,
   429→      sharpe: advanced?.sharpe != null ? fmtNum(advanced.sharpe) : "—",
   430→      sortino: advanced?.sortino != null ? fmtNum(advanced.sortino) : "—",
   431→      recovery:
   432→        advanced?.recoveryFactor != null ? fmtNum(advanced.recoveryFactor) : "—",
   433→      payoff: advanced?.payoffRatio != null ? fmtNum(advanced.payoffRatio) : "—",
   434→      avgDuration:
   435→        advanced?.avgDurationMin != null
   436→          ? advanced.avgDurationMin >= 60
   437→            ? t("durationHm", {
   438→                h: Math.floor(advanced.avgDurationMin / 60),
   439→                m: advanced.avgDurationMin % 60,
   440→              })
   441→            : t("durationM", { m: advanced.avgDurationMin })
   442→          : "—",
   443→      totalFees:
   444→        advanced != null ? fmtMoney(advanced.totalFees, currency) : "—",
   445→      rCoverage: advanced?.rCoverage != null ? fmtPct(advanced.rCoverage) : "—",
   446→      maxWinStreak: s ? fmtNum(s.maxWinStreak, 0) : "—",
   447→      maxLossStreak: s ? fmtNum(s.maxLossStreak, 0) : "—",
   448→      streakText:
   449→        streak > 0
   450→          ? t("streakWin", { count: streak })
   451→          : streak < 0
   452→            ? t("streakLoss", { count: Math.abs(streak) })
   453→            : "—",
   454→      streakTone: streak > 0 ? EMERALD : streak < 0 ? ROSE : "",
   455→    };
   456→  }, [stats, advanced, currency, t]);
   457→
   458→  if (!session) return null;
   459→
   460→  const hasAnyTrade = stats != null;
   461→  const showAllCta = !hasAnyTrade && period !== "all";
   462→
   463→  return (
   464→    <div className="space-y-6">
   465→      {/* ── Header + filters ── */}
   466→      <div className="flex flex-col gap-3">
   467→        <div className="flex flex-col gap-1">
   468→          <h2 className="text-xl font-bold tracking-tight sm:text-2xl">
   469→            {t("title")}
   470→          </h2>
   471→          <p className="text-sm text-muted-foreground">
   472→            {hasAnyTrade
   473→              ? t("subtitle", { count: stats.tradeCount })
   474→              : t("subtitleEmpty")}
   475→          </p>
   476→        </div>
   477→        <div className="flex flex-wrap items-center gap-2">
   478→          <Tabs
   479→            value={period}
   480→            onValueChange={(v) => {
   481→              if (PERIODS.includes(v as AnalyticsPeriod)) {
   482→                setPeriod(v as AnalyticsPeriod);
   483→              }
   484→            }}
   485→          >
   486→            <TabsList>
   487→              <TabsTrigger value="30d">{t("period30d")}</TabsTrigger>
   488→              <TabsTrigger value="90d">{t("period90d")}</TabsTrigger>
   489→              <TabsTrigger value="ytd">{t("periodYtd")}</TabsTrigger>
   490→              <TabsTrigger value="all">{t("periodAll")}</TabsTrigger>
   491→            </TabsList>
   492→          </Tabs>
   493→          <Select
   494→            value={accountId}
   495→            onValueChange={(v) => setAccountId(v === "all" ? "all" : v)}
   496→          >
   497→            <SelectTrigger className="w-[180px]" aria-label={t("accountFilter")}>
   498→              <SelectValue />
   499→            </SelectTrigger>
   500→            <SelectContent>
   501→              <SelectItem value="all">{t("allAccounts")}</SelectItem>
   502→              {accounts.map((a) => (
   503→                <SelectItem key={a.id} value={a.id}>
   504→                  {a.name}
   505→                </SelectItem>
   506→              ))}
   507→            </SelectContent>
   508→          </Select>
   509→          {/* v2.5 M2 — chart builder deep link */}
   510→          <Button
   511→            variant="outline"
   512→            size="sm"
   513→            className="ml-auto"
   514→            onClick={() => navigate("#/analytics/charts")}
   515→          >
   516→            <LineChart className="h-4 w-4" aria-hidden="true" />
   517→            {t("openChartBuilder")}
   518→          </Button>
   519→        </div>
   520→      </div>
   521→
   522→      {/* ── Loading ── */}
   523→      {isLoading ? (
   524→        <div className="space-y-6" role="status" aria-live="polite">
   525→          <span className="sr-only">{t("loading")}</span>
   526→          <div className="grid grid-cols-2 gap-3 lg:grid-cols-4">
   527→            {Array.from({ length: 8 }, (_, i) => (
   528→              <Skeleton key={i} className="h-24 w-full rounded-lg" />
   529→            ))}
   530→          </div>
   531→          <Skeleton className="h-72 w-full rounded-xl" />
   532→          <div className="grid gap-6 lg:grid-cols-2">
   533→            <Skeleton className="h-64 w-full rounded-xl" />
   534→            <Skeleton className="h-64 w-full rounded-xl" />
   535→          </div>
   536→        </div>
   537→      ) : isError ? (
   538→        /* ── Error ── */
   539→        <Card>
   540→          <CardContent className="flex flex-col items-start gap-3">
   541→            <div className="flex items-center gap-2">
   542→              <AlertTriangle className="h-5 w-5 text-destructive" aria-hidden="true" />
   543→              <p className="text-sm font-medium">{t("errorTitle")}</p>
   544→            </div>
   545→            <Button variant="outline" size="sm" onClick={() => void refetch()}>
   546→              <RotateCw className="h-4 w-4" aria-hidden="true" />
   547→              {t("errorRetry")}
   548→            </Button>
   549→          </CardContent>
   550→        </Card>
   551→      ) : !hasAnyTrade || !data ? (
   552→        /* ── Empty state ── */
   553→        <Card>
   554→          <CardHeader>
   555→            <CardTitle className="text-lg">{t("emptyTitle")}</CardTitle>
   556→            <CardDescription>{t("emptyDesc")}</CardDescription>
   557→          </CardHeader>
   558→          <CardContent className="flex flex-col gap-2 sm:flex-row sm:flex-wrap">
   559→            {showAllCta ? (
   560→              <Button onClick={() => setPeriod("all")}>
   561→                <BarChart3 className="h-4 w-4" aria-hidden="true" />
   562→                {t("emptyCtaAll")}
   563→              </Button>
   564→            ) : (
   565→              <Button onClick={() => navigate("#/trades/quick")}>
   566→                <Zap className="h-4 w-4" aria-hidden="true" />
   567→                {t("emptyCtaQuick")}
   568→              </Button>
