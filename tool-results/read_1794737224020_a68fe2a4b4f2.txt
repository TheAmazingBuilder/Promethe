   104→/** Local calendar day YYYY-MM-DD (client-side default, effect-only). */
   105→function todayLocalDay(): string {
   106→  const d = new Date();
   107→  const pad = (n: number) => String(n).padStart(2, "0");
   108→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}`;
   109→}

   111→function GoalEditor({
   112→  open,
   113→  onOpenChange,
   114→  editing,
   115→  currency,
   116→}: {
   117→  open: boolean;
   118→  onOpenChange: (open: boolean) => void;
   119→  editing: GoalDTO | null;
   120→  currency: string;
   121→}) {
   122→  const t = useTranslations("goalsView.form");
   123→  const tGoals = useTranslations("goalsView");
   124→  const tRoot = useTranslations();
   125→  const { toast } = useToast();
   126→  const create = useCreateGoal();
   127→  const update = useUpdateGoal();
   128→  const [form, setForm] = React.useState<GoalForm>(EMPTY_GOAL);
   129→  const [saving, setSaving] = React.useState(false);
   130→
   131→  React.useEffect(() => {
   132→    if (!open) return;
   133→    setForm(
   134→      editing
   135→        ? {
   136→            name: editing.name,
   137→            description: editing.description ?? "",
   138→            metric: (METRICS as string[]).includes(editing.metric)
   139→              ? (editing.metric as GoalForm["metric"])
   140→              : "NET_PNL",
   141→            targetValue: editing.targetValue,
   142→            period: (PERIODS as readonly string[]).includes(editing.period)
   143→              ? (editing.period as GoalForm["period"])
   144→              : "MONTHLY",
   145→            startDate: editing.startDate,
   146→            endDate: editing.endDate ?? "",
   147→            status: editing.status === "PAUSED" ? "PAUSED" : "ACTIVE",
   148→          }
   149→        : {
   150→            ...EMPTY_GOAL,
   151→            startDate: todayLocalDay(),
   152→          },
   153→    );
   154→  }, [open, editing]);
   155→
   156→  const set = <K extends keyof GoalForm>(key: K, value: GoalForm[K]) =>
   157→    setForm((f) => ({ ...f, [key]: value }));
   158→
   159→  const onSave = async () => {
   160→    if (!form.name.trim() || !form.targetValue.trim()) return;
   161→    setSaving(true);
   162→    try {
   163→      const payload: GoalInput = {
   164→        name: form.name.trim(),
   165→        description: form.description.trim() === "" ? null : form.description.trim(),
   166→        metric: form.metric,
   167→        targetValue: form.targetValue.trim(),
   168→        period: form.period,
   169→        startDate: form.startDate,
   170→        endDate: form.endDate.trim() === "" ? null : form.endDate.trim(),
   171→        status: form.status,
   172→      };
   173→      if (editing) {
   174→        await update.mutateAsync({ id: editing.id, patch: payload });
   175→        toast({ title: t("updated") });
   176→      } else {
   177→        await create.mutateAsync(payload);
   178→        toast({ title: t("created") });
   179→      }
   180→      onOpenChange(false);
   181→    } catch (error) {
   182→      const err = error instanceof ApiClientError ? error : null;
   183→      toast({
   184→        title: apiErrorMessage(tRoot, err?.code, error instanceof Error ? error.message : ""),
   185→        variant: "destructive",
   186→      });
   187→    } finally {
   188→      setSaving(false);
   189→    }
   190→  };
   191→
   192→  const targetHint =
   193→    form.metric === "WIN_RATE" || form.metric === "DISCIPLINE_SCORE" ? t("targetHintPct") : t("targetHintAmount");
   194→
   195→  return (
   196→    <Dialog open={open} onOpenChange={onOpenChange}>
   197→      <DialogContent className="max-h-[90vh] overflow-y-auto sm:max-w-lg">
   198→        <DialogHeader>
   199→          <DialogTitle>{editing ? t("editTitle") : t("createTitle")}</DialogTitle>
   200→          <DialogDescription className="sr-only">
   201→            {editing ? t("editTitle") : t("createTitle")}
   202→          </DialogDescription>
   203→        </DialogHeader>
   204→        <div className="flex flex-col gap-4">
   205→          <div className="space-y-1.5">
   206→            <label className="text-xs font-medium text-muted-foreground" htmlFor="goal-name">
   207→              {t("name")}
   208→            </label>
   209→            <Input
   210→              id="goal-name"
   211→              value={form.name}
   212→              onChange={(e) => set("name", e.target.value)}
   213→              placeholder={t("namePlaceholder")}
   214→              maxLength={80}
   215→            />
   216→          </div>
   217→          <div className="space-y-1.5">
   218→            <label className="text-xs font-medium text-muted-foreground" htmlFor="goal-desc">
   219→              {t("description")}
   220→            </label>
   221→            <Input
   222→              id="goal-desc"
   223→              value={form.description}
   224→              onChange={(e) => set("description", e.target.value)}
   225→              placeholder={t("descriptionPlaceholder")}
   226→              maxLength={2000}
   227→            />
   228→          </div>
   229→          <div className="grid gap-3 sm:grid-cols-2">
   230→            <div className="space-y-1.5">
   231→              <label className="text-xs font-medium text-muted-foreground" htmlFor="goal-metric">
   232→                {t("metric")}
   233→              </label>
   234→              <Select
   235→                value={form.metric}
   236→                onValueChange={(v) => set("metric", v as GoalForm["metric"])}
   237→              >
   238→                <SelectTrigger id="goal-metric">
   239→                  <SelectValue />
   240→                </SelectTrigger>
   241→                <SelectContent>
   242→                  {METRICS.map((m) => (
   243→                    <SelectItem key={m} value={m}>
   244→                      {tGoals(`metric.${m}`)}
   245→                    </SelectItem>
   246→                  ))}
   247→                </SelectContent>
   248→              </Select>
   249→            </div>
   250→            <div className="space-y-1.5">
   251→              <label className="text-xs font-medium text-muted-foreground" htmlFor="goal-target">
   252→                {t("target")}
   253→              </label>
   254→              <Input
   255→                id="goal-target"
   256→                inputMode="decimal"
   257→                value={form.targetValue}
   258→                onChange={(e) => set("targetValue", e.target.value)}
   259→                placeholder={
   260→                  form.metric === "WIN_RATE"
   261→                    ? "55"
   262→                    : form.metric === "NET_PNL"
   263→                      ? `1000 ${currency}`
   264→                      : "1"
   265→                }
   266→              />
   267→              <p className="text-[11px] text-muted-foreground">{targetHint}</p>
   268→            </div>
   269→          </div>
   270→          <div className="grid gap-3 sm:grid-cols-3">
   271→            <div className="space-y-1.5">
   272→              <label className="text-xs font-medium text-muted-foreground" htmlFor="goal-period">
   273→                {t("period")}
   274→              </label>
   275→              <Select
   276→                value={form.period}
   277→                onValueChange={(v) => set("period", v as GoalForm["period"])}
   278→              >
   279→                <SelectTrigger id="goal-period">
   280→                  <SelectValue />
   281→                </SelectTrigger>
   282→                <SelectContent>
   283→                  {PERIODS.map((p) => (
   284→                    <SelectItem key={p} value={p}>
   285→                      {tGoals(`period.${p}`)}
   286→                    </SelectItem>
   287→                  ))}
   288→                </SelectContent>
   289→              </Select>
   290→            </div>
   291→            <div className="space-y-1.5">
   292→              <label className="text-xs font-medium text-muted-foreground" htmlFor="goal-start">
   293→                {t("startDate")}
   294→              </label>
   295→              <Input
   296→                id="goal-start"
   297→                type="date"
   298→                value={form.startDate}
   299→                onChange={(e) => set("startDate", e.target.value)}
   300→              />
   301→            </div>
   302→            <div className="space-y-1.5">
   303→              <label className="text-xs font-medium text-muted-foreground" htmlFor="goal-end">
   304→                {t("endDate")}
   305→              </label>
   306→              <Input
   307→                id="goal-end"
   308→                type="date"
   309→                value={form.endDate}
   310→                onChange={(e) => set("endDate", e.target.value)}
   311→              />
   312→            </div>
   313→          </div>
   314→        </div>
   315→        <DialogFooter>
   316→          <Button variant="outline" onClick={() => onOpenChange(false)}>
   317→            {t("cancel")}

   319→          <Button
   320→            onClick={() => void onSave()}
   321→            disabled={saving || !form.name.trim() || !form.targetValue.trim()}

   324→          </Button>
   325→        </DialogFooter>

   333→function GoalCard({
   334→  goal,
   335→  currency,
   336→  onEdit,
   337→}: {
   338→  goal: GoalWithProgressDTO;
   339→  currency: string;
   340→  onEdit: () => void;
   341→}) {
   342→  const t = useTranslations("goalsView");
   343→  const { toast } = useToast();
   344→  const update = useUpdateGoal();
   345→  const del = useDeleteGoal();
   346→  const [confirmOpen, setConfirmOpen] = React.useState(false);
   347→
   348→  const paused = goal.status === "PAUSED";
   349→  const { progress } = goal;
   350→
   351→  const onToggle = async () => {
   352→    try {
   353→      await update.mutateAsync({
   354→        id: goal.id,
   355→        patch: { status: paused ? "ACTIVE" : "PAUSED" },
   356→      });
   357→      toast({ title: paused ? t("resumedToast") : t("pausedToast") });
   358→    } catch {
   359→      toast({ title: t("errorToast"), variant: "destructive" });
   360→    }
   361→  };
   362→
   363→  const onDelete = async () => {
   364→    try {
   365→      await del.mutateAsync(goal.id);
   366→      toast({ title: t("deletedToast") });
   367→      setConfirmOpen(false);
   368→    } catch {
   369→      toast({ title: t("errorToast"), variant: "destructive" });
   370→    }
   371→  };
   372→
   373→  return (
   374→    <Card className={paused ? "opacity-60" : undefined}>
   375→      <CardHeader>
   376→        <CardTitle className="flex items-center gap-2 text-base">
   377→          {progress.isAchieved ? (
   378→            <CheckCircle2 className="h-4 w-4 shrink-0 text-emerald-500" aria-hidden="true" />
   379→          ) : (
   380→            <Target className="h-4 w-4 shrink-0 text-primary" aria-hidden="true" />
   381→          )}
   382→          <span className="truncate">{goal.name}</span>
   383→          {paused && <Badge variant="secondary">{t("status.PAUSED")}</Badge>}
   384→          {progress.isExpired && !progress.isAchieved && (
   385→            <Badge className="bg-rose-500/15 text-rose-700 dark:text-rose-400">
   386→              {t("status.EXPIRED")}
   387→            </Badge>
   388→          )}
   389→        </CardTitle>
   390→        <CardDescription>
   391→          {t(`metric.${goal.metric}`)} · {t(`period.${goal.period}`)}
   392→          {goal.endDate ? ` · ${t("until")} ${fmtDateInTz(goal.endDate, "UTC")}` : ""}
   393→        </CardDescription>
   394→        <CardAction>
   395→          <div className="flex items-center gap-1">
   396→            <Button variant="ghost" size="icon" aria-label={t("pauseResume")} onClick={() => void onToggle()}>
   397→              {paused ? <Play className="h-4 w-4" /> : <Pause className="h-4 w-4" />}
   398→            </Button>
   399→            <Button variant="ghost" size="icon" aria-label={t("edit")} onClick={onEdit}>
   400→              <Pencil className="h-4 w-4" />
   401→            </Button>
   402→            <Button
   403→              variant="ghost"
   404→              size="icon"
   405→              aria-label={t("delete")}
   406→              onClick={() => setConfirmOpen(true)}
   407→            >
   408→              <Trash2 className="h-4 w-4" />
   409→            </Button>
   410→          </div>
   411→        </CardAction>
   412→      </CardHeader>
   413→      <CardContent className="space-y-3">
   414→        <div className="flex items-baseline justify-between gap-2">
   415→          <p className="text-2xl font-bold tabular-nums">
   416→            {formatGoalValue(goal.metric, progress.currentValue, currency)}
   417→          </p>
   418→          <p className="text-xs text-muted-foreground">
   419→            {t("targetLabel")} {formatGoalValue(goal.metric, progress.targetValue, currency)}
   420→          </p>
   421→        </div>
   422→        <div className="flex items-center gap-2">
   423→          <div className="h-2.5 flex-1 overflow-hidden rounded-full bg-muted">
   424→            <div
   425→              className={`h-full rounded-full transition-all ${goalBarTone(
   426→                progress.progressPercent,
   427→                progress.isAchieved
   428→              )}`}
   429→              style={{ width: `${Math.max(2, progress.progressPercent)}%` }}
   430→              role="progressbar"
   431→              aria-valuenow={progress.progressPercent}
   432→              aria-valuemin={0}
   433→              aria-valuemax={100}
   434→              aria-label={goal.name}
   435→            />
   436→          </div>
   437→          <span className="w-12 shrink-0 text-right text-sm font-semibold tabular-nums">
   438→            {Math.round(progress.progressPercent)}%
   439→          </span>
   440→        </div>
   441→        <p className="text-xs text-muted-foreground">
   442→          {t("window", {
   443→            from: fmtDateInTz(progress.windowStartDay, "UTC"),
   444→            to: fmtDateInTz(
   445→              progress.windowEndDay === "9999-12-31"
   446→                ? new Date().toISOString()
   447→                : progress.windowEndDay,
   448→              "UTC"
   449→            ),
   450→          })}
   451→        </p>
   452→        {goal.description ? (
   453→          <p className="text-sm text-muted-foreground">{goal.description}</p>
   454→        ) : null}
   455→
   456→        <Dialog open={confirmOpen} onOpenChange={setConfirmOpen}>
   457→          <DialogContent className="sm:max-w-sm">
   458→            <DialogHeader>
   459→              <DialogTitle>{t("confirmDeleteTitle")}</DialogTitle>
   460→              <DialogDescription>{t("confirmDeleteDesc")}</DialogDescription>
   461→            </DialogHeader>
   462→            <DialogFooter>
   463→              <Button variant="outline" onClick={() => setConfirmOpen(false)}>
   464→                {t("cancel")}
   465→              </Button>
   466→              <Button variant="destructive" onClick={() => void onDelete()}>
   467→                {t("delete")}
   468→              </Button>
   469→            </DialogFooter>
   470→          </DialogContent>
   471→        </Dialog>
   472→      </CardContent>
   473→    </Card>
   474→  );
   475→}

   516→      editing

   521→            value: editing.value,

   523→            isActive: editing.isActive,
   524→          }
   525→        : EMPTY_RULE,
   526→    );
   527→  }, [open, editing]);
   528→
   529→  const isCount = form.kind === "MAX_TRADES_PER_DAY";
   530→  const isCustom = form.kind === "CUSTOM";
   531→
   532→  const set = <K extends keyof RuleForm>(key: K, value: RuleForm[K]) =>
   533→    setForm((f) => ({ ...f, [key]: value }));
   534→
   535→  const onSave = async () => {
   536→    if (isCustom ? !form.description.trim() : !form.value.trim()) return;
   537→    setSaving(true);
   538→    try {
   539→      const payload: RuleInput = {
   540→        kind: form.kind,
   541→        value: isCustom ? "1" : form.value.trim(),
   542→        description: form.description.trim() === "" ? null : form.description.trim(),
   543→        isActive: form.isActive,
   544→      };
   545→      if (editing) {
   546→        await update.mutateAsync({ id: editing.id, patch: payload });
   547→        toast({ title: t("updated") });
   548→      } else {
   549→        await create.mutateAsync(payload);
   550→        toast({ title: t("created") });
   551→      }
   552→      onOpenChange(false);
   553→    } catch (error) {
   554→      const err = error instanceof ApiClientError ? error : null;
   555→      toast({
   556→        title: apiErrorMessage(tRoot, err?.code, error instanceof Error ? error.message : ""),
   557→        variant: "destructive",
   558→      });
   559→    } finally {
   560→      setSaving(false);
   561→    }
   562→  };
   563→
   564→  return (
   565→    <Dialog open={open} onOpenChange={onOpenChange}>
   566→      <DialogContent className="sm:max-w-md">
   567→        <DialogHeader>
   568→          <DialogTitle>{editing ? t("editTitle") : t("createTitle")}</DialogTitle>
   569→          <DialogDescription className="sr-only">
   570→            {editing ? t("editTitle") : t("createTitle")}
   571→          </DialogDescription>
   572→        </DialogHeader>
   573→        <div className="flex flex-col gap-4">
   574→          <div className="space-y-1.5">
   575→            <label className="text-xs font-medium text-muted-foreground" htmlFor="rule-kind">
   576→              {t("kind")}
   577→            </label>
   578→            <Select
   579→              value={form.kind}
   580→              onValueChange={(v) => set("kind", v as RuleForm["kind"])}
   581→            >
   582→              <SelectTrigger id="rule-kind">
   583→                <SelectValue />
   584→              </SelectTrigger>
   585→              <SelectContent>
   586→                {RULE_KIND_LIST.map((k) => (
   587→                  <SelectItem key={k} value={k}>
   588→                    {tRules(`kind.${k}`)}
   589→                  </SelectItem>
   590→                ))}
   591→              </SelectContent>
   592→            </Select>
   593→            <p className="text-[11px] text-muted-foreground">{t(`hint.${form.kind}`)}</p>
   594→          </div>
   595→          {!isCustom && (
   596→            <div className="space-y-1.5">
   597→              <label className="text-xs font-medium text-muted-foreground" htmlFor="rule-value">
   598→                {isCount ? t("valueCount") : t("valueAmount", { currency })}

   602→                inputMode="decimal"
   603→                value={form.value}

   609→          <div className="space-y-1.5">
   610→            <label className="text-xs font-medium text-muted-foreground" htmlFor="rule-desc">
   611→              {t("description")}
   612→            </label>
   613→            <Input
   614→              id="rule-desc"
   615→              value={form.description}
   616→              onChange={(e) => set("description", e.target.value)}
   617→              placeholder={t("descriptionPlaceholder")}
   618→              maxLength={500}
   619→            />
   620→          </div>
   621→          <label className="flex items-center gap-2 text-sm">
   622→            <Checkbox
   623→              id="rule-active"
   624→              checked={form.isActive}
   625→              onCheckedChange={(v) => set("isActive", v === true)}
   626→            />
   627→            {t("active")}
   628→          </label>
   629→          <p className="rounded-md border border-dashed p-2 text-[11px] text-muted-foreground">
   630→            {t("reEvalNote")}
   631→          </p>
   632→        </div>
   633→        <DialogFooter>
   634→          <Button variant="outline" onClick={() => onOpenChange(false)}>
   635→            {t("cancel")}
   636→          </Button>
   637→          <Button
   638→            onClick={() => void onSave()}
   639→            disabled={
   640→              saving || (isCustom ? !form.description.trim() : !form.value.trim())
   641→            }
   642→          >
   643→            {saving ? t("saving") : t("save")}
   644→          </Button>
   645→        </DialogFooter>
   646→      </DialogContent>
   647→    </Dialog>
   648→  );
   649→}
