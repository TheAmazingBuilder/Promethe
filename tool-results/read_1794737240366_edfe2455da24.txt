   143→  name: "",
   144→  description: "",
   145→  format: "NUMBER",
   146→  aggregate: "count",
   147→  groups: [],
   148→  useFormula: false,

   153→/** Strip empty filters/groups — the strict server schema rejects them. */
   154→function cleanGroups(groups: GroupDraft[]): CustomMetricInput["definition"]["filterGroups"] {
   155→  return groups
   156→    .map((g) => ({
   157→      filters: g.filters
   158→        .filter((f) => f.values.length > 0)
   159→        .map((f) => ({ field: f.field, values: f.values })),
   160→    }))
   161→    .filter((g) => g.filters.length > 0);
   162→}
   163→
   164→function buildInput(state: BuilderState): CustomMetricInput | null {
   165→  const name = state.name.trim();
   166→  if (name === "") return null;
   167→
   168→  const useFormula =
   169→    state.useFormula &&
   170→    state.expression.trim() !== "" &&
   171→    validateFormula(state.expression).ok;
   172→
   173→  const subMetrics = useFormula
   174→    ? SUB_NAMES.filter((n) => state.subs[n].enabled).map((n) => ({
   175→        name: n,
   176→        aggregate: state.subs[n].aggregate,
   177→        filterGroups: cleanGroups(state.subs[n].groups),
   178→      }))
   179→    : [];
   180→
   181→  if (useFormula && subMetrics.length === 0) return null;
   182→
   183→  return {
   184→    name,
   185→    description: state.description.trim() === "" ? null : state.description.trim(),
   186→    definition: {
   187→      aggregate: state.aggregate,
   188→      filterGroups: cleanGroups(state.groups),
   189→      formula: useFormula
   190→        ? { expression: state.expression.trim(), subMetrics }
   191→        : null,
   192→    },
   193→    format: state.format,
   194→  };
   195→}

   197→/** Defensively read a saved definition into the builder state. */
   198→function stateFromMetric(metric: {
   199→  name: string;
   200→  description: string | null;
   201→  format: string;
   202→  definition: unknown;
   203→}): BuilderState {
   204→  const def = metric.definition;
   205→  const isObject = (v: unknown): v is Record<string, unknown> =>
   206→    typeof v === "object" && v !== null;
   207→
   208→  const state: BuilderState = {
   209→    ...EMPTY_BUILDER,
   210→    subs: { A: { ...EMPTY_SUB }, B: { ...EMPTY_SUB }, C: { ...EMPTY_SUB } },
   211→    name: metric.name,
   212→    description: metric.description ?? "",
   213→    format: (FORMATS as readonly string[]).includes(metric.format)
   214→      ? (metric.format as MetricFormat)
   215→      : "NUMBER",
   216→  };
   217→
   218→  if (isObject(def)) {
   219→    if ((AGGREGATES as readonly string[]).includes(String(def.aggregate))) {
   220→      state.aggregate = def.aggregate as MetricAggregate;
   221→    }
   222→    if (Array.isArray(def.filterGroups)) {
   223→      state.groups = def.filterGroups
   224→        .filter(isObject)
   225→        .map((g) => ({
   226→          filters: (Array.isArray(g.filters) ? g.filters : [])
   227→            .filter(isObject)
   228→            .filter(
   229→              (f) =>
   230→                (FIELDS as readonly string[]).includes(String(f.field)) &&
   231→                Array.isArray(f.values)
   232→            )
   233→            .map((f) => ({
   234→              field: f.field as MetricFilterField,
   235→              values: (f.values as unknown[]).filter(
   236→                (v): v is string => typeof v === "string"
   237→              ),
   238→            })),
   239→        }))
   240→        .filter((g) => g.filters.length > 0);
   241→    }
   242→    if (isObject(def.formula)) {
   243→      const expression =
   244→        typeof def.formula.expression === "string" ? def.formula.expression : "";
   245→      const subs = Array.isArray(def.formula.subMetrics)
   246→        ? def.formula.subMetrics
   247→        : [];
   248→      if (expression !== "" && subs.length > 0) {
   249→        state.useFormula = true;
   250→        state.expression = expression;
   251→        for (const sub of subs) {
   252→          if (!isObject(sub)) continue;
   253→          const name = String(sub.name);
   254→          if (name !== "A" && name !== "B" && name !== "C") continue;
   255→          const groups = Array.isArray(sub.filterGroups)
   256→            ? (sub.filterGroups as unknown[])
   257→                .filter(isObject)
   258→                .map((g) => ({
   259→                  filters: (Array.isArray(g.filters) ? g.filters : [])
   260→                    .filter(isObject)
   261→                    .filter(
   262→                      (f) =>
   263→                        (FIELDS as readonly string[]).includes(String(f.field)) &&
   264→                        Array.isArray(f.values)
   265→                    )
   266→                    .map((f) => ({
   267→                      field: f.field as MetricFilterField,
   268→                      values: (f.values as unknown[]).filter(
   269→                        (v): v is string => typeof v === "string"
   270→                      ),
   271→                    })),
   272→                }))
   273→                .filter((g) => g.filters.length > 0)
   274→          : [];
   275→          state.subs[name] = {
   276→            enabled: true,
   277→            aggregate: (AGGREGATES as readonly string[]).includes(
   278→              String(sub.aggregate)
   279→            )
   280→              ? (sub.aggregate as MetricAggregate)
   281→              : "count",
   282→            groups,
   283→          };
   284→        }
   285→      }
   286→    }
   287→  }
   288→  return state;
   289→}

   293→interface ValueOption {
   294→  value: string;
   295→  label: string;
   296→}
   297→
   298→function useFieldOptions(): Record<MetricFilterField, ValueOption[]> {
   299→  const tRoot = useTranslations();
   300→  const instruments = useInstruments();
   301→  const strategies = useStrategies();
   302→  const tags = useTags();
   303→  const accounts = useAccounts();
   304→  const tMetric = useTranslations("dashboardsSettings.metric");
   305→
   306→  return React.useMemo(() => {
   307→    const weekdays = tMetric.raw("weekdayNames") as unknown as string[];
   308→    const pick = (path: string, v: string): string => {
   309→      try {
   310→        return tRoot(`${path}.${v}` as Parameters<typeof tRoot>[0]);
   311→      } catch {
   312→        return v;
   313→      }
   314→    };
   315→    const uniqueSymbols = Array.from(
   316→      new Set((instruments.data ?? []).map((i) => i.symbol))
   317→    ).sort();
   318→    return {
   319→      instrument: uniqueSymbols.map((s) => ({ value: s, label: s })),
   320→      strategy: (strategies.data ?? []).map((s) => ({
   321→        value: s.name,
   322→        label: s.name,
   323→      })),
   324→      session: TRADING_SESSIONS.map((s) => ({
   325→        value: s,
   326→        label: pick("trading.session", s),
   327→      })),
   328→      direction: (["LONG", "SHORT"] as const).map((d) => ({
   329→        value: d,
   330→        label: pick("trading.direction", d),
   331→      })),
   332→      tag: (tags.data ?? []).map((tag) => ({
   333→        value: tag.name,
   334→        label: tag.name,
   335→      })),
   336→      weekday: [1, 2, 3, 4, 5, 6, 7].map((n) => ({
   337→        value: String(n),
   338→        label: weekdays[n - 1] ?? String(n),
   339→      })),
   340→      hourRange: [],
   341→      emotion: EMOTIONS.map((e) => ({
   342→        value: e,
   343→        label: pick("trading.emotion", e),
   344→      })),
   345→      result: (["WIN", "LOSS", "BREAKEVEN"] as const).map((r) => ({
   346→        value: r,
   347→        label: pick("trading.result", r),
   348→      })),
   349→      account: (accounts.data ?? []).map((a) => ({
   350→        value: a.id,
   351→        label: a.name,
   352→      })),
   353→      setupType: [],
   354→    };
   355→  }, [instruments.data, strategies.data, tags.data, accounts.data, tRoot, tMetric]);
   356→}
   357→
   358→/** Display label of one stored filter value. */
   359→function useValueLabel() {
   360→  const options = useFieldOptions();
   361→  return React.useCallback(
   362→    (field: MetricFilterField, value: string): string => {
   363→      const found = options[field].find((o) => o.value === value);
   364→      return found?.label ?? value;
   365→    },
   366→    [options]
   367→  );
   368→}

   396→function HourRangeAdder({
   397→  onAdd,
   398→  t,
   399→}: {
   400→  onAdd: (range: string) => void;
   401→  t: ReturnType<typeof useTranslations>;
   402→}) {
   403→  const [from, setFrom] = React.useState("8");
   404→  const [to, setTo] = React.useState("10");
   405→  const valid =
   406→    /^\d{1,2}$/.test(from) &&
   407→    /^\d{1,2}$/.test(to) &&
   408→    Number(from) >= 0 &&
   409→    Number(from) < 24 &&
   410→    Number(to) > Number(from) &&
   411→    Number(to) <= 24;
   412→  return (
   413→    <span className="flex items-center gap-1">
   414→      <Input
   415→        type="number"
   416→        min={0}
   417→        max={23}
   418→        value={from}
   419→        onChange={(e) => setFrom(e.target.value)}
   420→        className="h-7 w-14 text-xs"
   421→        aria-label={t("hourFrom")}
   422→      />
   423→      <span className="text-xs text-muted-foreground">→</span>
   424→      <Input
   425→        type="number"
   426→        min={1}
   427→        max={24}
   428→        value={to}
   429→        onChange={(e) => setTo(e.target.value)}
   430→        className="h-7 w-14 text-xs"
   431→        aria-label={t("hourTo")}
   432→      />
   433→      <Button
   434→        type="button"
   435→        size="sm"
   436→        variant="outline"
   437→        className="h-7 px-2 text-xs"
   438→        disabled={!valid}
   439→        onClick={() => {
   440→          onAdd(`${Number(from)}-${Number(to)}`);
   441→          setFrom("8");
   442→          setTo("10");
   443→        }}
   444→      >
   445→        <Plus className="h-3 w-3" aria-hidden="true" />
   446→        {t("hourAdd")}
   447→      </Button>
   448→    </span>
   449→  );
   450→}

   492→function FilterGroupsEditor({
   493→  groups,
   494→  onChange,
   495→}: {
   496→  groups: GroupDraft[];
   497→  onChange: (groups: GroupDraft[]) => void;
   498→}) {
   499→  const t = useTranslations("dashboardsSettings.metric");
   500→  const options = useFieldOptions();
   501→  const valueLabel = useValueLabel();
   502→
   503→  const addGroup = () => onChange([...groups, { filters: [] }]);
   504→
   505→  const addFilter = (groupIndex: number) => {
   506→    onChange(
   507→      groups.map((g, i) =>
   508→        i !== groupIndex ? g : { filters: [...g.filters, { field: "instrument", values: [] }] }
   509→      )
   510→    );
   511→  };
   512→
   513→  return (
   514→    <div className="space-y-3">
   515→      {groups.length === 0 ? (
   516→        <p className="text-xs text-muted-foreground">{t("noFilters")}</p>
   517→      ) : null}
   518→      {groups.map((group, groupIndex) => (
   519→        <div
   520→          key={groupIndex}
   521→          className="space-y-2 rounded-md border border-dashed p-2.5"
   522→        >
   523→          <div className="flex items-center justify-between gap-2">
   524→            <p className="text-xs font-medium text-muted-foreground">
   525→              {t("groupLabel", { index: groupIndex + 1 })}
   526→            </p>
   527→            <div className="flex items-center gap-1">
   528→              <Button
   529→                type="button"
   530→                size="sm"
   531→                variant="ghost"
   532→                className="h-7 px-2 text-xs"
   533→                onClick={() => addFilter(groupIndex)}
   534→              >
   535→                <Plus className="h-3 w-3" aria-hidden="true" />
   536→                {t("addFilter")}
   537→              </Button>
   538→              <Button
   539→                type="button"
   540→                size="icon"
   541→                variant="ghost"
   542→                className="h-7 w-7"
   543→                aria-label={t("removeGroup")}
   544→                onClick={() =>
   545→                  onChange(groups.filter((_, i) => i !== groupIndex))
   546→                }
   547→              >
   548→                <Trash2 className="h-3.5 w-3.5" aria-hidden="true" />
   549→              </Button>
   550→            </div>
   551→          </div>
   552→
   553→          {group.filters.length === 0 ? (
   554→            <p className="text-xs text-muted-foreground">{t("emptyGroup")}</p>
   555→          ) : null}
   556→
   557→          {group.filters.map((filter, filterIndex) => {
   558→            const setFilter = (patch: Partial<FilterDraft>) =>
   559→              onChange(
   560→                groups.map((g, i) =>
   561→                  i !== groupIndex
   562→                    ? g
   563→                    : {
   564→                        filters: g.filters.map((f, j) =>
   565→                          j !== filterIndex ? f : { ...f, ...patch }
   566→                        ),
   567→                      }
   568→                )
   569→              );
   570→            const removeFilter = () =>
   571→              onChange(
   572→                groups.map((g, i) =>
   573→                  i !== groupIndex
   574→                    ? g
   575→                    : { filters: g.filters.filter((_, j) => j !== filterIndex) }
   576→                )
   577→              );
   578→            const toggleValue = (value: string, checked: boolean) => {
   579→              const next = checked
   580→                ? [...filter.values, value]
   581→                : filter.values.filter((v) => v !== value);
   582→              setFilter({ values: next.slice(0, 24) });
   583→            };
   584→            return (
   585→              <div key={filterIndex} className="space-y-1.5">
   586→                <div className="flex items-center gap-2">
   587→                  <Select
   588→                    value={filter.field}
   589→                    onValueChange={(v) =>
   590→                      setFilter({ field: v as MetricFilterField, values: [] })
   591→                    }
   592→                  >
   593→                    <SelectTrigger
   594→                      size="sm"
   595→                      className="h-7 w-44 text-xs"
   596→                      aria-label={t("fieldLabel")}
   597→                    >
   598→                      <SelectValue />
   599→                    </SelectTrigger>
   600→                    <SelectContent>
   601→                      {FIELDS.map((field) => (
   602→                        <SelectItem key={field} value={field}>
   603→                          {t(`field.${field}`)}
   604→                        </SelectItem>
   605→                      ))}
   606→                    </SelectContent>
   607→                  </Select>
   608→                  <span className="text-xs text-muted-foreground">
   609→                    {t("isOneOf")}
   610→                  </span>
   611→                  <Button
   612→                    type="button"
   613→                    size="icon"
   614→                    variant="ghost"
   615→                    className="ml-auto h-7 w-7"
   616→                    aria-label={t("removeFilter")}
   617→                    onClick={removeFilter}
   618→                  >
   619→                    <X className="h-3.5 w-3.5" aria-hidden="true" />
   620→                  </Button>
   621→                </div>
   622→
   623→                {filter.field === "hourRange" ? (
   624→                  <div className="flex flex-wrap items-center gap-1.5">
   625→                    <HourRangeAdder
   626→                      t={t}
   627→                      onAdd={(range) => {
   628→                        if (!filter.values.includes(range)) {
   629→                          setFilter({ values: [...filter.values, range] });
   630→                        }
   631→                      }}
   632→                    />
   633→                    {filter.values.map((range) => (
   634→                      <FilterChip
   635→                        key={range}
   636→                        label={range}

   638→                        onRemove={() => toggleValue(range, false)}
   639→                      />

   646→                      values={filter.values}

   648→                    />
   649→                    {filter.values.map((value) => (

   651→                        key={value}

   657→                  </div>
   658→                ) : (
   659→                  <div
   660→                    className="max-h-40 overflow-y-auto rounded-md border p-2"

   666→                        {t("noValues")}

   674→                              key={option.value}
   675→                              className={cn(
   676→                                "flex cursor-pointer items-center gap-1 rounded-full border px-2 py-0.5 text-xs transition-colors",
   677→                                checked
   678→                                  ? "border-primary bg-accent font-medium"
   679→                                  : "text-muted-foreground hover:bg-accent"
   680→                              )}
   681→                            >
   682→                              <Checkbox
   683→                                checked={checked}
   684→                                onCheckedChange={(v) =>
   685→                                  toggleValue(option.value, v === true)
   686→                                }
   687→                                className="h-3.5 w-3.5"
   688→                                aria-label={option.label}
   689→                              />
   690→                              {option.label}
   691→                            </label>
   692→                          );
   693→                        })}
   694→                      </div>
   695→                    )}
   696→                  </div>
   697→                )}
   698→              </div>
   699→            );
   700→          })}
   701→
   702→          {/* Group summary */}
   703→          <p className="text-[11px] text-muted-foreground">
   704→            {groupSummary(group, valueLabel, t)}
   705→          </p>
   706→        </div>
   707→      ))}
   708→
   709→      <Button
   710→        type="button"
   711→        size="sm"
   712→        variant="outline"
   713→        onClick={addGroup}
   714→        disabled={groups.length >= 6}
   715→      >
   716→        <Plus className="h-3.5 w-3.5" aria-hidden="true" />
   717→        {t("addGroup")}
   718→      </Button>
   719→    </div>
   720→  );
   721→}

   723→/** "instrument is XAUUSD OR EURUSD" — one line per group. */
   724→function groupSummary(
   725→  group: GroupDraft,
   726→  valueLabel: (field: MetricFilterField, value: string) => string,
   727→  t: ReturnType<typeof useTranslations>
   728→): string {
   729→  const parts = group.filters
   730→    .filter((f) => f.values.length > 0)
   731→    .map(
   732→      (f) =>
   733→        `${t(`field.${f.field}`)} ${t("isOneOf")} ${f.values
   734→          .map((v) => valueLabel(f.field, v))
   735→          .join(` ${t("or")} `)}`
   736→    );
   737→  if (parts.length === 0) return t("emptyGroup");
   738→  return parts.join(` ${t("and")} `);
   739→}

   741→// ── Preview ──────────────────────────────────────────────────────────────────
   742→
   743→interface PreviewResult {
   744→  value: string | null;
   745→  sampleSize: number;
   746→  lowSample: boolean;
   747→}
   748→
   749→function formatMetricValue(
   750→  value: string | null,
   751→  format: MetricFormat,
   752→  currency: string,
   753→  t: ReturnType<typeof useTranslations>
   754→): string {
   755→  if (value == null) return "—";
   756→  if (value === "inf") return "∞";
   757→  const n = Number(value);
   758→  if (!Number.isFinite(n)) return "—";
   759→  switch (format) {
   760→    case "CURRENCY":
   761→      return fmtMoney(n, currency);
   762→    case "PERCENT":
   763→      return fmtPct(n);
   764→    case "R":
   765→      return `${fmtNum(n)} R`;
   766→    case "PIPS":
   767→      return `${fmtNum(n)} pips`;
   768→    case "DURATION": {
   769→      const h = Math.floor(n / 60);
   770→      const m = Math.round(n % 60);
   771→      return h > 0 ? t("durationHm", { h, m }) : t("durationM", { m });
   772→    }
   773→    default:
   774→      return fmtNum(n);
   775→  }
   776→}
