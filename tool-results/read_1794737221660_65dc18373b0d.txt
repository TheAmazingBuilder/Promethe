   135→/** Trade detail sheet — read-only fiche with close dialog and screenshot flow. */
   136→export function TradeDetailView({ tradeId }: { tradeId: string }) {
   137→  // v2.5 M2 — overview / price chart / replay tabs
   138→  const [detailTab, setDetailTab] = React.useState<
   139→    "overview" | "chart" | "replay"
   140→  >("overview");
   141→  const t = useTranslations("trades.detail");
   142→  const tForm = useTranslations("trades.form");
   143→  const tCommon = useTranslations("common");
   144→  const tRoot = useTranslations();
   145→  const labels = useTradingLabels();
   146→  const { session } = useSession();
   147→  const timezone = session?.profile.timezone ?? "UTC";
   148→  const { toast } = useToast();
   149→
   150→  const tradeQuery = useTrade(tradeId || null);
   151→  const instruments = useInstruments();
   152→  const updateTrade = useUpdateTrade();
   153→  const deleteTrade = useDeleteTrade();
   154→  const closeTrade = useCloseTrade();
   155→  const uploadScreenshot = useUploadScreenshot();
   156→  const deleteScreenshot = useDeleteScreenshot();
   157→
   158→  const [deleteOpen, setDeleteOpen] = React.useState(false);
   159→  const [closeOpen, setCloseOpen] = React.useState(false);
   160→  const [shareOpen, setShareOpen] = React.useState(false);
   161→  const [exportOpen, setExportOpen] = React.useState(false);
   162→  const [exitPrice, setExitPrice] = React.useState("");
   163→  const [exitTime, setExitTime] = React.useState("");
   164→  const [exitReason, setExitReason] = React.useState("none");
   165→  const [commission, setCommission] = React.useState("");
   166→  const [swap, setSwap] = React.useState("");
   167→  const [fees, setFees] = React.useState("");
   168→  const [rate, setRate] = React.useState("");
   169→  const [useManual, setUseManual] = React.useState(false);
   170→  const [manualPnl, setManualPnl] = React.useState("");
   171→  const [uploading, setUploading] = React.useState(0);
   172→
   173→  const fileInputRef = React.useRef<HTMLInputElement>(null);
   174→
   175→  const trade = tradeQuery.data;
   176→  const spec = React.useMemo(() => {
   177→    if (!trade || !instruments.data) return null;
   178→    const found = instruments.data.find((i) => i.id === trade.instrumentId);
   179→    return found ? toSpec(found) : null;
   180→  }, [trade, instruments.data]);
   181→  const isEstimate =
   182→    trade != null && spec != null && spec.quoteCurrency !== trade.accountCurrency;
   183→
   184→  // Reset the close form each time the dialog opens.
   185→  React.useEffect(() => {
   186→    if (!closeOpen || !trade) return;
   187→    setExitPrice("");
   188→    setExitTime(isoToLocalInput(new Date().toISOString()));
   189→    setExitReason(trade.exitReason ?? "none");
   190→    setCommission(trade.commission);
   191→    setSwap(trade.swap);
   192→    setFees(trade.fees);
   193→    setRate("");
   194→    setUseManual(trade.pnlSource === "MANUAL");
   195→    setManualPnl("");
   196→  }, [closeOpen, trade]);
   197→
   198→  const mutationErrorToast = (err: unknown) => {
   199→    const e = err as ApiClientError;
   200→    toast({
   201→      title: apiErrorMessage(
   202→        tRoot,
   203→        e?.code,
   204→        err instanceof Error ? err.message : ""
   205→      ),
   206→      variant: "destructive",
   207→    });
   208→  };
   209→
   210→  const uploadBlobs = async (blobs: Blob[]) => {
   211→    if (!trade) return;
   212→    for (const blob of blobs) {
   213→      if (!looksLikeImage(blob) || blob.size > MAX_UPLOAD_BYTES) {
   214→        toast({ title: t("screensUploadError"), variant: "destructive" });
   215→        continue;
   216→      }
   217→      toast({ title: t("screensCompressing") });
   218→      const compressed = await compressImage(blob);
   219→      setUploading((n) => n + 1);
   220→      await new Promise<void>((resolve) => {
   221→        uploadScreenshot.mutate(
   222→          { tradeId: trade.id, blob: compressed, label: null, caption: null },
   223→          {
   224→            onSuccess: () => {
   225→              toast({ title: t("screensUploaded") });
   226→              resolve();
   227→            },
   228→            onError: (err) => {
   229→              mutationErrorToast(err);
   230→              resolve();
   231→            },
   232→          }
   233→        );
   234→      });
   235→      setUploading((n) => Math.max(0, n - 1));
   236→    }
   237→  };
   238→
   239→  const onFiles = (files: FileList | null) => {
   240→    if (!files || files.length === 0) return;
   241→    void uploadBlobs(Array.from(files));
   242→  };
   243→
   244→  const onPasteScreens = (e: React.ClipboardEvent<HTMLElement>) => {
   245→    const image = Array.from(e.clipboardData.files).find((f) =>
   246→      f.type.startsWith("image/")
   247→    );
   248→    if (image) {
   249→      e.preventDefault();
   250→      void uploadBlobs([image]);
   251→      return;
   252→    }
   253→    void (async () => {
   254→      const blob = await readClipboardImage();
   255→      if (blob) void uploadBlobs([blob]);
   256→    })();
   257→  };
   258→
   259→  const toggleReview = () => {
   260→    if (!trade) return;
   261→    updateTrade.mutate(
   262→      {
   263→        id: trade.id,
   264→        patch: {
   265→          reviewStatus:
   266→            trade.reviewStatus === "REVIEWED" ? "NOT_REVIEWED" : "REVIEWED",
   267→        },
   268→      },
   269→      {
   270→        onSuccess: () => toast({ title: t("reviewToast") }),
   271→        onError: mutationErrorToast,
   272→      }
   273→    );
   274→  };
   275→
   276→  const confirmDelete = () => {
   277→    if (!trade) return;
   278→    deleteTrade.mutate(trade.id, {
   279→      onSuccess: () => {
   280→        toast({ title: t("deletedToast") });
   281→        navigate("#/trades");
   282→      },
   283→      onError: mutationErrorToast,
   284→    });
   285→  };
   286→
   287→  const submitClose = () => {
   288→    if (!trade) return;
   289→    const price = exitPrice.trim();
   290→    if (!price) return;
   291→    closeTrade.mutate(
   292→      {
   293→        id: trade.id,
   294→        input: {
   295→          exitPrice: price,
   296→          exitTime: localInputToIso(exitTime),
   297→          exitReason:
   298→            exitReason === "none"
   299→              ? null
   300→              : (exitReason as CloseTradeInput["exitReason"]),
   301→          commission: commission.trim() || null,
   302→          swap: swap.trim() || null,
   303→          fees: fees.trim() || null,
   304→          quoteToAccountRate: isEstimate ? rate.trim() || null : null,
   305→          pnlSource: useManual ? "MANUAL" : null,
   306→          manualNetPnl: useManual && manualPnl.trim() ? manualPnl.trim() : null,
   307→        },
   308→      },
   309→      {
   310→        onSuccess: () => {
   311→          toast({ title: t("closedToast") });
   312→          setCloseOpen(false);
   313→        },
   314→        onError: mutationErrorToast,
   315→      }
   316→    );
   317→  };
   318→
   319→  // ── Load states ──
   320→  if (tradeQuery.isLoading) {
   321→    return (
   322→      <div className="space-y-4">
   323→        <Skeleton className="h-8 w-24" />
   324→        <Skeleton className="h-10 w-2/3" />
   325→        <Skeleton className="h-4 w-1/2" />
   326→        <div className="grid grid-cols-2 gap-3 lg:grid-cols-4">
   327→          {Array.from({ length: 4 }).map((_, i) => (
   328→            <Skeleton key={i} className="h-24 w-full" />
   329→          ))}
   330→        </div>
   331→        <Skeleton className="h-40 w-full" />
   332→      </div>
   333→    );
   334→  }
   335→
   336→  if (tradeQuery.isError) {
   337→    const err = tradeQuery.error as ApiClientError | null;
   338→    const notFound = err?.status === 404;
   339→    return (
   340→      <div className="space-y-4">
   341→        <Button
   342→          variant="ghost"
   343→          size="sm"
   344→          className="-ml-2 text-muted-foreground"
   345→          onClick={() => navigate("#/trades")}
   346→        >
   347→          <ArrowLeft className="h-4 w-4" aria-hidden="true" />
   348→          {tCommon("back")}
   349→        </Button>
   350→        <Card className="gap-0 py-4">
   351→          <CardContent className="px-4">
   352→            <div className="flex flex-col items-center gap-3 py-10 text-center">
   353→              <AlertCircle
   354→                className="h-8 w-8 text-muted-foreground"
   355→                aria-hidden="true"
   356→              />
   357→              <p className="text-sm text-muted-foreground">
   358→                {notFound ? t("notFound") : t("loadError")}
   359→              </p>
   360→              {notFound ? (
   361→                <Button
   362→                  variant="outline"
   363→                  size="sm"
   364→                  onClick={() => navigate("#/trades")}
   365→                >
   366→                  {tCommon("back")}
   367→                </Button>
   368→              ) : (
   369→                <Button
   370→                  variant="outline"
   371→                  size="sm"
   372→                  onClick={() => void tradeQuery.refetch()}
   373→                >
   374→                  {tCommon("retry")}
   375→                </Button>
   376→              )}
   377→            </div>
   378→          </CardContent>
   379→        </Card>
   380→      </div>
   381→    );
   382→  }
   383→
   384→  if (!trade) {
   385→    return (
   386→      <div className="space-y-4">
   387→        <Button
   388→          variant="ghost"
   389→          size="sm"
   390→          className="-ml-2 text-muted-foreground"
   391→          onClick={() => navigate("#/trades")}
   392→        >
   393→          <ArrowLeft className="h-4 w-4" aria-hidden="true" />
   394→          {tCommon("back")}
   395→        </Button>
   396→        <Card className="gap-0 py-4">
   397→          <CardContent className="px-4">
   398→            <div className="flex flex-col items-center gap-3 py-10 text-center">
   399→              <AlertCircle
   400→                className="h-8 w-8 text-muted-foreground"
   401→                aria-hidden="true"
   402→              />
   403→              <p className="text-sm text-muted-foreground">{t("notFound")}</p>
   404→              <Button
   405→                variant="outline"
   406→                size="sm"
   407→                onClick={() => navigate("#/trades")}
   408→              >
   409→                {tCommon("back")}
   410→              </Button>
   411→            </div>
   412→          </CardContent>
   413→        </Card>
   414→      </div>
   415→    );
   416→  }
   417→
   418→  // ── Derived data ──
   419→  const digits = trade.digits;
   420→  const costs = toD(trade.commission)
   421→    .plus(toD(trade.swap))
   422→    .plus(toD(trade.fees));
   423→  const subtitle = [
   424→    trade.accountName,
   425→    trade.strategyName ?? null,
   426→    trade.session ? labels.session(trade.session) : null,
   427→    trade.timeframe ?? null,
   428→    labels.orderType(trade.orderType),
   429→    trade.setupType ?? null,
   430→    trade.marketCondition ? labels.marketCondition(trade.marketCondition) : null,
   431→  ]
   432→    .filter((v): v is string => v !== null)
   433→    .join(" · ");
   434→
   435→  const levels: Array<[string, string]> = [
   436→    [tForm("entryPrice"), fmtPrice(trade.entryPrice, digits)],
   437→    [tForm("exitPrice"), fmtPrice(trade.exitPrice, digits)],
   438→    [tForm("stopLoss"), fmtPrice(trade.stopLoss, digits)],
   439→    [tForm("takeProfit"), fmtPrice(trade.takeProfit, digits)],
   440→    [t("initialStop"), fmtPrice(trade.initialStopLoss, digits)],
   441→    [t("initialTp"), fmtPrice(trade.initialTakeProfit, digits)],
   442→    [t("mfe"), fmtPrice(trade.mfePrice, digits)],
   443→    [t("mae"), fmtPrice(trade.maePrice, digits)],
   444→  ];
   445→
   446→  const context: Array<[string, string]> = [
   447→    [t("account"), trade.accountName],
   448→    [t("strategy"), trade.strategyName ?? "—"],
   449→    [t("sessionLabel"), labels.session(trade.session)],
   450→    [t("timeframe"), trade.timeframe ?? "—"],
   451→    [t("orderType"), labels.orderType(trade.orderType)],
   452→    [t("setupType"), trade.setupType ?? "—"],
   453→    [t("marketCondition"), labels.marketCondition(trade.marketCondition)],
   454→    ...(trade.exitReason
   455→      ? ([[tForm("exitReason"), labels.exitReason(trade.exitReason)]] as Array<
   456→          [string, string]
   457→        >)
   458→      : []),
   459→  ];
   460→
   461→  const checklistEntries = Object.entries(trade.checklistResults ?? {});
   462→
   463→  return (
   464→    <div className="space-y-4">
   465→      {/* ── Header ── */}
   466→      <Button
   467→        variant="ghost"
   468→        size="sm"
   469→        className="-ml-2 text-muted-foreground"
   470→        onClick={() => navigate("#/trades")}
   471→      >
   472→        <ArrowLeft className="h-4 w-4" aria-hidden="true" />
   473→        {tCommon("back")}
   474→      </Button>
   475→
   476→      <div className="flex flex-col gap-3 lg:flex-row lg:items-start lg:justify-between">
   477→        <div className="min-w-0 space-y-1.5">
   478→          <div className="flex flex-wrap items-center gap-2">
   479→            <h2 className="text-2xl font-bold tabular-nums">{trade.symbol}</h2>
   480→            <DirectionBadge direction={trade.direction} />
   481→            <StatusBadge status={trade.status} />
   482→            <ResultBadge result={trade.resultType} />
   483→            {trade.isDraft && <DraftBadge />}
   484→          </div>
   485→          <p className="text-sm text-muted-foreground">{subtitle}</p>
   486→          <p className="text-xs text-muted-foreground">
   487→            {t("headerEntry")} :{" "}
   488→            <span className="tabular-nums">
   489→              {fmtDateTimeInTz(trade.entryTime, timezone)} ·{" "}
   490→              {fmtPrice(trade.entryPrice, digits)}
   491→            </span>
   492→            <span aria-hidden="true"> · </span>
   493→            {t("headerExit")} :{" "}
   494→            <span className="tabular-nums">
   495→              {trade.exitTime
   496→                ? `${fmtDateTimeInTz(trade.exitTime, timezone)} · ${fmtPrice(trade.exitPrice, digits)}`
   497→                : "—"}
   498→            </span>
   499→          </p>
   500→        </div>
   501→        <div className="flex flex-wrap items-center gap-2">
   502→          {trade.status === "OPEN" && (
   503→            <Button onClick={() => setCloseOpen(true)}>
   504→              <Flag className="h-4 w-4" aria-hidden="true" />
   505→              {t("closeSubmit")}
   506→            </Button>
   507→          )}
   508→          <Button variant="outline" onClick={() => setShareOpen(true)}>
   509→            <Share2 className="h-4 w-4" aria-hidden="true" />
   510→            {tRoot("shareDialog.trigger")}
   511→          </Button>
   512→          <Button variant="outline" onClick={() => setExportOpen(true)}>
   513→            <ImageDown className="h-4 w-4" aria-hidden="true" />
   514→            {tRoot("exportCard.trigger")}
   515→          </Button>
   516→          <Button
   517→            variant="outline"
   518→            onClick={toggleReview}
   519→            disabled={updateTrade.isPending}
   520→          >
   521→            {updateTrade.isPending ? (
   522→              <Loader2 className="h-4 w-4 animate-spin" aria-hidden="true" />
   523→            ) : (
   524→              <CheckCircle2 className="h-4 w-4" aria-hidden="true" />
   525→            )}
   526→            {trade.reviewStatus === "REVIEWED"
   527→              ? t("reviewUnmark")
   528→              : t("reviewMark")}
   529→          </Button>
   530→          <Button
   531→            variant="outline"
   532→            onClick={() => navigate(`#/trades/${trade.id}/edit`)}
   533→          >
   534→            <Pencil className="h-4 w-4" aria-hidden="true" />
   535→            {t("edit")}
   536→          </Button>
   537→          <Button
   538→            variant="outline"
   539→            className="text-destructive hover:text-destructive"
   540→            onClick={() => setDeleteOpen(true)}
   541→          >
   542→            <Trash2 className="h-4 w-4" aria-hidden="true" />
   543→            {t("delete")}
   544→          </Button>
   545→        </div>
   546→      </div>
   547→
   548→      {/* ── v2.5 M2 — overview / price chart / replay tabs ── */}
   549→      <Tabs value={detailTab} onValueChange={(v) => setDetailTab(v as typeof detailTab)}>
   550→        <TabsList>
   551→          <TabsTrigger value="overview">{t("tabOverview")}</TabsTrigger>
   552→          <TabsTrigger value="chart">{t("tabChart")}</TabsTrigger>
   553→          <TabsTrigger value="replay">{t("tabReplay")}</TabsTrigger>
   554→        </TabsList>
   555→      </Tabs>
   556→
   557→      {detailTab === "chart" ? (
   558→        <LazyPriceChart trade={trade} />
   559→      ) : detailTab === "replay" ? (
   560→        <LazyReplayPlayer trade={trade} />
   561→      ) : (
   562→        <>
   563→
   564→      {/* ── KPI grid ── */}
   565→      <section className="grid grid-cols-2 gap-3 lg:grid-cols-4">
   566→        <KpiCell
   567→          label={t("netPnl")}
   568→          value={fmtSignedMoney(trade.netPnl, trade.pnlCurrency)}
   569→          valueClassName={`text-2xl ${pnlClass(trade.netPnl)}`}
   570→          hint={
   571→            isEstimate || trade.pnlSource === "MANUAL" ? (
   572→              <div className="flex flex-wrap items-center gap-1.5">
   573→                {isEstimate && (
   574→                  <Tooltip>
   575→                    <TooltipTrigger asChild>
   576→                      <Badge
   577→                        variant="outline"
   578→                        className="cursor-help border-amber-500/40 bg-amber-500/10 text-[10px] text-amber-700 dark:text-amber-400"
   579→                      >
   580→                        {t("estimate")}
   581→                      </Badge>
   582→                    </TooltipTrigger>
   583→                    <TooltipContent>
   584→                      {t("estimateHint", {
