   104→   104→/** Local calendar day YYYY-MM-DD (client-side default, effect-only). */
   105→   105→function todayLocalDay(): string {
   106→   106→  const d = new Date();
   107→   107→  const pad = (n: number) => String(n).padStart(2, "0");
   108→   108→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}`;
   109→   109→}

   125→   125→  const { toast } = useToast();
   126→   126→  const create = useCreateGoal();

   150→   150→            ...EMPTY_GOAL,

   171→   171→        status: form.status,
   172→   172→      };

   174→   174→        await update.mutateAsync({ id: editing.id, patch: payload });

   177→   177→        await create.mutateAsync(payload);
   178→   178→        toast({ title: t("created") });

   184→   184→        title: apiErrorMessage(tRoot, err?.code, error instanceof Error ? error.message : ""),

   189→   189→    }

   207→   207→              {t("name")}
   208→   208→            </label>
   209→   209→            <Input
   210→   210→              id="goal-name"
   211→   211→              value={form.name}
   212→   212→              onChange={(e) => set("name", e.target.value)}
   213→   213→              placeholder={t("namePlaceholder")}
   214→   214→              maxLength={80}
   215→   215→            />
   216→   216→          </div>
   217→   217→          <div className="space-y-1.5">
   218→   218→            <label className="text-xs font-medium text-muted-foreground" htmlFor="goal-desc">
   219→   219→              {t("description")}
   220→   220→            </label>
   221→   221→            <Input
   222→   222→              id="goal-desc"
   223→   223→              value={form.description}
   224→   224→              onChange={(e) => set("description", e.target.value)}
   225→   225→              placeholder={t("descriptionPlaceholder")}
   226→   226→              maxLength={2000}
   227→   227→            />
   228→   228→          </div>
   229→   229→          <div className="grid gap-3 sm:grid-cols-2">
   230→   230→            <div className="space-y-1.5">

   237→   237→              >

   239→   239→                  <SelectValue />

   333→   333→function GoalCard({
   334→   334→  goal,
   335→   335→  currency,
   336→   336→  onEdit,
   337→   337→}: {
   338→   338→  goal: GoalWithProgressDTO;
   339→   339→  currency: string;
   340→   340→  onEdit: () => void;
   341→   341→}) {
   342→   342→  const t = useTranslations("goalsView");
   343→   343→  const { toast } = useToast();
   344→   344→  const update = useUpdateGoal();
   345→   345→  const del = useDeleteGoal();
   346→   346→  const [confirmOpen, setConfirmOpen] = React.useState(false);
   347→   347→
   348→   348→  const paused = goal.status === "PAUSED";
   349→   349→  const { progress } = goal;
   350→   350→
   351→   351→  const onToggle = async () => {
   352→   352→    try {
   353→   353→      await update.mutateAsync({
   354→   354→        id: goal.id,
   355→   355→        patch: { status: paused ? "ACTIVE" : "PAUSED" },
   356→   356→      });
   357→   357→      toast({ title: paused ? t("resumedToast") : t("pausedToast") });
   358→   358→    } catch {
   359→   359→      toast({ title: t("errorToast"), variant: "destructive" });
   360→   360→    }
   361→   361→  };
   362→   362→
   363→   363→  const onDelete = async () => {
   364→   364→    try {
   365→   365→      await del.mutateAsync(goal.id);
   366→   366→      toast({ title: t("deletedToast") });
   367→   367→      setConfirmOpen(false);
   368→   368→    } catch {
   369→   369→      toast({ title: t("errorToast"), variant: "destructive" });
   370→   370→    }
   371→   371→  };
   372→   372→
   373→   373→  return (
   374→   374→    <Card className={paused ? "opacity-60" : undefined}>
   375→   375→      <CardHeader>
   376→   376→        <CardTitle className="flex items-center gap-2 text-base">
   377→   377→          {progress.isAchieved ? (
   378→   378→            <CheckCircle2 className="h-4 w-4 shrink-0 text-emerald-500" aria-hidden="true" />
   379→   379→          ) : (
   380→   380→            <Target className="h-4 w-4 shrink-0 text-primary" aria-hidden="true" />
   381→   381→          )}
   382→   382→          <span className="truncate">{goal.name}</span>
   383→   383→          {paused && <Badge variant="secondary">{t("status.PAUSED")}</Badge>}
   384→   384→          {progress.isExpired && !progress.isAchieved && (
   385→   385→            <Badge className="bg-rose-500/15 text-rose-700 dark:text-rose-400">
   386→   386→              {t("status.EXPIRED")}
   387→   387→            </Badge>
   388→   388→          )}
   389→   389→        </CardTitle>
   390→   390→        <CardDescription>
   391→   391→          {t(`metric.${goal.metric}`)} · {t(`period.${goal.period}`)}
   392→   392→          {goal.endDate ? ` · ${t("until")} ${fmtDateInTz(goal.endDate, "UTC")}` : ""}
   393→   393→        </CardDescription>
   394→   394→        <CardAction>
   395→   395→          <div className="flex items-center gap-1">
   396→   396→            <Button variant="ghost" size="icon" aria-label={t("pauseResume")} onClick={() => void onToggle()}>
   397→   397→              {paused ? <Play className="h-4 w-4" /> : <Pause className="h-4 w-4" />}
   398→   398→            </Button>
   399→   399→            <Button variant="ghost" size="icon" aria-label={t("edit")} onClick={onEdit}>
   400→   400→              <Pencil className="h-4 w-4" />
   401→   401→            </Button>
   402→   402→            <Button
   403→   403→              variant="ghost"
   404→   404→              size="icon"
   405→   405→              aria-label={t("delete")}
   406→   406→              onClick={() => setConfirmOpen(true)}
   407→   407→            >
   408→   408→              <Trash2 className="h-4 w-4" />
   409→   409→            </Button>
   410→   410→          </div>
   411→   411→        </CardAction>
   412→   412→      </CardHeader>
   413→   413→      <CardContent className="space-y-3">
   414→   414→        <div className="flex items-baseline justify-between gap-2">
   415→   415→          <p className="text-2xl font-bold tabular-nums">
   416→   416→            {formatGoalValue(goal.metric, progress.currentValue, currency)}
   417→   417→          </p>
   418→   418→          <p className="text-xs text-muted-foreground">
   419→   419→            {t("targetLabel")} {formatGoalValue(goal.metric, progress.targetValue, currency)}
   420→   420→          </p>
   421→   421→        </div>
   422→   422→        <div className="flex items-center gap-2">
   423→   423→          <div className="h-2.5 flex-1 overflow-hidden rounded-full bg-muted">
   424→   424→            <div
   425→   425→              className={`h-full rounded-full transition-all ${goalBarTone(
   426→   426→                progress.progressPercent,
   427→   427→                progress.isAchieved
   428→   428→              )}`}
   429→   429→              style={{ width: `${Math.max(2, progress.progressPercent)}%` }}
   430→   430→              role="progressbar"
   431→   431→              aria-valuenow={progress.progressPercent}
   432→   432→              aria-valuemin={0}
   433→   433→              aria-valuemax={100}
   434→   434→              aria-label={goal.name}
   435→   435→            />
   436→   436→          </div>
   437→   437→          <span className="w-12 shrink-0 text-right text-sm font-semibold tabular-nums">
   438→   438→            {Math.round(progress.progressPercent)}%
   439→   439→          </span>
   440→   440→        </div>
   441→   441→        <p className="text-xs text-muted-foreground">
   442→   442→          {t("window", {
   443→   443→            from: fmtDateInTz(progress.windowStartDay, "UTC"),
   444→   444→            to: fmtDateInTz(
   445→   445→              progress.windowEndDay === "9999-12-31"
   446→   446→                ? new Date().toISOString()
   447→   447→                : progress.windowEndDay,
   448→   448→              "UTC"
   449→   449→            ),
   450→   450→          })}
   451→   451→        </p>
   452→   452→        {goal.description ? (
   453→   453→          <p className="text-sm text-muted-foreground">{goal.description}</p>
   454→   454→        ) : null}
   455→   455→
   456→   456→        <Dialog open={confirmOpen} onOpenChange={setConfirmOpen}>
   457→   457→          <DialogContent className="sm:max-w-sm">
   458→   458→            <DialogHeader>
   459→   459→              <DialogTitle>{t("confirmDeleteTitle")}</DialogTitle>
   460→   460→              <DialogDescription>{t("confirmDeleteDesc")}</DialogDescription>
   461→   461→            </DialogHeader>
   462→   462→            <DialogFooter>
   463→   463→              <Button variant="outline" onClick={() => setConfirmOpen(false)}>
   464→   464→                {t("cancel")}
   465→   465→              </Button>
   466→   466→              <Button variant="destructive" onClick={() => void onDelete()}>
   467→   467→                {t("delete")}
   468→   468→              </Button>
   469→   469→            </DialogFooter>
   470→   470→          </DialogContent>
   471→   471→        </Dialog>
   472→   472→      </CardContent>
   473→   473→    </Card>
   474→   474→  );
   475→   475→}

   493→   493→function RuleEditor({
   494→   494→  open,
   495→   495→  onOpenChange,
   496→   496→  editing,
   497→   497→  currency,
   498→   498→}: {
   499→   499→  open: boolean;
   500→   500→  onOpenChange: (open: boolean) => void;
   501→   501→  editing: RuleDTO | null;
   502→   502→  currency: string;
   503→   503→}) {
   504→   504→  const t = useTranslations("goalsView.rules.form");
   505→   505→  const tRules = useTranslations("goalsView.rules");
   506→   506→  const tRoot = useTranslations();
   507→   507→  const { toast } = useToast();
   508→   508→  const create = useCreateRule();
   509→   509→  const update = useUpdateRule();
   510→   510→  const [form, setForm] = React.useState<RuleForm>(EMPTY_RULE);
   511→   511→  const [saving, setSaving] = React.useState(false);
   512→   512→
   513→   513→  React.useEffect(() => {
   514→   514→    if (!open) return;
   515→   515→    setForm(
   516→   516→      editing
   517→   517→        ? {
   518→   518→            kind: (RULE_KIND_LIST as readonly string[]).includes(editing.kind)
   519→   519→              ? (editing.kind as RuleForm["kind"])
   520→   520→              : "MAX_TRADES_PER_DAY",
   521→   521→            value: editing.value,
   522→   522→            description: editing.description ?? "",
   523→   523→            isActive: editing.isActive,
   524→   524→          }
   525→   525→        : EMPTY_RULE,
   526→   526→    );
   527→   527→  }, [open, editing]);
   528→   528→
   529→   529→  const isCount = form.kind === "MAX_TRADES_PER_DAY";
   530→   530→  const isCustom = form.kind === "CUSTOM";
   531→   531→
   532→   532→  const set = <K extends keyof RuleForm>(key: K, value: RuleForm[K]) =>
   533→   533→    setForm((f) => ({ ...f, [key]: value }));
   534→   534→
   535→   535→  const onSave = async () => {
   536→   536→    if (isCustom ? !form.description.trim() : !form.value.trim()) return;
   537→   537→    setSaving(true);
   538→   538→    try {
   539→   539→      const payload: RuleInput = {
   540→   540→        kind: form.kind,
   541→   541→        value: isCustom ? "1" : form.value.trim(),
   542→   542→        description: form.description.trim() === "" ? null : form.description.trim(),
   543→   543→        isActive: form.isActive,
   544→   544→      };
   545→   545→      if (editing) {
   546→   546→        await update.mutateAsync({ id: editing.id, patch: payload });
   547→   547→        toast({ title: t("updated") });
   548→   548→      } else {
   549→   549→        await create.mutateAsync(payload);
   550→   550→        toast({ title: t("created") });
   551→   551→      }
   552→   552→      onOpenChange(false);
   553→   553→    } catch (error) {
   554→   554→      const err = error instanceof ApiClientError ? error : null;
   555→   555→      toast({
   556→   556→        title: apiErrorMessage(tRoot, err?.code, error instanceof Error ? error.message : ""),
   557→   557→        variant: "destructive",
   558→   558→      });
   559→   559→    } finally {
   560→   560→      setSaving(false);
   561→   561→    }
   562→   562→  };
   563→   563→
   564→   564→  return (
   565→   565→    <Dialog open={open} onOpenChange={onOpenChange}>
   566→   566→      <DialogContent className="sm:max-w-md">
   567→   567→        <DialogHeader>
   568→   568→          <DialogTitle>{editing ? t("editTitle") : t("createTitle")}</DialogTitle>
   569→   569→          <DialogDescription className="sr-only">
   570→   570→            {editing ? t("editTitle") : t("createTitle")}
   571→   571→          </DialogDescription>
   572→   572→        </DialogHeader>
   573→   573→        <div className="flex flex-col gap-4">
   574→   574→          <div className="space-y-1.5">
   575→   575→            <label className="text-xs font-medium text-muted-foreground" htmlFor="rule-kind">
   576→   576→              {t("kind")}
   577→   577→            </label>
   578→   578→            <Select
   579→   579→              value={form.kind}
   580→   580→              onValueChange={(v) => set("kind", v as RuleForm["kind"])}
   581→   581→            >
   582→   582→              <SelectTrigger id="rule-kind">
   583→   583→                <SelectValue />
   584→   584→              </SelectTrigger>
   585→   585→              <SelectContent>
   586→   586→                {RULE_KIND_LIST.map((k) => (
   587→   587→                  <SelectItem key={k} value={k}>
   588→   588→                    {tRules(`kind.${k}`)}
   589→   589→                  </SelectItem>
   590→   590→                ))}
   591→   591→              </SelectContent>
   592→   592→            </Select>
   593→   593→            <p className="text-[11px] text-muted-foreground">{t(`hint.${form.kind}`)}</p>
   594→   594→          </div>
   595→   595→          {!isCustom && (
   596→   596→            <div className="space-y-1.5">
   597→   597→              <label className="text-xs font-medium text-muted-foreground" htmlFor="rule-value">
   598→   598→                {isCount ? t("valueCount") : t("valueAmount", { currency })}
   599→   599→              </label>
   600→   600→              <Input
   601→   601→                id="rule-value"
   602→   602→                inputMode="decimal"
   603→   603→                value={form.value}
   604→   604→                onChange={(e) => set("value", e.target.value)}
   605→   605→                placeholder={isCount ? "3" : "500"}
   606→   606→              />
   607→   607→            </div>
   608→   608→          )}
   609→   609→          <div className="space-y-1.5">
   610→   610→            <label className="text-xs font-medium text-muted-foreground" htmlFor="rule-desc">
   611→   611→              {t("description")}
   612→   612→            </label>
   613→   613→            <Input
   614→   614→              id="rule-desc"
   615→   615→              value={form.description}
   616→   616→              onChange={(e) => set("description", e.target.value)}
   617→   617→              placeholder={t("descriptionPlaceholder")}
   618→   618→              maxLength={500}
   619→   619→            />
   620→   620→          </div>
   621→   621→          <label className="flex items-center gap-2 text-sm">
   622→   622→            <Checkbox
   623→   623→              id="rule-active"
   624→   624→              checked={form.isActive}
   625→   625→              onCheckedChange={(v) => set("isActive", v === true)}
   626→   626→            />
   627→   627→            {t("active")}
   628→   628→          </label>
   629→   629→          <p className="rounded-md border border-dashed p-2 text-[11px] text-muted-foreground">
   630→   630→            {t("reEvalNote")}
   631→   631→          </p>
   632→   632→        </div>
   633→   633→        <DialogFooter>
   634→   634→          <Button variant="outline" onClick={() => onOpenChange(false)}>
   635→   635→            {t("cancel")}
   636→   636→          </Button>
   637→   637→          <Button
   638→   638→            onClick={() => void onSave()}
   639→   639→            disabled={
   640→   640→              saving || (isCustom ? !form.description.trim() : !form.value.trim())
   641→   641→            }
   642→   642→          >
   643→   643→            {saving ? t("saving") : t("save")}
   644→   644→          </Button>
   645→   645→        </DialogFooter>
   646→   646→      </DialogContent>
   647→   647→    </Dialog>
   648→   648→  );
   649→   649→}
