   468→                    variant="ghost"
   469→                    size="sm"
   470→                    onClick={() => setSelected(selected?.id === a.id ? null : a)}

   491→
   492→      {/* Create / edit dialog */}

   497→        accounts={accounts.data ?? []}
   498→        strategies={strategies.data ?? []}
   499→        onClose={() => {

   501→          setEditing(null);
   502→        }}
   503→      />
   504→    </div>

   507→
   508→function fmtSide(
   509→  v: number | null,
   510→  row: { pct?: boolean; r?: boolean; money?: boolean }
   511→): string {
   512→  if (v == null) return "—";
   513→  if (row.pct) return `${fmtNum(v, 1)} %`;
   514→  if (row.r) return `${fmtNum(v, 2)} R`;
   515→  if (row.money) return fmtNum(v, 0);
   516→  return String(v);
   517→}

   521→function AnnotationDialog({
   522→  open,
   523→  initialDate,
   524→  annotation,
   525→  accounts,
   526→  strategies,
   527→  onClose,
   528→}: {
   529→  open: boolean;
   530→  initialDate: string;
   531→  annotation: Annotation | null;
   532→  accounts: Array<{ id: string; name: string }>;
   533→  strategies: Array<{ id: string; name: string }>;
   534→  onClose: () => void;
   535→}) {
   536→  const t = useTranslations("chartsBuilder.annotations");
   537→  const tCommon = useTranslations("common");
   538→  const tRoot = useTranslations();
   539→  const { toast } = useToast();
   540→  const qc = useQueryClient();
   541→
   542→  const [label, setLabel] = React.useState("");
   543→  const [note, setNote] = React.useState("");
   544→  const [kind, setKind] = React.useState<string>("OTHER");
   545→  const [scope, setScope] = React.useState<string>("GLOBAL");
   546→  const [accountId, setAccountId] = React.useState("");
   547→  const [strategyId, setStrategyId] = React.useState("");
   548→  const [endDate, setEndDate] = React.useState("");
   549→  const [busy, setBusy] = React.useState(false);
   550→
   551→  React.useEffect(() => {
   552→    if (!open) return;
   553→    setLabel(annotation?.label ?? "");
   554→    setNote(annotation?.note ?? "");
   555→    setKind(annotation?.kind ?? "OTHER");
   556→    setScope(annotation?.scope ?? "GLOBAL");
   557→    setAccountId(annotation?.accountId ?? "");
   558→    setStrategyId(annotation?.strategyId ?? "");
   559→    setEndDate(
   560→      annotation?.endDate != null
   561→        ? new Date(annotation.endDate).toISOString().slice(0, 10)
   562→        : ""
   563→    );
   564→  }, [open, annotation]);
   565→
   566→  const submit = async () => {
   567→    if (!label.trim()) return;
   568→    setBusy(true);
   569→    try {
   570→      const body = {
   571→        date: annotation?.date ?? initialDate,
   572→        endDate: endDate ? new Date(`${endDate}T12:00:00.000Z`).toISOString() : null,
   573→        label: label.trim(),
   574→        note: note.trim() || null,
   575→        kind,
   576→        scope,
   577→        accountId: scope === "ACCOUNT" ? accountId || null : null,
   578→        strategyId: scope === "STRATEGY" ? strategyId || null : null,
   579→      };
   580→      if (annotation) {
   581→        await apiFetch(`/api/charts/annotations/${annotation.id}`, {
   582→          method: "PATCH",
   583→          body: JSON.stringify(body),
   584→        });
   585→      } else {
   586→        await apiFetch("/api/charts/annotations", {
   587→          method: "POST",
   588→          body: JSON.stringify(body),
   589→        });
   590→      }
   591→      toast({ title: annotation ? t("updatedToast") : t("createdToast") });
   592→      void qc.invalidateQueries({ queryKey: ["charts", "annotations"] });
   593→      onClose();
   594→    } catch (error) {
   595→      const err = error instanceof ApiClientError ? error : null;
   596→      toast({
   597→        title: tCommon("error"),
   598→        description: apiErrorMessage(
   599→          tRoot,
   600→          err?.code,
   601→          error instanceof Error ? error.message : ""
   602→        ),
   603→        variant: "destructive",
   604→      });
   605→    } finally {
   606→      setBusy(false);
   607→    }
   608→  };
   609→
   610→  const dayValue = new Date(initialDate).toISOString().slice(0, 10);
   611→
   612→  return (
   613→    <Dialog open={open} onOpenChange={(v) => !v && onClose()}>
   614→      <DialogContent className="sm:max-w-md">
   615→        <DialogHeader>
   616→          <DialogTitle>
   617→            {annotation ? t("editTitle") : t("createTitle")}
   618→          </DialogTitle>
   619→          <DialogDescription>
   620→            {annotation
   621→              ? t("editDesc")
   622→              : t("createDesc", { date: dayValue })}
   623→          </DialogDescription>
   624→        </DialogHeader>
   625→        <div className="space-y-3">
   626→          <div className="space-y-1.5">
   627→            <Label htmlFor="ann-label">{t("labelLabel")}</Label>
   628→            <Input
   629→              id="ann-label"
   630→              value={label}
   631→              placeholder={t("labelPlaceholder")}
   632→              onChange={(e) => setLabel(e.target.value)}
   633→            />
   634→          </div>
   635→          <div className="grid grid-cols-2 gap-3">
   636→            <div className="space-y-1.5">
   637→              <Label>{t("kindLabel")}</Label>
   638→              <Select value={kind} onValueChange={setKind}>
   639→                <SelectTrigger>
   640→                  <SelectValue />
   641→                </SelectTrigger>
   642→                <SelectContent>
   643→                  {ANNOTATION_KINDS.map((k) => (
   644→                    <SelectItem key={k} value={k}>
   645→                      {t(`kinds.${k}` as never)}
   646→                    </SelectItem>
   647→                  ))}
   648→                </SelectContent>
   649→              </Select>
   650→            </div>
   651→            <div className="space-y-1.5">
   652→              <Label htmlFor="ann-end">{t("endDateLabel")}</Label>
   653→              <Input
   654→                id="ann-end"
   655→                type="date"
   656→                value={endDate}
   657→                onChange={(e) => setEndDate(e.target.value)}
   658→              />
   659→            </div>
   660→          </div>
   661→          <div className="space-y-1.5">
   662→            <Label>{t("scopeLabel")}</Label>
   663→            <Select value={scope} onValueChange={setScope}>
   664→              <SelectTrigger>
   665→                <SelectValue />
   666→              </SelectTrigger>
   667→              <SelectContent>
   668→                {ANNOTATION_SCOPES.map((s) => (
   669→                  <SelectItem key={s} value={s}>
   670→                    {t(`scopes.${s}` as never)}
   671→                  </SelectItem>
   672→                ))}
   673→              </SelectContent>
   674→            </Select>
   675→          </div>
   676→          {scope === "ACCOUNT" && (
   677→            <div className="space-y-1.5">
   678→              <Label>{t("accountLabel")}</Label>
   679→              <Select value={accountId} onValueChange={setAccountId}>
   680→                <SelectTrigger>
   681→                  <SelectValue placeholder={t("accountPlaceholder")} />
   682→                </SelectTrigger>
   683→                <SelectContent>
   684→                  {accounts.map((a) => (
   685→                    <SelectItem key={a.id} value={a.id}>
   686→                      {a.name}
   687→                    </SelectItem>
   688→                  ))}
   689→                </SelectContent>
   690→              </Select>
   691→            </div>
   692→          )}
   693→          {scope === "STRATEGY" && (
   694→            <div className="space-y-1.5">
   695→              <Label>{t("strategyLabel")}</Label>
   696→              <Select value={strategyId} onValueChange={setStrategyId}>
   697→                <SelectTrigger>
   698→                  <SelectValue placeholder={t("strategyPlaceholder")} />
   699→                </SelectTrigger>
   700→                <SelectContent>
   701→                  {strategies.map((s) => (
   702→                    <SelectItem key={s.id} value={s.id}>
   703→                      {s.name}
   704→                    </SelectItem>
   705→                  ))}
   706→                </SelectContent>
   707→              </Select>
   708→            </div>
   709→          )}
   710→          <div className="space-y-1.5">
   711→            <Label htmlFor="ann-note">{t("noteLabel")}</Label>
   712→            <Textarea
   713→              id="ann-note"
   714→              value={note}
   715→              className="h-20"
   716→              placeholder={t("notePlaceholder")}
   717→              onChange={(e) => setNote(e.target.value)}
   718→            />
   719→          </div>
   720→        </div>
   721→        <DialogFooter>
   722→          <Button onClick={() => void submit()} disabled={busy || !label.trim()}>
   723→            {annotation ? tCommon("save") : t("create")}
   724→          </Button>
   725→        </DialogFooter>
   726→      </DialogContent>
   727→    </Dialog>
   728→  );
   729→}
