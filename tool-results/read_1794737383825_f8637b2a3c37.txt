    90→function nowIso(): string {
    91→  return (

    98→  return {
    99→    ...tpl,

   102→    session: tpl.session as TradeTemplateData["session"],
   103→    orderType: tpl.orderType as TradeTemplateData["orderType"],
   104→  };
   105→}

   160→  const { toast } = useToast();

   162→
   163→  const [form, setForm] = React.useState<QuickFormState>(emptyForm);
   164→  const [templateId, setTemplateId] = React.useState("");
   165→  const [submitting, setSubmitting] = React.useState(false);
   166→  const [saved, setSaved] = React.useState<{ symbol: string; closed: boolean } | null>(
   167→    null
   168→  );
   169→
   170→  const entryPriceRef = React.useRef<HTMLInputElement>(null);
   171→  const baseInitRef = React.useRef(false);
   172→  const riskInitRef = React.useRef(false);
   173→  const autoTemplateRef = React.useRef(false);
   174→  const directionTouchedRef = React.useRef(false);
   175→  const mutationIdRef = React.useRef<string>(
   176→    typeof crypto !== "undefined" && "randomUUID" in crypto
   177→      ? crypto.randomUUID()
   178→      : `qt-${Date.now()}-${Math.random().toString(36).slice(2)}`
   179→  );
   180→
   181→  const accountsQuery = useAccounts();
   182→  const accounts = accountsQuery.data;
   183→  const { data: instruments } = useInstruments();
   184→  const { data: strategies } = useStrategies();
   185→  const { data: templates } = useTemplates();
   186→
   187→  const createTrade = useCreateTrade();
   188→  const recordTemplateUse = useRecordTemplateUse();
   189→
   190→  // ── Defaults on mount: remembered account / instrument / timeframe ──
   191→  React.useEffect(() => {
   192→    if (baseInitRef.current || !accounts) return;
   193→    baseInitRef.current = true;
   194→    const prefs = session?.profile.preferences ?? {};
   195→    const valid = (id: string | null | undefined): string | null =>
   196→      id && accounts.some((a) => a.id === id) ? id : null;
   197→    const accountId =
   198→      valid(prefs.lastAccountId) ??
   199→      valid(session?.profile.defaultAccountId) ??
   200→      accounts.find((a) => a.isActive)?.id ??
   201→      "";
   202→    setForm((f) => ({
   203→      ...f,
   204→      accountId: f.accountId || accountId,
   205→      instrumentId: f.instrumentId || prefs.lastInstrumentId || "",
   206→      timeframe: f.timeframe || prefs.lastTimeframe || "",
   207→    }));
   208→  }, [accounts, session]);
   209→
   210→  // ── Remembered lot size for the pre-selected instrument ──
   211→  React.useEffect(() => {
   212→    if (riskInitRef.current || !instruments || !form.instrumentId) return;
   213→    riskInitRef.current = true;
   214→    const symbol = instruments.find((i) => i.id === form.instrumentId)?.symbol;
   215→    const remembered = symbol
   216→      ? session?.profile.preferences?.rememberedLots?.[symbol]
   217→      : undefined;
   218→    if (remembered != null && remembered !== "") {
   219→      setForm((f) => ({ ...f, riskMode: "lots", lots: remembered }));
   220→    }
   221→  }, [instruments, form.instrumentId, session]);
   222→
   223→  // ── Derived data ──
   224→  const spec = React.useMemo(
   225→    () => (instruments ?? []).find((i) => i.id === form.instrumentId) ?? null,
   226→    [instruments, form.instrumentId]
   227→  );
   228→  const account = React.useMemo(
   229→    () => (accounts ?? []).find((a) => a.id === form.accountId) ?? null,
   230→    [accounts, form.accountId]
   231→  );
   232→  const accountCurrency = account?.currency ?? "";
   233→  const symbol = spec?.symbol ?? "";
   234→
   235→  const entry = form.entryPrice.trim();
   236→  const stop = form.stopPips.trim();
   237→
   238→  /** SL/TP prices + position size derived from the current inputs. */
   239→  const derived = React.useMemo(() => {
   240→    if (!spec || !entry || !stop) return null;
   241→    const s = toSpec(spec);
   242→    const stopLoss = stopPriceFromPips(form.direction, entry, stop, s);
   243→    const takeProfit = form.tpPips.trim()
   244→      ? takeProfitPriceFromPips(form.direction, entry, form.tpPips.trim(), s)
   245→      : null;
   246→
   247→    let sizing: PositionSizeResult | null = null;
   248→    if (form.riskMode !== "lots" && account) {
   249→      sizing = calcPositionSize({
   250→        accountBalance: account.initialBalance,
   251→        riskPercent: form.riskMode === "percent" ? form.riskPercent.trim() || null : null,
   252→        riskAmount: form.riskMode === "amount" ? form.riskAmount.trim() || null : null,
   253→        entry,
   254→        stopLoss,
   255→        instrument: s,
   256→        direction: form.direction,
   257→      });
   258→    }
   259→    const lots =
   260→      form.riskMode === "lots"
   261→        ? form.lots.trim()
   262→        : sizing?.valid
   263→          ? sizing.lots.toString()
   264→          : "";
   265→    return { spec: s, stopLoss, takeProfit, sizing, lots };
   266→  }, [spec, entry, stop, form.tpPips, form.direction, form.riskMode, form.riskPercent, form.riskAmount, form.lots, account]);
   267→
   268→  const effectiveLots = derived?.lots ?? "";
   269→  const sizingError =
   270→    derived?.sizing && !derived.sizing.valid
   271→      ? sizingErrorMessage(
   272→          tRoot,
   273→          derived.sizing.reason,
   274→          derived.sizing.minLots.toString()
   275→        )
   276→      : null;
   277→
   278→  const previewReady = Boolean(spec && entry && stop && effectiveLots);
   279→  const preview = React.useMemo(() => {
   280→    if (!previewReady || !derived) return null;
   281→    try {
   282→      return previewTrade({
   283→        instrument: derived.spec,
   284→        direction: form.direction,
   285→        lots: effectiveLots,
   286→        entry,
   287→        exit:
   288→          (form.closed && form.exitPrice.trim()) ||
   289→          derived.takeProfit ||
   290→          entry,
   291→        stopLoss: derived.stopLoss,
   292→        takeProfit: derived.takeProfit,
   293→        accountCurrency: accountCurrency || null,
   294→        accountBalance: account?.initialBalance ?? null,
   295→      });
   296→    } catch {
   297→      return null;
   298→    }
   299→  }, [previewReady, derived, form.direction, effectiveLots, entry, form.closed, form.exitPrice, accountCurrency, account]);
   300→
   301→  // ── Handlers ──
   302→  const set = <K extends keyof QuickFormState>(key: K, value: QuickFormState[K]) =>
   303→    setForm((f) => ({ ...f, [key]: value }));
   304→
   305→  const setDirection = (value: string) => {
   306→    if (value !== "LONG" && value !== "SHORT") return;
   307→    directionTouchedRef.current = true;
   308→    set("direction", value);
   309→  };
   310→
   311→  const riskValue =
   312→    form.riskMode === "percent"
   313→      ? form.riskPercent
   314→      : form.riskMode === "amount"
   315→        ? form.riskAmount
   316→        : form.lots;
   317→  const setRiskValue = (value: string) => {
   318→    if (form.riskMode === "percent") set("riskPercent", value);
   319→    else if (form.riskMode === "amount") set("riskAmount", value);
   320→    else set("lots", value);
   321→  };
   322→
   323→  function applyTemplate(tpl: TemplateDTO) {
   324→    setTemplateId(tpl.id);
   325→    const strategyRisk =
   326→      strategies?.find((s) => s.id === tpl.strategyId)?.defaultRiskPercent ?? null;
   327→    const prefill = applyTradeTemplate(asTemplateData(tpl), {
   328→      instruments: (instruments ?? []).map(toSpec),
   329→      entryPrice: entry || null,
   330→      direction: directionTouchedRef.current ? form.direction : null,
   331→      strategyDefaultRiskPercent: strategyRisk,
   332→    });
   333→    setForm((f) => ({
   334→      ...f,
   335→      strategyId: prefill.strategyId ?? f.strategyId,
   336→      instrumentId: prefill.instrumentId ?? f.instrumentId,
   337→      direction: (prefill.direction ?? f.direction) as Direction,
   338→      timeframe: prefill.timeframe ?? f.timeframe,
   339→      stopPips: prefill.stopLossPips ?? f.stopPips,
   340→      tpPips: prefill.takeProfitPips ?? f.tpPips,
   341→      notes: prefill.notes ?? f.notes,
   342→      tagIds:
   343→        prefill.tagIds.length > 0
   344→          ? Array.from(new Set([...f.tagIds, ...prefill.tagIds]))
   345→          : f.tagIds,
   346→      riskMode: prefill.lots != null ? "lots" : f.riskMode,
   347→      riskPercent: prefill.riskPercent ?? f.riskPercent,
   348→      riskAmount: prefill.riskAmount ?? f.riskAmount,
   349→      lots: prefill.lots ?? f.lots,
   350→    }));
   351→    toast({ title: t("templateApplied", { name: tpl.name }) });
   352→    recordTemplateUse.mutate(tpl.id);
   353→  }
   354→
   355→  // ── Auto-apply ?template=<id> from the hash (templates "use" action) ──
   356→  React.useEffect(() => {
   357→    if (autoTemplateRef.current || !templates) return;
   358→    autoTemplateRef.current = true;
   359→    const hash = typeof window === "undefined" ? "" : window.location.hash;
   360→    const qIndex = hash.indexOf("?");
   361→    if (qIndex === -1) return;
   362→    const id = new URLSearchParams(hash.slice(qIndex + 1)).get("template");
   363→    if (!id) return;
   364→    const tpl = templates.find((x) => x.id === id);
   365→    if (tpl) applyTemplate(tpl);
   366→  }, [templates]);
   367→
   368→  async function onSubmit(event: React.FormEvent) {
   369→    event.preventDefault();
   370→    if (!form.instrumentId || !entry || !effectiveLots) {
   371→      toast({
   372→        title: t("validationTitle"),
   373→        description: t("validationDesc"),
   374→        variant: "destructive",
   375→      });
   376→      return;
   377→    }
   378→    const raw: TradeInput = {
   379→      accountId: form.accountId,
   380→      strategyId: form.strategyId || null,
   381→      instrumentId: form.instrumentId,
   382→      direction: form.direction,
   383→      orderType: "MARKET",
   384→      status: form.closed ? "CLOSED" : "OPEN",
   385→      timeframe: (form.timeframe || null) as Timeframe | null,
   386→      session: null,
   387→      setupType: null,
   388→      marketCondition: null,
   389→      entryTime: form.entryTime || nowIso(),
   390→      exitTime: form.closed ? nowIso() : null,
   391→      entryPrice: entry,
   392→      exitPrice:
   393→        form.closed && form.exitPrice.trim() ? form.exitPrice.trim() : null,
   394→      stopLoss: derived?.stopLoss ?? null,
   395→      takeProfit: derived?.takeProfit ?? null,
   396→      mfePrice: null,
   397→      maePrice: null,
   398→      lots: effectiveLots,
   399→      riskPercent:
   400→        form.riskMode === "percent" && form.riskPercent.trim()
   401→          ? form.riskPercent.trim()
   402→          : null,
   403→      riskAmount:
   404→        form.riskMode === "amount" && form.riskAmount.trim()
   405→          ? form.riskAmount.trim()
   406→          : null,
   407→      plannedRiskPercent: null,
   408→      quoteToAccountRate: null,
   409→      commission: "0",
   410→      swap: "0",
   411→      fees: "0",
   412→      pnlSource: "CALCULATED",
   413→      manualNetPnl: null,
   414→      exitReason:
   415→        form.closed && form.exitReason
   416→          ? (form.exitReason as ExitReason)
   417→          : null,
   418→      emotionBefore: null,
   419→      emotionDuring: null,
   420→      emotionAfter: null,
   421→      confidence: null,
   422→      setupQualityRating: null,
   423→      executionRating: null,
   424→      followedPlan: null,
   425→      tagIds: form.tagIds,
   426→      checklistResults: null,
   427→      preTradeAnalysis: "",
   428→      postTradeReview: "",
   429→      notes: form.notes.trim() || null,
   430→      tradingViewUrl: null,
   431→      reviewStatus: "NOT_REVIEWED",
   432→      isDraft: false,
   433→      clientMutationId: mutationIdRef.current,
   434→    };
   435→    const parsed = tradeInputSchema.safeParse(raw);
   436→    if (!parsed.success) {
   437→      toast({
   438→        title: t("validationTitle"),
   439→        description: t("validationDesc"),
   440→        variant: "destructive",
   441→      });
   442→      return;
   443→    }
   444→
   445→    setSubmitting(true);
   446→    try {
   447→      const result: MutationResult = await createTrade.mutateAsync(parsed.data as TradeInput);
   448→      toast({
   449→        title: result.deduplicated
   450→          ? t("savedToast")
   451→          : form.closed
   452→            ? t("savedClosedToast")
   453→            : t("savedToast"),
   454→      });
   455→
   456→      // Persist quick-entry defaults (fire and forget).
   457→      const prefs = session?.profile.preferences ?? {};
   458→      void apiFetch("/api/profile", {
   459→        method: "PATCH",
   460→        body: JSON.stringify({
   461→          preferences: {
   462→            lastAccountId: form.accountId || null,
   463→            lastInstrumentId: form.instrumentId || null,
   464→            lastTimeframe: form.timeframe || null,
   465→            rememberedLots: {
   466→              ...(prefs.rememberedLots ?? {}),
   467→              ...(symbol ? { [symbol]: effectiveLots } : {}),
   468→            },
   469→          },
   470→        }),
   471→      }).catch(() => {});
   472→
   473→      setSaved({ symbol: symbol || form.instrumentId, closed: form.closed });
   474→    } catch (error) {
   475→      const err = error instanceof ApiClientError ? error : null;
   476→      toast({
   477→        title: apiErrorMessage(
   478→          tRoot,
   479→          err?.code,
   480→          error instanceof Error ? error.message : ""
   481→        ),
   482→        variant: "destructive",
   483→      });
   484→    } finally {
   485→      setSubmitting(false);
   486→    }
   487→  }
   488→
   489→  const onAnother = () => {
   490→    setSaved(null);
   491→    mutationIdRef.current = `qt-${Date.now()}-${Math.random().toString(36).slice(2)}`;
   492→    setForm((f) => ({
   493→      ...f,
   494→      entryTime: nowIso(),
   495→      entryPrice: "",
   496→      closed: false,
   497→      exitPrice: "",
   498→      exitReason: "",
   499→      notes: "",
   500→    }));
   501→    requestAnimationFrame(() => entryPriceRef.current?.focus());
   502→  };
   503→
   504→  // ── Gates ──
   505→  if (accountsQuery.isError) {
   506→    return (
   507→      <Card>
   508→        <CardContent className="flex flex-col items-center gap-3 py-10 text-center">
   509→          <p className="text-sm text-muted-foreground">{t("loadError")}</p>
   510→          <Button variant="outline" onClick={() => void accountsQuery.refetch()}>
   511→            {tc("retry")}
   512→          </Button>
   513→        </CardContent>
   514→      </Card>
   515→    );
   516→  }
   517→  if (accounts && accounts.filter((a) => a.isActive).length === 0) {
   518→    return (
   519→      <Card>
   520→        <CardContent className="py-10 text-center">
   521→          <p className="text-sm text-muted-foreground">{t("accountMissing")}</p>
   522→        </CardContent>
   523→      </Card>
   524→    );
   525→  }
   526→
   527→  const favoriteIds =
   528→    session?.profile.preferences?.favoriteInstrumentIds ?? [];
   529→  const slPriceText =
   530→    derived && spec && stop ? fmtPrice(derived.stopLoss, spec.digits) : null;
   531→  const tpPriceText =
   532→    derived && spec && form.tpPips.trim() && derived.takeProfit
   533→      ? fmtPrice(derived.takeProfit, spec.digits)
   534→      : null;
   535→
   536→  return (
   537→    <div className="mx-auto w-full max-w-2xl space-y-4">
   538→      <div className="space-y-1 text-center sm:text-left">
   539→        <h2 className="text-xl font-bold tracking-tight sm:text-2xl">{t("title")}</h2>
   540→        <p className="text-sm text-muted-foreground">{t("subtitle")}</p>
   541→      </div>
   542→
   543→      <Card>
   544→        <CardHeader className="pb-3">
   545→          <CardTitle className="text-base sr-only">{t("title")}</CardTitle>
   546→          <CardDescription className="sr-only">{t("subtitle")}</CardDescription>
   547→          <Field label={t("template")} htmlFor="qt-template">
   548→            <Select
   549→              value={templateId}
   550→              onValueChange={(id) => {
   551→                if (!id) return; // spurious Radix empty sync — ignore
   552→                const tpl = templates?.find((x) => x.id === id);
   553→                if (tpl) applyTemplate(tpl);
   554→              }}
   555→            >
   556→              <SelectTrigger id="qt-template" className="w-full">
   557→                <SelectValue placeholder={t("templateNone")} />
   558→              </SelectTrigger>
   559→              <SelectContent className="max-h-72">
   560→                {(templates ?? []).map((tpl) => (
   561→                  <SelectItem key={tpl.id} value={tpl.id}>
   562→                    {tpl.name}
   563→                  </SelectItem>
   564→                ))}
   565→              </SelectContent>
   566→            </Select>
   567→          </Field>
   568→        </CardHeader>
   569→        <CardContent className="space-y-4">
   570→          {saved && (
   571→            <div className="flex flex-col items-center gap-3 rounded-lg border border-emerald-500/40 bg-emerald-500/10 p-4 text-center">
   572→              <CheckCircle2
   573→                className="h-8 w-8 text-emerald-500 dark:text-emerald-400"
   574→                aria-hidden="true"
   575→              />
   576→              <div>
   577→                <p className="text-sm font-semibold">
   578→                  {saved.closed ? t("savedClosedToast") : t("savedToast")}
   579→                </p>
   580→                {saved.symbol && (
   581→                  <p className="text-xs tabular-nums text-muted-foreground">
   582→                    {saved.symbol} · {fmtNum(effectiveLots, 2, 2)} {tc("lots")}
   583→                  </p>
