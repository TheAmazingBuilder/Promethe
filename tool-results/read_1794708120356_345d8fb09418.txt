   123→   123→/** Recharts tooltip: date + equity (+ drawdown below peak, as money). */
   124→   124→function EquityTooltipContent({
   125→   125→  active,
   126→   126→  payload,
   127→   127→  currency,
   128→   128→  timezone,
   129→   129→}: {
   130→   130→  active?: boolean;
   131→   131→  payload?: Array<{ payload?: EquityDatum }>;
   132→   132→  currency: string;
   133→   133→  timezone: string;
   134→   134→}) {
   135→   135→  if (!active) return null;
   136→   136→  const point = payload?.[0]?.payload;
   137→   137→  if (!point) return null;
   138→   138→  return (
   139→   139→    <div className="rounded-lg border bg-background px-3 py-2 text-xs shadow-md">
   140→   140→      <p className="font-medium">{fmtDateInTz(point.date, timezone)}</p>
   141→   141→      <p className="mt-1 tabular-nums">{fmtMoney(point.equity, currency)}</p>
   142→   142→      {point.drawdown > 0 ? (
   143→   143→        <p className={`tabular-nums ${ROSE}`}>
   144→   144→          −{fmtMoney(point.drawdown, currency)}
   145→   145→        </p>
   146→   146→      ) : null}
   147→   147→    </div>
   148→   148→  );
   149→   149→}

   189→   189→/** Risk cell for an open position: amount, else %, else em dash. */
   190→   190→function openPositionRisk(p: OpenPositionDTO): string {
   191→   191→  if (p.riskAmount) return fmtMoney(p.riskAmount, p.pnlCurrency);
   192→   192→  if (p.riskPercent) return fmtPct(p.riskPercent);
   193→   193→  return "—";
   194→   194→}
