   470→                        return next;
   471→                      })
   472→                    }
   473→                  >
   474→                    {visible.has(a.id) ? (
   475→                      <Eye aria-hidden="true" />
   476→                    ) : (
   477→                      <EyeOff aria-hidden="true" />
   478→                    )}
   479→                  </Button>
   480→                  <Button
   481→                    variant="ghost"
   482→                    size="sm"

   485→                    {t("stats")}

   498→                </div>
   499→              ))}
   500→            </div>
   501→          )}

   504→
   505→      {/* Create / edit dialog */}

   510→        accounts={accounts.data ?? []}
   511→        strategies={strategies.data ?? []}
   512→        onClose={() => {
   513→          setCreating(null);
   514→          setEditing(null);
   515→        }}
   516→      />
   517→    </div>

   520→
   521→function fmtSide(
   522→  v: number | null,
   523→  row: { pct?: boolean; r?: boolean; money?: boolean },
   524→  fmt: FormatApi,
   525→  currency: string
   526→): string {
   527→  if (v == null) return "—";
   528→  if (row.pct) return fmt.pct(v, 1);
   529→  if (row.r) return `${fmtNum(v, 2)} R`;
   530→  if (row.money) return fmt.money(v, currency, 0);
   531→  return String(v);
   532→}

   570→    setKind(annotation?.kind ?? "OTHER");
   571→    setScope(annotation?.scope ?? "GLOBAL");
   572→    setAccountId(annotation?.accountId ?? "");
   573→    setStrategyId(annotation?.strategyId ?? "");
   574→    setEndDate(
   575→      annotation?.endDate != null
   576→        ? new Date(annotation.endDate).toISOString().slice(0, 10)
   577→        : ""
   578→    );
   579→  }, [open, annotation]);
   580→
   581→  const submit = async () => {
   582→    if (!label.trim()) return;
   583→    setBusy(true);
   584→    try {
   585→      const body = {
   586→        date: annotation?.date ?? initialDate,
   587→        endDate: endDate ? new Date(`${endDate}T12:00:00.000Z`).toISOString() : null,
   588→        label: label.trim(),
   589→        note: note.trim() || null,
   590→        kind,
   591→        scope,
   592→        accountId: scope === "ACCOUNT" ? accountId || null : null,
   593→        strategyId: scope === "STRATEGY" ? strategyId || null : null,
   594→      };
   595→      if (annotation) {
   596→        await apiFetch(`/api/charts/annotations/${annotation.id}`, {
   597→          method: "PATCH",
   598→          body: JSON.stringify(body),
   599→        });
   600→      } else {
   601→        await apiFetch("/api/charts/annotations", {
   602→          method: "POST",
   603→          body: JSON.stringify(body),
   604→        });
   605→      }
   606→      toast({ title: annotation ? t("updatedToast") : t("createdToast") });
   607→      void qc.invalidateQueries({ queryKey: ["charts", "annotations"] });
   608→      onClose();
   609→    } catch (error) {
   610→      const err = error instanceof ApiClientError ? error : null;
   611→      toast({
   612→        title: tCommon("error"),
   613→        description: apiErrorMessage(
   614→          tRoot,
   615→          err?.code,
   616→          error instanceof Error ? error.message : ""
   617→        ),
   618→        variant: "destructive",
   619→      });
   620→    } finally {
   621→      setBusy(false);
   622→    }
   623→  };
   624→
   625→  const dayValue = new Date(initialDate).toISOString().slice(0, 10);
   626→
   627→  return (
   628→    <Dialog open={open} onOpenChange={(v) => !v && onClose()}>
   629→      <DialogContent className="sm:max-w-md">
   630→        <DialogHeader>
   631→          <DialogTitle>
   632→            {annotation ? t("editTitle") : t("createTitle")}
   633→          </DialogTitle>
   634→          <DialogDescription>
   635→            {annotation
   636→              ? t("editDesc")
   637→              : t("createDesc", { date: dayValue })}
   638→          </DialogDescription>
   639→        </DialogHeader>
   640→        <div className="space-y-3">
   641→          <div className="space-y-1.5">
   642→            <Label htmlFor="ann-label">{t("labelLabel")}</Label>
   643→            <Input
   644→              id="ann-label"
   645→              value={label}
   646→              placeholder={t("labelPlaceholder")}
   647→              onChange={(e) => setLabel(e.target.value)}
   648→            />
   649→          </div>
   650→          <div className="grid grid-cols-2 gap-3">
   651→            <div className="space-y-1.5">
   652→              <Label>{t("kindLabel")}</Label>
   653→              <Select value={kind} onValueChange={setKind}>
   654→                <SelectTrigger>
   655→                  <SelectValue />
   656→                </SelectTrigger>
   657→                <SelectContent>
   658→                  {ANNOTATION_KINDS.map((k) => (
   659→                    <SelectItem key={k} value={k}>
   660→                      {t(`kinds.${k}` as never)}
   661→                    </SelectItem>
   662→                  ))}
   663→                </SelectContent>
   664→              </Select>
   665→            </div>
   666→            <div className="space-y-1.5">
   667→              <Label htmlFor="ann-end">{t("endDateLabel")}</Label>
   668→              <Input
   669→                id="ann-end"
   670→                type="date"
   671→                value={endDate}
   672→                onChange={(e) => setEndDate(e.target.value)}
   673→              />
   674→            </div>
   675→          </div>
   676→          <div className="space-y-1.5">
   677→            <Label>{t("scopeLabel")}</Label>
   678→            <Select value={scope} onValueChange={setScope}>
   679→              <SelectTrigger>
   680→                <SelectValue />
   681→              </SelectTrigger>
   682→              <SelectContent>
   683→                {ANNOTATION_SCOPES.map((s) => (
   684→                  <SelectItem key={s} value={s}>
   685→                    {t(`scopes.${s}` as never)}
   686→                  </SelectItem>
   687→                ))}
   688→              </SelectContent>
   689→            </Select>
   690→          </div>
   691→          {scope === "ACCOUNT" && (
   692→            <div className="space-y-1.5">
   693→              <Label>{t("accountLabel")}</Label>
   694→              <Select value={accountId} onValueChange={setAccountId}>
   695→                <SelectTrigger>
   696→                  <SelectValue placeholder={t("accountPlaceholder")} />
   697→                </SelectTrigger>
   698→                <SelectContent>
   699→                  {accounts.map((a) => (
   700→                    <SelectItem key={a.id} value={a.id}>
   701→                      {a.name}
   702→                    </SelectItem>
   703→                  ))}
   704→                </SelectContent>
   705→              </Select>
   706→            </div>
   707→          )}
   708→          {scope === "STRATEGY" && (
   709→            <div className="space-y-1.5">
   710→              <Label>{t("strategyLabel")}</Label>
   711→              <Select value={strategyId} onValueChange={setStrategyId}>
   712→                <SelectTrigger>
   713→                  <SelectValue placeholder={t("strategyPlaceholder")} />
   714→                </SelectTrigger>
   715→                <SelectContent>
   716→                  {strategies.map((s) => (
   717→                    <SelectItem key={s.id} value={s.id}>
   718→                      {s.name}
   719→                    </SelectItem>
   720→                  ))}
   721→                </SelectContent>
   722→              </Select>
   723→            </div>
   724→          )}
   725→          <div className="space-y-1.5">
   726→            <Label htmlFor="ann-note">{t("noteLabel")}</Label>
   727→            <Textarea
   728→              id="ann-note"
   729→              value={note}
   730→              className="h-20"
   731→              placeholder={t("notePlaceholder")}
   732→              onChange={(e) => setNote(e.target.value)}
   733→            />
   734→          </div>
   735→        </div>
   736→        <DialogFooter>
   737→          <Button onClick={() => void submit()} disabled={busy || !label.trim()}>
   738→            {annotation ? tCommon("save") : t("create")}
   739→          </Button>
   740→        </DialogFooter>
   741→      </DialogContent>
   742→    </Dialog>
   743→  );
   744→}
