   149→/** All widget layouts keyed by widget id (the undo snapshot shape). */

   152→function snapshotLayouts(widgets: DashboardWidgetDTO[]): LayoutMap {
   153→  const out: LayoutMap = {};
   154→  for (const w of widgets) out[w.id] = w.layouts;
   155→  return out;
   156→}

   158→function layoutsEqual(a?: WidgetLayouts, b?: WidgetLayouts): boolean {
   159→  for (const bp of ["lg", "md", "sm"] as const) {
   160→    const x = a?.[bp];
   161→    const y = b?.[bp];
   162→    if (x == null && y == null) continue;
   163→    if (x == null || y == null) return false;
   164→    if (x.x !== y.x || x.y !== y.y || x.w !== y.w || x.h !== y.h) return false;
   165→  }
   166→  return true;
   167→}

   169→function pickItem(l: LayoutItem): LayoutItem {
   170→  return {
   171→    i: l.i,
   172→    x: l.x,
   173→    y: l.y,
   174→    w: l.w,
   175→    h: l.h,
   176→    minW: l.minW,
   177→    maxW: l.maxW,
   178→    minH: l.minH,
   179→    maxH: l.maxH,
   180→  };
   181→}

   183→/** Builds the react-grid-layout layouts (lg/md/sm) from the widget DTOs. */
   184→function buildRglLayouts(
   185→  widgets: DashboardWidgetDTO[]
   186→): ResponsiveLayouts<string> {
   187→  const make = (bp: "lg" | "md" | "sm"): LayoutItem[] =>
   188→    widgets.map((w) => {
   189→      const def = getWidgetDefinition(w.type);
   190→      const l =
   191→        w.layouts[bp] ??
   192→        w.layouts.lg ?? { x: 0, y: 0, w: def?.defaultLayout.w ?? 4, h: def?.defaultLayout.h ?? 3 };
   193→      const cw = Math.max(1, Math.min(l.w, GRID_COLS));
   194→      return {
   195→        i: w.id,
   196→        x: Math.max(0, Math.min(l.x, GRID_COLS - cw)),
   197→        y: Math.max(0, l.y),
   198→        w: cw,
   199→        h: Math.max(1, l.h),
   200→        minW: def?.minSize.w,
   201→        maxW: def?.maxSize.w,
   202→        minH: def?.minSize.h,
   203→        maxH: def?.maxSize.h,
   204→      };
   205→    });
   206→  return { lg: make("lg"), md: make("md"), sm: make("sm") };
   207→}

   209→/** Chart body height derived from the grid row height (40px) minus the header. */
   210→function chartHeightPx(widget: DashboardWidgetDTO): number {
   211→  const h = widget.layouts.lg?.h ?? 4;
   212→  return Math.max(120, h * 40 - 110);
   213→}

   319→export function DashboardsView() {
   320→  const { session } = useSession();
   321→  const { locale } = useI18n();
   322→  const t = useTranslations("dashboards");
   323→  const tDash = useTranslations("dashboard");
   324→  const tAnalytics = useTranslations("analytics");
   325→  const tRoot = useTranslations();
   326→  const { toast } = useToast();
   327→  const isMobile = useIsMobile();
   328→
   329→  // ── URL state (p = preset, a = account, d = dashboard) ──
   330→  const [preset, setPreset] = React.useState<PeriodPreset>("last30");
   331→  const [accountFilter, setAccountFilter] = React.useState<string>("all");
   332→  const [urlDashboardId, setUrlDashboardId] = React.useState<string | null>(null);
   333→  const [hydrated, setHydrated] = React.useState(false);
   334→  const [storedId, setStoredId] = React.useState<string | null>(null);
   335→
   336→  React.useEffect(() => {
   337→    const read = () => {
   338→      const params = readHashQuery();
   339→      const p = params.get("p");
   340→      if (p && isPeriodPreset(p)) setPreset(p);
   341→      const a = params.get("a");
   342→      setAccountFilter(a ?? "all");
   343→      setUrlDashboardId(params.get("d"));
   344→    };
   345→    read();
   346→    setHydrated(true);
   347→    setStoredId(window.localStorage.getItem(LAST_DASHBOARD_KEY));
   348→    window.addEventListener("hashchange", read);
   349→    return () => window.removeEventListener("hashchange", read);
   350→  }, []);
   351→
   352→  // ── Data ──
   353→  const {
   354→    data: dashboardsData,
   355→    isLoading: dashboardsLoading,
   356→    isError: dashboardsError,
   357→    refetch: refetchDashboards,
   358→  } = useDashboards();
   359→  const { data: accountsData } = useAccounts();
   360→  const accounts = accountsData ?? [];
   361→  const dashboards = dashboardsData?.dashboards ?? [];
   362→  const requirements = dashboardsData?.requirements ?? null;
   363→
   364→  const active = React.useMemo<DashboardDTO | null>(() => {
   365→    if (dashboards.length === 0) return null;
   366→    const wanted = urlDashboardId ?? storedId;
   367→    return (
   368→      dashboards.find((d) => d.id === wanted) ??
   369→      dashboards.find((d) => d.isDefault) ??
   370→      dashboards[0]!
   371→    );
   372→  }, [dashboards, urlDashboardId, storedId]);
   373→
   374→  const effectiveAccountId =
   375→    accountFilter !== "all" && accounts.some((a) => a.id === accountFilter)
   376→      ? accountFilter
   377→      : null;
   378→
   379→  const {
   380→    data: dashboardData,
   381→    isPending: dataPending,
   382→    isError: dataError,
   383→    refetch: refetchData,
   384→  } = useDashboardData(active?.id ?? null, preset, effectiveAccountId);
   385→
   386→  // Keep the URL query + last-opened dashboard in sync without navigating.
   387→  React.useEffect(() => {
   388→    if (!hydrated) return;
   389→    const sp = new URLSearchParams();
   390→    sp.set("p", preset);
   391→    if (effectiveAccountId) sp.set("a", effectiveAccountId);
   392→    if (active?.id) sp.set("d", active.id);
   393→    const next = `#/?${sp.toString()}`;
   394→    if (window.location.hash !== next) {
   395→      window.history.replaceState(null, "", next);
   396→    }
   397→  }, [hydrated, preset, effectiveAccountId, active?.id]);
   398→
   399→  React.useEffect(() => {
   400→    if (active?.id) {
   401→      window.localStorage.setItem(LAST_DASHBOARD_KEY, active.id);
   402→    }
   403→  }, [active?.id]);
   404→
   405→  // ── Edit mode, announcements ──
   406→  const [editMode, setEditMode] = React.useState(false);
   407→  const [catalogOpen, setCatalogOpen] = React.useState(false);
   408→  const [announcement, setAnnouncement] = React.useState("");
   409→  const announce = React.useCallback((message: string) => {
   410→    setAnnouncement(message);
   411→  }, []);
   412→
   413→  const enterEdit = () => {
   414→    setEditMode(true);
   415→    announce(t("edit.editModeOn"));
   416→  };
   417→  const exitEdit = () => {
   418→    setEditMode(false);
   419→    announce(t("edit.editModeOff"));
   420→  };
   421→
   422→  // Command palette hook: "Customize dashboard".
   423→  React.useEffect(() => {
   424→    const onCustomize = () => {
   425→      setEditMode(true);
   426→      setAnnouncement(t("edit.editModeOn"));
   427→    };
   428→    window.addEventListener("tj:customize-dashboard", onCustomize);
   429→    return () =>
   430→      window.removeEventListener("tj:customize-dashboard", onCustomize);
   431→  }, [t]);
   432→
   433→  // ── Mutations ──
   434→  const createDashboard = useCreateDashboard();
   435→  const patchDashboard = usePatchDashboard();
   436→  const deleteDashboard = useDeleteDashboard();
   437→  const reorderDashboards = useReorderDashboards();
   438→  const addWidget = useAddWidget();
   439→  const deleteWidget = useDeleteWidget();
   440→  const patchWidgetNormal = usePatchWidget();
   441→  const patchWidgetOptimistic = usePatchWidget({ optimistic: true });
   442→
   443→  // ── Grid layouts state (desktop/tablet) ──
   444→  const [gridState, setGridState] = React.useState<ResponsiveLayouts<string> | null>(null);
   445→  const gridsRef = React.useRef<ResponsiveLayouts<string> | null>(null);
   446→  const savedLayoutsRef = React.useRef<LayoutMap>({});
   447→  const undoSnapshotRef = React.useRef<LayoutMap | null>(null);
   448→  const [canUndo, setCanUndo] = React.useState(false);
   449→  const saveTimerRef = React.useRef<ReturnType<typeof setTimeout> | null>(null);
   450→  const editModeRef = React.useRef(editMode);
   451→
   452→  React.useEffect(() => {
   453→    editModeRef.current = editMode;
   454→  }, [editMode]);
   455→
   456→  const layoutsSignature = React.useMemo(
   457→    () =>
   458→      active?.widgets.map((w) => `${w.id}:${JSON.stringify(w.layouts)}`).join("|") ??
   459→      "",
   460→    [active]
   461→  );
   462→
   463→  // Rebuild the grid from the server whenever the widget set / layouts change
   464→  // (add, remove, config refresh, optimistic autosave landing in the cache).
   465→  React.useEffect(() => {
   466→    savedLayoutsRef.current = snapshotLayouts(active?.widgets ?? []);
   467→    setGridState(null);
   468→  }, [layoutsSignature, active]);
   469→
   470→  const layouts = React.useMemo<ResponsiveLayouts<string>>(
   471→    () => gridState ?? buildRglLayouts(active?.widgets ?? []),
   472→    [gridState, active, layoutsSignature]
   473→  );
   474→
   475→  /** Debounced layout save — patches only the widgets that actually moved. */
   476→  const saveLayouts = React.useCallback(() => {
   477→    if (!active) return;
   478→    const all = gridsRef.current;
   479→    if (!all) return;
   480→    let saved = 0;
   481→    for (const w of active.widgets) {
   482→      const lg = all.lg?.find((l) => l.i === w.id);
   483→      if (!lg) continue;
   484→      const next: WidgetLayouts = {
   485→        lg: { x: lg.x, y: lg.y, w: lg.w, h: lg.h },
   486→      };
   487→      const md = all.md?.find((l) => l.i === w.id);
   488→      if (md) next.md = { x: md.x, y: md.y, w: md.w, h: md.h };
   489→      const sm = all.sm?.find((l) => l.i === w.id);
   490→      if (sm) next.sm = { x: sm.x, y: sm.y, w: sm.w, h: sm.h };
   491→      if (layoutsEqual(savedLayoutsRef.current[w.id], next)) continue;
   492→      savedLayoutsRef.current[w.id] = next;
   493→      saved += 1;
   494→      patchWidgetOptimistic.mutate({
   495→        dashboardId: active.id,
   496→        widgetId: w.id,
   497→        patch: { layouts: next },
   498→      });
   499→    }
   500→    if (saved > 0) {
   501→      setAnnouncement(t("edit.layoutSaved"));
   502→    }
   503→  }, [active, patchWidgetOptimistic, t]);
   504→
   505→  const scheduleSave = React.useCallback(() => {
   506→    if (saveTimerRef.current) clearTimeout(saveTimerRef.current);
   507→    saveTimerRef.current = setTimeout(() => {
   508→      saveLayouts();
   509→    }, SAVE_DEBOUNCE_MS);
   510→  }, [saveLayouts]);
   511→
   512→  React.useEffect(() => {
   513→    return () => {
   514→      if (saveTimerRef.current) clearTimeout(saveTimerRef.current);
   515→    };
   516→  }, []);
   517→
   518→  const snapshotForUndo = React.useCallback(() => {
   519→    undoSnapshotRef.current = { ...savedLayoutsRef.current };
   520→    setCanUndo(true);
   521→  }, []);
   522→
   523→  const onGridLayoutChange = (
   524→    _layout: readonly LayoutItem[],
   525→    allLayouts: ResponsiveLayouts<string>
   526→  ) => {
   527→    const normalized: ResponsiveLayouts<string> = {
   528→      lg: (allLayouts.lg ?? []).map(pickItem),
   529→      md: (allLayouts.md ?? []).map(pickItem),
   530→      sm: (allLayouts.sm ?? []).map(pickItem),
   531→    };
   532→    gridsRef.current = normalized;
   533→    setGridState((prev) =>
   534→      prev && JSON.stringify(prev) === JSON.stringify(normalized)
   535→        ? prev
   536→        : normalized
   537→    );
   538→    if (editModeRef.current) scheduleSave();
   539→  };
   540→
   541→  const onGridInteractionStart = React.useCallback(() => {
   542→    snapshotForUndo();
   543→  }, [snapshotForUndo]);
   544→
   545→  const undoLayouts = () => {
   546→    const snap = undoSnapshotRef.current;
   547→    if (!snap || !active) return;
   548→    for (const w of active.widgets) {
   549→      const restored = snap[w.id];
   550→      if (!restored) continue;
   551→      if (layoutsEqual(savedLayoutsRef.current[w.id], restored)) continue;
   552→      savedLayoutsRef.current[w.id] = restored;
   553→      patchWidgetOptimistic.mutate({
   554→        dashboardId: active.id,
   555→        widgetId: w.id,
   556→        patch: { layouts: restored },
   557→      });
   558→    }
   559→    undoSnapshotRef.current = null;
   560→    setCanUndo(false);
   561→    setAnnouncement(t("edit.undoDone"));
   562→  };
   563→
   564→  /** Restores every widget to its registry default footprint. */
