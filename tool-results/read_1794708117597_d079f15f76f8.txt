    93→function KpiCard({
    94→  label,
    95→  value,
    96→  valueClassName,
    97→  sub,
    98→  compact = false,
    99→}: {
   100→  label: string;
   101→  value: React.ReactNode;
   102→  valueClassName?: string;
   103→  sub?: React.ReactNode;
   104→  compact?: boolean;
   105→}) {
   106→  return (
   107→    <div className="flex flex-col gap-2 rounded-lg border p-4">
   108→      <p className="text-xs font-medium text-muted-foreground">{label}</p>
   109→      <p
   110→        className={`tabular-nums ${
   111→          compact ? "text-xl font-semibold" : "text-2xl font-bold"
   112→        } ${valueClassName ?? ""}`}
   113→      >
   114→        {value}
   115→      </p>
   116→      {sub != null ? (
   117→        <p className="text-xs tabular-nums text-muted-foreground">{sub}</p>
   118→      ) : null}
   119→    </div>
   120→  );
   121→}

   123→/** Recharts tooltip: date + equity (+ drawdown below peak, as money). */
   124→function EquityTooltipContent({
   125→  active,
   126→  payload,
   127→  currency,
   128→  timezone,
   129→}: {
   130→  active?: boolean;
   131→  payload?: Array<{ payload?: EquityDatum }>;
   132→  currency: string;
   133→  timezone: string;
   134→}) {
   135→  if (!active) return null;
   136→  const point = payload?.[0]?.payload;
   137→  if (!point) return null;
   138→  return (
   139→    <div className="rounded-lg border bg-background px-3 py-2 text-xs shadow-md">
   140→      <p className="font-medium">{fmtDateInTz(point.date, timezone)}</p>
   141→      <p className="mt-1 tabular-nums">{fmtMoney(point.equity, currency)}</p>
   142→      {point.drawdown > 0 ? (
   143→        <p className={`tabular-nums ${ROSE}`}>
   144→          −{fmtMoney(point.drawdown, currency)}
   145→        </p>
   146→      ) : null}
   147→    </div>
   148→  );
   149→}

   189→/** Risk cell for an open position: amount, else %, else em dash. */
   190→function openPositionRisk(p: OpenPositionDTO): string {
   191→  if (p.riskAmount) return fmtMoney(p.riskAmount, p.pnlCurrency);
   192→  if (p.riskPercent) return fmtPct(p.riskPercent);
   193→  return "—";
   194→}
